6   package dev.metaschema.cli.util;
7   
8   import static org.assertj.core.api.Assertions.assertThat;
9   import static org.junit.jupiter.api.Assertions.assertNotNull;
10  import static org.junit.jupiter.api.Assertions.assertSame;
11  
12  import org.junit.jupiter.api.DisplayName;
13  import org.junit.jupiter.api.Nested;
14  import org.junit.jupiter.api.Test;
15  
16  import dev.metaschema.core.metapath.format.IPathFormatter;
17  
18  /**
19   * Tests for {@link LoggingValidationHandler} factory methods and configuration.
20   * <p>
21   * These tests verify the handler creation API that must be preserved when
22   * migrating from Jansi to JLine. The actual ANSI output behavior is tested
23   * through integration tests in {@link dev.metaschema.cli.CLITest}.
24   */
25  class LoggingValidationHandlerTest {
26  
27    @Nested
28    @DisplayName("Factory Method Tests")
29    class FactoryMethodTests {
30  
31      @Test
32      @DisplayName("instance() returns non-null singleton")
33      void testInstanceReturnsNonNull() {
34        LoggingValidationHandler handler = LoggingValidationHandler.instance();
35        assertNotNull(handler);
36      }
37  
38      @Test
39      @DisplayName("instance() returns same instance on repeated calls")
40      void testInstanceReturnsSameInstance() {
41        LoggingValidationHandler first = LoggingValidationHandler.instance();
42        LoggingValidationHandler second = LoggingValidationHandler.instance();
43        assertSame(first, second, "instance() should return the same singleton");
44      }
45  
46      @Test
47      @DisplayName("instance() and instance(false) return the same singleton")
48      void testInstanceAndInstanceFalseReturnSameSingleton() {
49        LoggingValidationHandler defaultInstance = LoggingValidationHandler.instance();
50        LoggingValidationHandler noExceptionInstance = LoggingValidationHandler.instance(false);
51        assertSame(defaultInstance, noExceptionInstance,
52            "instance() and instance(false) should return the same singleton");
53      }
54  
55      @Test
56      @DisplayName("instance(false) returns handler that does not log exceptions")
57      void testInstanceWithoutExceptionLogging() {
58        LoggingValidationHandler handler = LoggingValidationHandler.instance(false);
59        assertNotNull(handler);
60        assertThat(handler.isLogExceptions()).isFalse();
61      }
62  
63      @Test
64      @DisplayName("instance(true) returns handler that logs exceptions")
65      void testInstanceWithExceptionLogging() {
66        LoggingValidationHandler handler = LoggingValidationHandler.instance(true);
67        assertNotNull(handler);
68        assertThat(handler.isLogExceptions()).isTrue();
69      }
70  
71      @Test
72      @DisplayName("instance(true) and instance(false) return different singletons")
73      void testInstanceWithDifferentExceptionSettings() {
74        LoggingValidationHandler withExceptions = LoggingValidationHandler.instance(true);
75        LoggingValidationHandler withoutExceptions = LoggingValidationHandler.instance(false);
76  
77        assertThat(withExceptions).isNotSameAs(withoutExceptions);
78        assertThat(withExceptions.isLogExceptions()).isTrue();
79        assertThat(withoutExceptions.isLogExceptions()).isFalse();
80      }
81  
82      @Test
83      @DisplayName("withPathFormatter creates handler with custom formatter")
84      void testWithPathFormatter() {
85        LoggingValidationHandler handler = LoggingValidationHandler.withPathFormatter(
86            IPathFormatter.METAPATH_PATH_FORMATER);
87        assertNotNull(handler);
88        // New instance should not log exceptions by default
89        assertThat(handler.isLogExceptions()).isFalse();
90      }
91  
92      @Test
93      @DisplayName("withSettings creates handler with custom settings - exceptions enabled")
94      void testWithSettingsExceptionsEnabled() {
95        LoggingValidationHandler handler = LoggingValidationHandler.withSettings(
96            true,
97            IPathFormatter.METAPATH_PATH_FORMATER);
98        assertNotNull(handler);
99        assertThat(handler.isLogExceptions()).isTrue();
100     }
101 
102     @Test
103     @DisplayName("withSettings creates handler with custom settings - exceptions disabled")
104     void testWithSettingsExceptionsDisabled() {
105       LoggingValidationHandler handler = LoggingValidationHandler.withSettings(
106           false,
107           IPathFormatter.METAPATH_PATH_FORMATER);
108       assertNotNull(handler);
109       assertThat(handler.isLogExceptions()).isFalse();
110     }
111   }
112 
113   @Nested
114   @DisplayName("Configuration Tests")
115   class ConfigurationTests {
116 
117     @Test
118     @DisplayName("default instance does not log exceptions")
119     void testDefaultInstanceDoesNotLogExceptions() {
120       LoggingValidationHandler handler = LoggingValidationHandler.instance();
121       assertThat(handler.isLogExceptions()).isFalse();
122     }
123 
124     @Test
125     @DisplayName("handler created with withPathFormatter uses default exception logging")
126     void testWithPathFormatterUsesDefaultExceptionLogging() {
127       LoggingValidationHandler handler = LoggingValidationHandler.withPathFormatter(
128           IPathFormatter.METAPATH_PATH_FORMATER);
129       // Default is to not log exceptions
130       assertThat(handler.isLogExceptions()).isFalse();
131     }
132   }
133 }
