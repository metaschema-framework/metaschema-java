7   
8   import static org.junit.jupiter.api.Assertions.assertAll;
9   import static org.junit.jupiter.api.Assertions.assertEquals;
10  import static org.junit.jupiter.api.Assertions.assertNotNull;
11  
12  import gov.nist.secauto.metaschema.core.model.IAssemblyDefinition;
13  import gov.nist.secauto.metaschema.core.model.IConstraintLoader;
14  import gov.nist.secauto.metaschema.core.model.MetaschemaException;
15  import gov.nist.secauto.metaschema.core.model.constraint.IConstraintSet;
16  import gov.nist.secauto.metaschema.core.qname.IEnhancedQName;
17  import gov.nist.secauto.metaschema.databind.IBindingContext;
18  import gov.nist.secauto.metaschema.databind.model.IBoundModule;
19  
20  import org.junit.jupiter.api.Test;
21  
22  import java.io.IOException;
23  import java.nio.file.Files;
24  import java.nio.file.Path;
25  import java.nio.file.Paths;
26  import java.util.List;
27  
28  class BindingConstraintLoaderTest {
29    private static final String NS = "http://csrc.nist.gov/ns/test/metaschema/meta-constraints";
30  
31    @Test
32    void testValid() throws MetaschemaException, IOException {
33      IBindingContext bindingContext = IBindingContext.newInstance();
34      IConstraintLoader loader = new BindingConstraintLoader(bindingContext);
35  
36      List<IConstraintSet> constraints = loader.load(
37          Paths.get("src/test/resources/content/constraints/meta-constraints/meta-constraints-valid.yaml"));
38      assertEquals(1, constraints.size());
39  
40      Path compileDir = Paths.get("target/generated-test-modules/meta-constraints/");
41      Files.createDirectories(compileDir);
42  
43      bindingContext = IBindingContext.builder()
44          .compilePath(compileDir)
45          .constraintSet(constraints)
46          .build();
47  
48      IBindingMetaschemaModule metaschema = bindingContext.loadMetaschema(
49          Paths.get("src/test/resources/content/constraints/meta-constraints/metaschema.xml"));
50      IBoundModule module = bindingContext.registerModule(metaschema);
51  
52      final IAssemblyDefinition level1
53          = module.getAssemblyDefinitionByName(IEnhancedQName.of(NS, "level1").getIndexPosition());
54      final IAssemblyDefinition level2
55          = module.getAssemblyDefinitionByName(IEnhancedQName.of(NS, "level2").getIndexPosition());
56      final IAssemblyDefinition level3
57          = module.getAssemblyDefinitionByName(IEnhancedQName.of(NS, "level3").getIndexPosition());
58  
59      assert level1 != null;
60      assert level2 != null;
61      assert level3 != null;
62  
63      level1.getConstraints();
64      level2.getConstraints();
65      level3.getConstraints();
66  
67      assertAll(
68          () -> assertNotNull(level1),
69          () -> assertEquals(1, level1 == null ? 0 : level1.getLetExpressions().size(), "level 1 let"),
70          () -> assertEquals(1, level1 == null ? 0 : level1.getExpectConstraints().size(), "level 1 expect"),
71          () -> assertNotNull(level2),
72          () -> assertEquals(1, level2 == null ? 0 : level2.getLetExpressions().size(), "level 2 let"),
73          () -> assertEquals(1, level2 == null ? 0 : level2.getExpectConstraints().size(), "level 2 expect"),
74          () -> assertNotNull(level3),
75          () -> assertEquals(1, level3 == null ? 0 : level3.getLetExpressions().size(), "level 3 let"),
76          () -> assertEquals(1, level3 == null ? 0 : level3.getExpectConstraints().size(), "level 3 expect"));
77    }
78  }
