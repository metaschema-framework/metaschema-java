6   package dev.metaschema.databind.io.xml;
7   
8   import static org.junit.jupiter.api.Assertions.assertEquals;
9   import static org.junit.jupiter.api.Assertions.assertFalse;
10  import static org.junit.jupiter.api.Assertions.assertThrows;
11  import static org.junit.jupiter.api.Assertions.assertTrue;
12  
13  import org.junit.jupiter.api.Test;
14  import org.w3c.dom.Document;
15  import org.w3c.dom.Element;
16  
17  import java.util.ArrayList;
18  import java.util.Collections;
19  import java.util.List;
20  
21  import javax.xml.parsers.DocumentBuilderFactory;
22  import javax.xml.parsers.ParserConfigurationException;
23  
24  class XmlAnyContentTest {
25  
26    private static Element createElement(String name) throws ParserConfigurationException {
27      DocumentBuilderFactory factory = DocumentBuilderFactory.newInstance();
28      Document doc = factory.newDocumentBuilder().newDocument();
29      return doc.createElement(name);
30    }
31  
32    @Test
33    void testEmptyListIsEmpty() {
34      XmlAnyContent content = new XmlAnyContent(Collections.emptyList());
35      assertTrue(content.isEmpty(), "isEmpty() should return true for empty list");
36    }
37  
38    @Test
39    void testEmptyListGetElements() {
40      XmlAnyContent content = new XmlAnyContent(Collections.emptyList());
41      assertTrue(content.getElements().isEmpty(), "getElements() should return empty list");
42    }
43  
44    @Test
45    void testNonEmptyListIsNotEmpty() throws ParserConfigurationException {
46      Element elem = createElement("test");
47      XmlAnyContent content = new XmlAnyContent(List.of(elem));
48      assertFalse(content.isEmpty(), "isEmpty() should return false for non-empty list");
49    }
50  
51    @Test
52    void testNonEmptyListGetElements() throws ParserConfigurationException {
53      Element elem1 = createElement("first");
54      Element elem2 = createElement("second");
55      List<Element> elements = List.of(elem1, elem2);
56  
57      XmlAnyContent content = new XmlAnyContent(elements);
58      List<Element> result = content.getElements();
59  
60      assertEquals(2, result.size(), "getElements() should return all elements");
61      assertEquals(elem1, result.get(0), "First element should match");
62      assertEquals(elem2, result.get(1), "Second element should match");
63    }
64  
65    @Test
66    void testReturnedListIsUnmodifiable() throws ParserConfigurationException {
67      Element elem = createElement("test");
68      List<Element> mutableList = new ArrayList<>();
69      mutableList.add(elem);
70  
71      XmlAnyContent content = new XmlAnyContent(mutableList);
72      List<Element> returned = content.getElements();
73  
74      assertThrows(UnsupportedOperationException.class, () -> returned.add(createElement("extra")),
75          "Returned list should be unmodifiable");
76    }
77  
78    @Test
79    void testDefensiveCopyFromMutableInput() throws ParserConfigurationException {
80      Element elem = createElement("original");
81      List<Element> mutableList = new ArrayList<>();
82      mutableList.add(elem);
83  
84      XmlAnyContent content = new XmlAnyContent(mutableList);
85  
86      // Modify the original list after construction
87      mutableList.add(createElement("added-after"));
88  
89      assertEquals(1, content.getElements().size(),
90          "XmlAnyContent should not be affected by changes to the original list");
91    }
92  }
