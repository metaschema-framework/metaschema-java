6   package dev.metaschema.databind.io.json;
7   
8   import static org.junit.jupiter.api.Assertions.assertNotNull;
9   
10  import org.junit.jupiter.api.Test;
11  
12  import java.io.IOException;
13  import java.nio.file.Paths;
14  
15  import dev.metaschema.core.model.MetaschemaException;
16  import dev.metaschema.core.util.ObjectUtils;
17  import dev.metaschema.databind.IBindingContext;
18  import dev.metaschema.databind.codegen.AbstractMetaschemaTest;
19  import dev.metaschema.databind.io.DeserializationFeature;
20  import dev.metaschema.databind.io.IBoundLoader;
21  
22  class JsonParserTest
23      extends AbstractMetaschemaTest {
24    @Test
25    void testIssue308Regression() throws IOException, MetaschemaException {
26      IBindingContext bindingContext = newBindingContext();
27  
28      bindingContext.loadMetaschema(ObjectUtils.notNull(
29          Paths.get("src/test/resources/metaschema/308-choice-regression/metaschema.xml")));
30  
31      IBoundLoader loader = bindingContext.newBoundLoader();
32      loader.enableFeature(DeserializationFeature.DESERIALIZE_VALIDATE_CONSTRAINTS);
33      // Disable required field validation because dynamically compiled binding
34      // classes
35      // don't preserve choice group information (see issue #594). The metaschema has
36      // a
37      // choice between x and y, and the example provides y, which should satisfy the
38      // choice.
39      loader.disableFeature(DeserializationFeature.DESERIALIZE_VALIDATE_REQUIRED_FIELDS);
40      Object obj = loader.load(ObjectUtils.notNull(
41          Paths.get("src/test/resources/metaschema/308-choice-regression/example.json")));
42      assertNotNull(obj);
43    }
44  }
