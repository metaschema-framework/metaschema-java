52    @Test
53    void testLoader() throws MalformedURLException, IOException, BindingException {
54      File bindingConfigFile = new File("src/test/resources/metaschema/binding-config.xml");
55  
56      DefaultBindingConfiguration config = new DefaultBindingConfiguration();
57      config.load(bindingConfigFile);
58  
59      assertEquals("gov.nist.itl.metaschema.codegen.xml.example.assembly",
60          config.getPackageNameForNamespace("http://csrc.nist.gov/ns/metaschema/testing/assembly"));
61  
62      context.checking(new Expectations() {
63        { // NOPMD - intentional
64          oneOf(module).getLocation();
65          will(returnValue(METASCHEMA_LOCATION));
66          allowing(definition).getContainingModule();
67          will(returnValue(module));
68          allowing(definition).getModelType();
69          will(returnValue(DEFINITION_MODEL_TYPE));
70          allowing(definition).getName();
71          will(returnValue(DEFINITION_NAME));
72        }
73      });
74      IDefinitionBindingConfiguration defConfig = config.getBindingConfigurationForDefinition(
75          ObjectUtils.notNull(definition));
76      assertNotNull(defConfig);
77      assertEquals(DEFINITION__CLASS_NAME, defConfig.getClassName());
78    }

81    void testCollectionClassOverride() throws IOException, BindingException {
82      // Test loading binding configuration with collection-class overrides
83      File bindingConfigFile = new File("src/test/resources/metaschema/binding-config-with-collection-class.xml");
84      URI assemblyMetaschemaLocation = new File("src/test/resources/metaschema/assembly/metaschema.xml")
85          .getAbsoluteFile().toURI();
86  
87      DefaultBindingConfiguration config = new DefaultBindingConfiguration();
88      config.load(bindingConfigFile);
89  
90      // Create mock for the top-level assembly definition
91      IAssemblyDefinition topLevelDefinition = context.mock(IAssemblyDefinition.class, "topLevel");
92      IModule assemblyModule = context.mock(IModule.class, "assemblyModule");
93      INamedModelInstanceAbsolute childrenInstance = context.mock(INamedModelInstanceAbsolute.class, "childrenInstance");
94  
95      context.checking(new Expectations() {
96        {
97          // Module expectations
98          allowing(assemblyModule).getLocation();
99          will(returnValue(assemblyMetaschemaLocation));
100 
101         // Definition expectations
102         allowing(topLevelDefinition).getContainingModule();
103         will(returnValue(assemblyModule));
104         allowing(topLevelDefinition).getModelType();
105         will(returnValue(ModelType.ASSEMBLY));
106         allowing(topLevelDefinition).getName();
107         will(returnValue("top-level"));
108 
109         // Instance expectations
110         allowing(childrenInstance).getContainingDefinition();
111         will(returnValue(topLevelDefinition));
112         allowing(childrenInstance).getName();
113         will(returnValue("children"));
114       }
115     });
116 
117     // Get the property binding configuration
118     IPropertyBindingConfiguration propertyConfig = config.getPropertyBindingConfiguration(
119         ObjectUtils.notNull(topLevelDefinition),
120         "children");
121 
122     // Verify collection class override is available
123     assertNotNull(propertyConfig, "Property binding configuration should exist for 'children'");
124     assertEquals(ArrayList.class.getName(), propertyConfig.getCollectionClassName(),
125         "Collection class should be ArrayList");
126 
127     // Test grandchild's fields property

129

131       {

134         allowing(grandchildDefinition).getModelType();
135         will(returnValue(ModelType.ASSEMBLY));
136         allowing(grandchildDefinition).getName();
137         will(returnValue("grandchild"));
138       }
139     });
140 
141     IPropertyBindingConfiguration fieldsPropertyConfig = config.getPropertyBindingConfiguration(
142         ObjectUtils.notNull(grandchildDefinition),
143         "fields");
144 
145     assertNotNull(fieldsPropertyConfig, "Property binding configuration should exist for 'fields'");
146     assertEquals(CopyOnWriteArrayList.class.getName(), fieldsPropertyConfig.getCollectionClassName(),
147         "Collection class should be CopyOnWriteArrayList");
148 
149     // Test that unconfigured property returns null
150     IPropertyBindingConfiguration unconfiguredProperty = config.getPropertyBindingConfiguration(
151         ObjectUtils.notNull(topLevelDefinition),
152         "nonexistent");
153     assertNull(unconfiguredProperty, "Unconfigured property should return null");

264   void testDuplicatePropertyBindingLastWins() throws IOException, BindingException {
265     // Test that duplicate property bindings use last-wins semantics
266     File bindingConfigFile = new File("src/test/resources/metaschema/binding-config-edge-cases.xml");
267     URI assemblyMetaschemaLocation = new File("src/test/resources/metaschema/assembly/metaschema.xml")
268         .getAbsoluteFile().toURI();
269 
270     DefaultBindingConfiguration config = new DefaultBindingConfiguration();
271     config.load(bindingConfigFile);
272 
273     IAssemblyDefinition duplicateDefinition = context.mock(IAssemblyDefinition.class, "duplicateDef");
274     IModule duplicateModule = context.mock(IModule.class, "duplicateModule");
275 
276     context.checking(new Expectations() {
277       {
278         allowing(duplicateModule).getLocation();
279         will(returnValue(assemblyMetaschemaLocation));
280         allowing(duplicateDefinition).getContainingModule();
281         will(returnValue(duplicateModule));
282         allowing(duplicateDefinition).getModelType();
283         will(returnValue(ModelType.ASSEMBLY));
284         allowing(duplicateDefinition).getName();
285         will(returnValue("edge-case-assembly"));
286       }
287     });
288 
289     // When duplicate property bindings exist, the last one should win
290     // The config has ArrayList first, then LinkedList - LinkedList should be
291     // returned
292     IPropertyBindingConfiguration duplicateConfig = config.getPropertyBindingConfiguration(
293         ObjectUtils.notNull(duplicateDefinition),
294         "duplicate-property");
295     assertNotNull(duplicateConfig, "Duplicate property binding should return a configuration");
296     assertEquals(LinkedList.class.getName(), duplicateConfig.getCollectionClassName(),
297         "Duplicate property bindings should use last-wins semantics");
298   }

301   void testChoiceGroupBindingParsing() throws IOException, BindingException {
302     // Test that choice-group-binding elements are parsed from XML config
303     File bindingConfigFile = new File("src/test/resources/metaschema/binding-config-with-choice-groups.xml");
304     URI assemblyMetaschemaLocation = new File("src/test/resources/metaschema/assembly/metaschema.xml")
305         .getAbsoluteFile().toURI();
306 
307     DefaultBindingConfiguration config = new DefaultBindingConfiguration();
308     config.load(bindingConfigFile);
309 
310     IAssemblyDefinition testAssemblyDefinition = context.mock(IAssemblyDefinition.class, "testAssembly");
311     IModule testModule = context.mock(IModule.class, "testModule");
312 
313     context.checking(new Expectations() {
314       {
315         allowing(testModule).getLocation();
316         will(returnValue(assemblyMetaschemaLocation));
317         allowing(testAssemblyDefinition).getContainingModule();
318         will(returnValue(testModule));
319         allowing(testAssemblyDefinition).getModelType();
320         will(returnValue(ModelType.ASSEMBLY));
321         allowing(testAssemblyDefinition).getName();
322         will(returnValue("test-assembly"));
323       }
324     });
325 
326     // Get the definition binding configuration
327     IDefinitionBindingConfiguration defConfig = config.getBindingConfigurationForDefinition(
328         ObjectUtils.notNull(testAssemblyDefinition));
329 
330     assertNotNull(defConfig, "Definition binding configuration should exist");
331 
332     // Verify choice group bindings are accessible
333     assertNotNull(defConfig.getChoiceGroupBindings(),
334         "Choice group bindings map should not be null");
335     assertEquals(3, defConfig.getChoiceGroupBindings().size(),
336         "Should have 3 choice group bindings");
337 
338     // Verify "mixed-content" choice group binding
339     IChoiceGroupBindingConfiguration mixedContentConfig
340         = defConfig.getChoiceGroupBindings().get("mixed-content");
341     assertNotNull(mixedContentConfig, "mixed-content choice group binding should exist");
342     assertEquals("mixed-content", mixedContentConfig.getGroupAsName());
343     assertEquals("dev.metaschema.core.model.IModelElement",
344         mixedContentConfig.getItemTypeName());
345     assertEquals(true, mixedContentConfig.isUseWildcard(),
346         "mixed-content should use wildcard");
347 
348     // Verify "typed-items" choice group binding
349     IChoiceGroupBindingConfiguration typedItemsConfig
350         = defConfig.getChoiceGroupBindings().get("typed-items");
351     assertNotNull(typedItemsConfig, "typed-items choice group binding should exist");
352     assertEquals("typed-items", typedItemsConfig.getGroupAsName());
353     assertEquals("java.lang.String", typedItemsConfig.getItemTypeName());
354     assertEquals(false, typedItemsConfig.isUseWildcard(),
355         "typed-items should not use wildcard");
356 
357     // Verify "untyped-items" choice group binding
358     IChoiceGroupBindingConfiguration untypedItemsConfig
359         = defConfig.getChoiceGroupBindings().get("untyped-items");
360     assertNotNull(untypedItemsConfig, "untyped-items choice group binding should exist");
361     assertEquals("untyped-items", untypedItemsConfig.getGroupAsName());
362     assertNull(untypedItemsConfig.getItemTypeName(),
363         "untyped-items should not have an item type");
364   }
