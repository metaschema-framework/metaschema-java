6   package dev.metaschema.databind.codegen;
7   
8   import static org.junit.jupiter.api.Assertions.assertEquals;
9   import static org.junit.jupiter.api.Assertions.assertNotNull;
10  import static org.junit.jupiter.api.Assertions.assertTrue;
11  
12  import org.junit.jupiter.api.Test;
13  
14  import java.io.IOException;
15  import java.lang.reflect.Field;
16  import java.lang.reflect.Method;
17  import java.nio.file.Paths;
18  
19  import dev.metaschema.core.model.IAnyContent;
20  import dev.metaschema.core.model.IBoundObject;
21  import dev.metaschema.core.model.MetaschemaException;
22  import dev.metaschema.core.util.ObjectUtils;
23  import dev.metaschema.databind.io.BindingException;
24  import dev.metaschema.databind.model.annotations.BoundAny;
25  
26  /**
27   * Tests that the code generator properly emits {@link BoundAny}-annotated
28   * fields for assemblies containing {@code <any/>} in their model.
29   */
30  class AnyCodegenTest
31      extends AbstractMetaschemaTest {
32  
33    @Test
34    void testAnyFieldGenerated()
35        throws MetaschemaException, IOException, ClassNotFoundException, BindingException,
36        NoSuchMethodException {
37      Class<? extends IBoundObject> rootClass = compileModule(
38          ObjectUtils.notNull(Paths.get("src/test/resources/metaschema/any/metaschema.xml")),
39          null,
40          "com.example.ns.any_test.Root",
41          ObjectUtils.notNull(generationDir));
42  
43      // Verify a field annotated with @BoundAny exists
44      Field anyField = findBoundAnyField(rootClass);
45      assertNotNull(anyField, "Generated class should have a field annotated with @BoundAny");
46  
47      // Verify the field type is IAnyContent
48      assertEquals(IAnyContent.class, anyField.getType(),
49          "@BoundAny field should be of type IAnyContent");
50  
51      // Verify getter exists and returns IAnyContent
52      Method getter = rootClass.getMethod("getAny");
53      assertNotNull(getter, "Generated class should have getAny() method");
54      assertTrue(IAnyContent.class.isAssignableFrom(getter.getReturnType()),
55          "getAny() should return IAnyContent");
56  
57      // Verify setter exists and accepts IAnyContent
58      Method setter = rootClass.getMethod("setAny", IAnyContent.class);
59      assertNotNull(setter, "Generated class should have setAny(IAnyContent) method");
60    }
61  
62    /**
63     * Find the field annotated with {@link BoundAny} in the given class.
64     *
65     * @param clazz
66     *          the class to search
67     * @return the field, or {@code null} if not found
68     */
69    private static Field findBoundAnyField(Class<?> clazz) {
70      for (Field field : clazz.getDeclaredFields()) {
71        if (field.isAnnotationPresent(BoundAny.class)) {
72          return field;
73        }
74      }
75      return null;
76    }
77  }
