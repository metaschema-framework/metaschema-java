50    @BeforeEach

56    /**
57     * Generates code and returns the content of the generated Java file.
58     *
59     * @param metaschemaPath
60     *          the path to the Metaschema module
61     * @param className
62     *          the simple name of the class to find
63     * @return the content of the generated Java file
64     * @throws Exception
65     *           if generation fails
66     */
67    @NonNull
68    private String generateAndReadClass(@NonNull Path metaschemaPath, @NonNull String className) throws Exception {
69      IBindingContext context = newBindingContext();
70      IModule module = context.loadMetaschema(metaschemaPath);
71  
72      DefaultBindingConfiguration bindingConfiguration = new DefaultBindingConfiguration();
73      ModuleCompilerHelper.compileModule(module, classDir, bindingConfiguration);
74  
75      // Find the generated Java file
76      Path generatedFile = findGeneratedFile(classDir, className + ".java");
77      return ObjectUtils.notNull(Files.readString(generatedFile));
78    }
79  
80    /**
81     * Finds a generated file by name within the given directory tree.
82     *
83     * @param dir
84     *          the root directory to search
85     * @param fileName
86     *          the name of the file to find
87     * @return the path to the found file
88     * @throws IOException
89     *           if the file cannot be found or an I/O error occurs
90     */
91    @SuppressWarnings("resource")
92    @NonNull
93    private static Path findGeneratedFile(@NonNull Path dir, @NonNull String fileName) throws IOException {
94      return ObjectUtils.notNull(Files.walk(dir)
95          .filter(p -> p.getFileName().toString().equals(fileName))
96          .findFirst()
97          .orElseThrow(() -> new IOException("Could not find generated file: " + fileName)));
98    }

100   @Test
101   void testFieldJavadocDoesNotContainQuotes() throws Exception {
102     String content = generateAndReadClass(TEST_METASCHEMA, TEST_CLASS_NAME);
103 
104     // Pattern to find Javadoc comments that start with a quote
105     // Good: /** Some description */
106     // Bad: /** "Some description" */ (single-line or multi-line)
107     // Match both single-line and multi-line Javadoc with leading quotes
108     Pattern quotedJavadoc = Pattern.compile("/\\*\\*\\s*(\\n\\s*\\*\\s*)?\"[^\"]+\"");
109 
110     assertFalse(quotedJavadoc.matcher(content).find(),
111         "Generated Javadoc should not contain quoted descriptions. Found in:\n" + content);
112   }

132   @Test
133   void testGetterHasReturnTag() throws Exception {
134     String content = generateAndReadClass(TEST_METASCHEMA, TEST_CLASS_NAME);
135 
136     // Check that getter methods have @return tag
137     // This assumes the metaschema has at least one field that generates a getter
138     // Use DOTALL and allow for @Nullable annotation between Javadoc and method
139     Pattern getterWithReturn
140         = Pattern.compile("@return.*?\\*/\\s*(@Nullable\\s+)?public\\s+\\w+\\s+get", Pattern.DOTALL);
141 
142     assertTrue(getterWithReturn.matcher(content).find(),
143         "Getter methods should have @return tag. Content:\n" + content);
144   }

223   /**
224    * Tests that a class-level Javadoc is generated even when the metaschema
225    * definition only has a formal-name and no description.
226    *
227    * @throws Exception
228    *           if generation fails
229    */
230   @Test
231   void testClassHasJavadocWithOnlyFormalName() throws Exception {
232     Path noDescMetaschema = ObjectUtils.notNull(
233         Paths.get("src/test/resources/metaschema/no_description/metaschema.xml"));
234     String content = generateAndReadClass(noDescMetaschema, "ItemWithNameOnly");
235 
236     // The class should have a Javadoc comment derived from the formal-name
237     // Pattern: /** followed by content then */ before the class declaration
238     Pattern classJavadoc = Pattern.compile("/\\*\\*.*?\\*/\\s*@MetaschemaAssembly", Pattern.DOTALL);
239 
240     assertTrue(classJavadoc.matcher(content).find(),
241         "Class with only formal-name should have class-level Javadoc. Content:\n" + content);
242   }
