143   @Test
144   void testValidChoiceInstancesCreated() {
145     IBindingContext context = IBindingContext.newInstance();
146     IBoundDefinitionModelAssembly definition = (IBoundDefinitionModelAssembly) context
147         .getBoundDefinitionForClass(ValidChoiceAssembly.class);
148 
149     assertNotNull(definition, "Definition should not be null");
150 
151     List<IChoiceInstance> choices = definition.getChoiceInstances();
152     assertNotNull(choices, "Choice instances should not be null");
153     assertEquals(1, choices.size(), "Should have exactly one choice instance");
154 
155     IChoiceInstance choice = choices.get(0);
156     assertNotNull(choice, "Choice instance should not be null");
157     assertEquals(2, choice.getNamedModelInstances().size(),
158         "Choice should have 2 alternatives");
159   }

161   @Test
162   void testInvalidNonAdjacentChoiceThrowsException() {
163     IBindingContext context = IBindingContext.newInstance();
164 
165     // Attempting to get definition for class with non-adjacent choice fields
166     // should throw IllegalStateException when model is accessed
167     IBoundDefinitionModelAssembly definition = (IBoundDefinitionModelAssembly) context
168         .getBoundDefinitionForClass(InvalidChoiceAssembly.class);
169     assertNotNull(definition, "Definition should not be null before accessing model");
170 
171     assertThrows(IllegalStateException.class, () -> {
172       // Force model initialization by accessing the choice instances
173       definition.getChoiceInstances();
174     }, "Should throw exception for non-adjacent choice fields");
175   }

177   @Test
178   void testMultipleChoiceGroups() {
179     IBindingContext context = IBindingContext.newInstance();
180     IBoundDefinitionModelAssembly definition = (IBoundDefinitionModelAssembly) context
181         .getBoundDefinitionForClass(MultiChoiceAssembly.class);
182 
183     assertNotNull(definition, "Definition should not be null");
184 
185     List<IChoiceInstance> choices = definition.getChoiceInstances();
186     assertNotNull(choices, "Choice instances should not be null");
187     assertEquals(2, choices.size(), "Should have exactly two choice instances");
188 
189     // Each choice should have 2 alternatives
190     for (IChoiceInstance choice : choices) {
191       assertEquals(2, choice.getNamedModelInstances().size(),
192           "Each choice should have 2 alternatives");
193     }
194   }

196   @Test
197   void testChoiceInstanceProperties() {
198     IBindingContext context = IBindingContext.newInstance();
199     IBoundDefinitionModelAssembly definition = (IBoundDefinitionModelAssembly) context
200         .getBoundDefinitionForClass(ValidChoiceAssembly.class);
201     assertNotNull(definition, "Definition should not be null");
202 
203     List<IChoiceInstance> choices = definition.getChoiceInstances();
204     IChoiceInstance choice = choices.get(0);
205 
206     // Verify IChoiceInstance properties
207     // Annotation-based bindings default to optional choices (minOccurs = 0)
208     assertEquals(0, choice.getMinOccurs(), "minOccurs should be 0 for optional choice");
209     assertEquals(1, choice.getMaxOccurs(), "maxOccurs should be 1");
210     assertNotNull(choice.getContainingDefinition(), "Containing definition should not be null");
211     assertEquals(definition, choice.getContainingDefinition(),
212         "Containing definition should be the parent assembly");
213 
214     // Verify the choice contains the expected fields
215     assertTrue(choice.getFieldInstances().size() > 0 || choice.getAssemblyInstances().size() > 0,
216         "Choice should contain field or assembly instances");
217   }
