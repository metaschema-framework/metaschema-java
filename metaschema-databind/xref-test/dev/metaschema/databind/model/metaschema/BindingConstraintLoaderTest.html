153   @Test
154   void testTraceReportConstraintLoading() throws IOException {
155     StringBuilder trace = new StringBuilder();
156     trace.append("=== Tracing Report Constraint Loading ===\n\n");
157 
158     // Step 1: Parse YAML directly to binding object
159     trace.append("STEP 1: Parse YAML to binding object\n");
160     IBindingContext bindingContext = IBindingContext.newInstance();
161     IBoundLoader loader = bindingContext.newBoundLoader();
162     URI resourceUri = ObjectUtils
163         .notNull(Paths.get("src/test/resources/content/constraints/meta-constraints/meta-constraints-report.yaml")
164             .toUri());
165 
166     Object constraintsDocument = loader.load(resourceUri);
167     trace.append("  Document type: ").append(constraintsDocument.getClass().getName()).append("\n");
168     assertEquals(MetaschemaMetaConstraints.class, constraintsDocument.getClass(),
169         "Should parse to MetaschemaMetaConstraints");
170 
171     MetaschemaMetaConstraints metaConstraints = (MetaschemaMetaConstraints) constraintsDocument;
172 
173     // Step 2: Check contexts
174     trace.append("\nSTEP 2: Check parsed contexts\n");
175     List<dev.metaschema.databind.model.metaschema.binding.MetapathContext> contexts
176         = metaConstraints.getContexts();
177     trace.append("  Number of contexts: ").append(contexts.size()).append("\n");
178     assertFalse(contexts.isEmpty(), "Should have at least one context");
179 
180     // Step 3: Check first context's constraints
181     trace.append("\nSTEP 3: Check first context's constraints\n");
182     dev.metaschema.databind.model.metaschema.binding.MetapathContext firstContext = contexts.get(0);
183     trace.append("  First context metapaths: ");
184     for (var mp : firstContext.getMetapaths()) {
185       trace.append(mp.getTarget()).append(" ");
186     }
187     trace.append("\n");
188 
189     AssemblyConstraints assemblyConstraints = firstContext.getConstraints();
190     trace.append("  Constraints object: ").append(assemblyConstraints == null ? "null" : "present").append("\n");
191     assertNotNull(assemblyConstraints, "Constraints should not be null");
192 
193     // Step 4: Check rules in constraints
194     trace.append("\nSTEP 4: Check rules in AssemblyConstraints\n");
195     List<? extends ITargetedConstraintBase> rules = assemblyConstraints.getRules();
196     trace.append("  Number of rules: ").append(rules.size()).append("\n");
197 
198     if (!rules.isEmpty()) {
199       for (int i = 0; i < rules.size(); i++) {
200         Object rule = rules.get(i);
201         trace.append("  Rule ").append(i).append(": ");
202         if (rule == null) {
203           trace.append("null\n");
204         } else {
205           trace.append(rule.getClass().getName()).append("\n");
206           if (rule instanceof TargetedReportConstraint) {
207             TargetedReportConstraint report = (TargetedReportConstraint) rule;
208             trace.append("    ID: ").append(report.getId()).append("\n");
209             trace.append("    Test: ").append(report.getTest()).append("\n");
210             trace.append("    Target: ").append(report.getTarget()).append("\n");
211           } else if (rule instanceof IConstraintBase) {
212             IConstraintBase constraint = (IConstraintBase) rule;
213             trace.append("    ID: ").append(constraint.getId()).append("\n");
214           }
215         }
216       }
217     } else {
218       trace.append("  WARNING: No rules found!\n");
219     }
220 
221     // Step 5: Check if TargetedReportConstraint is in the rules
222     trace.append("\nSTEP 5: Look for TargetedReportConstraint\n");
223     boolean foundReport = false;
224     for (Object rule : rules) {
225       if (rule instanceof TargetedReportConstraint) {
226         foundReport = true;
227         trace.append("  FOUND TargetedReportConstraint!\n");
228         break;
229       }
230     }
231     if (!foundReport) {
232       trace.append("  NOT FOUND - TargetedReportConstraint missing from rules\n");
233     }
234 
235     // Output trace for debugging
236     System.out.println(trace);
237 
238     // The assertion that matters
239     assertEquals(1, rules.size(), "Should have exactly one rule. Trace:\n" + trace);
240     assertEquals(TargetedReportConstraint.class, rules.get(0).getClass(),
241         "Rule should be TargetedReportConstraint. Trace:\n" + trace);
242   }

250   void testConstraintBindingSupportParsesReportConstraint() throws IOException {
251     StringBuilder trace = new StringBuilder();
252     trace.append("=== Testing ConstraintBindingSupport.parse() ===\n\n");
253 
254     // Step 1: Parse YAML directly to binding object
255     IBindingContext bindingContext = IBindingContext.newInstance();
256     IBoundLoader loader = bindingContext.newBoundLoader();
257     URI resourceUri = ObjectUtils
258         .notNull(Paths.get("src/test/resources/content/constraints/meta-constraints/meta-constraints-report.yaml")
259             .toUri());
260 
261     MetaschemaMetaConstraints metaConstraints = (MetaschemaMetaConstraints) loader.load(resourceUri);
262     var contexts = metaConstraints.getContexts();
263     assertFalse(contexts.isEmpty(), "Should have at least one context");
264

268 
269     trace.append("YAML parsed successfully. Rules count: ").append(assemblyConstraints.getRules().size()).append("\n");
270 
271     // Step 2: Create AssemblyConstraintSet and parse constraints into it
272     StaticContext staticContext = StaticContext.builder()
273         .baseUri(resourceUri)
274         .useWildcardWhenNamespaceNotDefaulted(true)
275         .build();
276     ISource source = ISource.externalSource(staticContext, false);
277 
278     AssemblyConstraintSet constraintSet = new AssemblyConstraintSet(source);
279 
280     trace.append("Before ConstraintBindingSupport.parse():\n");
281     trace.append("  expect constraints: ").append(constraintSet.getExpectConstraints().size()).append("\n");
282     trace.append("  report constraints: ").append(constraintSet.getReportConstraints().size()).append("\n");
283     trace.append("  all constraints: ").append(constraintSet.getConstraints().size()).append("\n");
284 
285     // Parse using ConstraintBindingSupport
286     ConstraintBindingSupport.parse(constraintSet, assemblyConstraints, source);
287 
288     trace.append("\nAfter ConstraintBindingSupport.parse():\n");
289     trace.append("  expect constraints: ").append(constraintSet.getExpectConstraints().size()).append("\n");
290     trace.append("  report constraints: ").append(constraintSet.getReportConstraints().size()).append("\n");
291     trace.append("  all constraints: ").append(constraintSet.getConstraints().size()).append("\n");
292 
293     // List all constraints with their types
294     for (IConstraint c : constraintSet.getConstraints()) {
295       trace.append("    - ").append(c.getClass().getSimpleName()).append(": id=").append(c.getId()).append("\n");
296     }
297 
298     // Report constraints specifically
299     trace.append("\nReport constraints details:\n");
300     for (IReportConstraint rc : constraintSet.getReportConstraints()) {
301       trace.append("    - id=").append(rc.getId())
302           .append(", test=").append(rc.getTest().getPath())
303           .append(", target=").append(rc.getTarget().getPath()).append("\n");
304     }
305 
306     System.out.println(trace);
307 
308     // Assertions
309     assertEquals(1, constraintSet.getReportConstraints().size(),
310         "Should have exactly one report constraint. Trace:\n" + trace);
311     assertEquals("level1-report", constraintSet.getReportConstraints().get(0).getId(),
312         "Report constraint should have correct id");
313   }
