6   package dev.metaschema.databind.model.metaschema;
7   
8   import static org.junit.jupiter.api.Assertions.assertInstanceOf;
9   import static org.junit.jupiter.api.Assertions.assertNotNull;
10  
11  import org.junit.jupiter.api.Test;
12  
13  import java.io.IOException;
14  import java.nio.file.Paths;
15  
16  import dev.metaschema.core.model.IAssemblyDefinition;
17  import dev.metaschema.core.model.IModule;
18  import dev.metaschema.core.model.MetaschemaException;
19  import dev.metaschema.core.model.util.ModuleUtils;
20  import dev.metaschema.core.util.ObjectUtils;
21  import dev.metaschema.databind.codegen.AbstractMetaschemaTest;
22  import dev.metaschema.databind.model.metaschema.impl.DefinitionAssemblyGlobal;
23  
24  class AnyInstanceLoadingTest
25      extends AbstractMetaschemaTest {
26  
27    @Test
28    void testAssemblyWithAnyInstanceIsLoaded() throws MetaschemaException, IOException {
29      IBindingModuleLoader loader = newBindingContext().newModuleLoader();
30      loader.allowEntityResolution();
31  
32      IModule module = loader.load(ObjectUtils.notNull(
33          Paths.get("src/test/resources/metaschema/any/metaschema.xml")));
34  
35      Integer rootIndex = ModuleUtils.parseModelName(module, "root").getIndexPosition();
36      IAssemblyDefinition rootDef = module.getScopedAssemblyDefinitionByName(rootIndex);
37      assertNotNull(rootDef, "root assembly definition should be found");
38  
39      // Access the model container to verify the any instance was loaded
40      DefinitionAssemblyGlobal globalDef = assertInstanceOf(DefinitionAssemblyGlobal.class, rootDef);
41      assertNotNull(globalDef.getModelContainer().getAnyInstance(),
42          "any instance should not be null for assembly with <any/>");
43    }
44  }
