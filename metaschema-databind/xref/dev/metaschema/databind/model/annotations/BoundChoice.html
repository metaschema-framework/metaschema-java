6   package dev.metaschema.databind.model.annotations;
7   
8   import static java.lang.annotation.ElementType.FIELD;
9   import static java.lang.annotation.ElementType.METHOD;
10  import static java.lang.annotation.RetentionPolicy.RUNTIME;
11  
12  import java.lang.annotation.Documented;
13  import java.lang.annotation.Retention;
14  import java.lang.annotation.Target;
15  
16  import edu.umd.cs.findbugs.annotations.NonNull;
17  
18  /**
19   * Identifies that the annotation target is a bound property that participates
20   * in a Metaschema choice.
21   * <p>
22   * A choice represents mutually exclusive alternatives in a Metaschema model.
23   * Fields with the same {@link #choiceId()} are part of the same choice and
24   * exactly one of them must be provided when the choice is required.
25   * <p>
26   * This is distinct from {@link BoundChoiceGroup}, which represents a
27   * polymorphic collection with a type discriminator.
28   * <p>
29   * <strong>Adjacency requirement:</strong> All fields with the same
30   * {@code choiceId} must be declared consecutively in the class, reflecting the
31   * Metaschema model where choice alternatives occupy the same position in the
32   * serialization order.
33   */
34  @Documented
35  @Retention(RUNTIME)
36  @Target({ FIELD, METHOD })
37  public @interface BoundChoice {
38    /**
39     * Identifies which choice this field belongs to.
40     * <p>
41     * Fields with the same choiceId are mutually exclusive alternatives. The
42     * choiceId must be unique within the containing assembly and consistent across
43     * all alternatives in the choice.
44     *
45     * @return the choice identifier
46     */
47    @NonNull
48    String choiceId();
49  }
