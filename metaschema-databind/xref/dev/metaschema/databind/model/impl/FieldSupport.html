6   package dev.metaschema.databind.model.impl;
7   
8   import java.lang.reflect.Field;
9   
10  import edu.umd.cs.findbugs.annotations.NonNull;
11  
12  public final class FieldSupport {
13  
14    /**
15     * Ensure that the provided field can be accessed.
16     *
17     * @param field
18     *          the field to check
19     */
20    @SuppressWarnings("PMD.AvoidAccessibilityAlteration")
21    public static void bindField(@NonNull Field field) {
22      field.setAccessible(true);
23    }
24  
25    private FieldSupport() {
26      // disable construction
27    }
28  }
