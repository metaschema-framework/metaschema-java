6   package dev.metaschema.databind.model.metaschema;
7   
8   import dev.metaschema.core.model.IInstance;
9   import edu.umd.cs.findbugs.annotations.NonNull;
10  
11  /**
12   * Represents a Metaschema instance loaded via data binding.
13   * <p>
14   * This interface provides access to binding-specific metadata for flag, field,
15   * and assembly instances.
16   */
17  public interface IBindingInstance extends IInstance, IBindingModelElement {
18    @Override
19    @NonNull
20    IBindingDefinitionModel getContainingDefinition();
21  
22    @Override
23    default IBindingMetaschemaModule getContainingModule() {
24      return getContainingDefinition().getContainingModule();
25    }
26  }
