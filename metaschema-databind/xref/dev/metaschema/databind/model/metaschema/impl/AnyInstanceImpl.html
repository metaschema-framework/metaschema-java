6   package dev.metaschema.databind.model.metaschema.impl;
7   
8   import dev.metaschema.core.datatype.markup.MarkupMultiline;
9   import dev.metaschema.core.model.IAnyInstance;
10  import dev.metaschema.core.model.IContainerModelAbsolute;
11  import dev.metaschema.core.model.IModule;
12  import edu.umd.cs.findbugs.annotations.NonNull;
13  import edu.umd.cs.findbugs.annotations.Nullable;
14  
15  /**
16   * Implements an {@code any} instance for module-loaded assembly definitions.
17   * <p>
18   * This class provides a simple implementation of {@link IAnyInstance} that is
19   * used when a Metaschema module is loaded from its binding representation. It
20   * stores a reference to the containing model container (typically an assembly
21   * definition) and delegates other behavior to the default methods on
22   * {@link IAnyInstance}.
23   */
24  public final class AnyInstanceImpl implements IAnyInstance {
25    @NonNull
26    private final IContainerModelAbsolute parentContainer;
27  
28    /**
29     * Construct a new {@code any} instance for a module-loaded assembly.
30     *
31     * @param parentContainer
32     *          the model container (assembly definition) that owns this instance
33     */
34    public AnyInstanceImpl(@NonNull IContainerModelAbsolute parentContainer) {
35      this.parentContainer = parentContainer;
36    }
37  
38    @Override
39    public IContainerModelAbsolute getParentContainer() {
40      return parentContainer;
41    }
42  
43    @Override
44    public IModule getContainingModule() {
45      return getContainingDefinition().getContainingModule();
46    }
47  
48    @Override
49    @Nullable
50    public MarkupMultiline getRemarks() {
51      // any instances do not have remarks
52      return null;
53    }
54  }
