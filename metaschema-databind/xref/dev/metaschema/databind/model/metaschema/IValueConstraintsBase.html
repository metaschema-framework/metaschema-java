6   package dev.metaschema.databind.model.metaschema;
7   
8   import java.util.List;
9   
10  import dev.metaschema.core.model.IBoundObject;
11  import dev.metaschema.core.util.CollectionUtil;
12  import dev.metaschema.databind.model.metaschema.binding.ConstraintLetExpression;
13  import edu.umd.cs.findbugs.annotations.NonNull;
14  
15  /**
16   * Base interface for value-level constraint bindings.
17   * <p>
18   * This interface provides access to constraints that apply to flag and field
19   * values.
20   */
21  public interface IValueConstraintsBase extends IBoundObject {
22    /**
23     * Get the let expressions defined for this constraint set.
24     *
25     * <p>
26     * The default implementation returns an empty list. Implementations with let
27     * expressions should override this method.
28     *
29     * @return the list of let expressions, or an empty list if none are defined
30     */
31    @NonNull
32    default List<ConstraintLetExpression> getLets() {
33      return CollectionUtil.emptyList();
34    }
35  
36    /**
37     * Get the constraint rules defined for this constraint set.
38     *
39     * @return the list of constraint rules
40     */
41    List<? extends IConstraintBase> getRules();
42  }
