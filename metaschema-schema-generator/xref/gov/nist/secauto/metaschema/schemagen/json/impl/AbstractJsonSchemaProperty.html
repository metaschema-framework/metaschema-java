6   package gov.nist.secauto.metaschema.schemagen.json.impl;
7   
8   import com.fasterxml.jackson.databind.node.ObjectNode;
9   
10  import gov.nist.secauto.metaschema.core.model.IInstance;
11  
12  import edu.umd.cs.findbugs.annotations.NonNull;
13  
14  /**
15   * A JSON schema for a given Metaschema-based definition instance, which is part
16   * of a larger JSON schema.
17   *
18   * @param <I>
19   *          the Java type of the Metaschema definition instance
20   */
21  public abstract class AbstractJsonSchemaProperty<I extends IInstance>
22      implements IJsonSchemaProperty {
23    @NonNull
24    private final I instance;
25  
26    /**
27     * Construct a new JSON schema property based on a Metaschema definition
28     * instance.
29     *
30     * @param instance
31     *          the Metaschema definition instance
32     */
33    protected AbstractJsonSchemaProperty(@NonNull I instance) {
34      this.instance = instance;
35    }
36  
37    /**
38     * Get the associated Metaschema instance.
39     *
40     * @return the instance
41     */
42    @NonNull
43    public I getInstance() {
44      return instance;
45    }
46  
47    @Override
48    public void generate(@NonNull ObjectNode node, @NonNull IJsonGenerationState state) {
49      generateMetadata(node, state);
50  
51      generateBody(node, state);
52      assert !node.isEmpty();
53    }
54  
55    /**
56     * Generate human-focused documentation and other metadata.
57     *
58     * @param node
59     *          the property JSON object
60     * @param state
61     *          the schema generation state used for context
62     */
63    protected void generateMetadata(
64        @NonNull ObjectNode node,
65        @NonNull IJsonGenerationState state) {
66      // do nothing by default
67    }
68  
69    /**
70     * Generate the JSON schema body.
71     *
72     * @param node
73     *          the property JSON object
74     * @param state
75     *          the schema generation state used for context
76     */
77    protected abstract void generateBody(
78        @NonNull ObjectNode node,
79        @NonNull IJsonGenerationState state);
80  }
