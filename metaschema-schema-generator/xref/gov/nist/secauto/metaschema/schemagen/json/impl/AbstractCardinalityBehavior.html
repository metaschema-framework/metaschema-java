6   package gov.nist.secauto.metaschema.schemagen.json.impl;
7   
8   import com.fasterxml.jackson.databind.node.ArrayNode;
9   import com.fasterxml.jackson.databind.node.ObjectNode;
10  
11  import gov.nist.secauto.metaschema.core.model.IModelInstance;
12  import gov.nist.secauto.metaschema.core.util.ObjectUtils;
13  
14  import java.util.Collection;
15  
16  import edu.umd.cs.findbugs.annotations.NonNull;
17  
18  /**
19   * Supports production of a JSON schema property construction for a Metaschema
20   * instance that has {@link IModelInstance#getJsonGroupAsBehavior()}.
21   */
22  public abstract class AbstractCardinalityBehavior
23      implements ICardinalityBehavior {
24    /**
25     * Generates the type reference(s).
26     *
27     * @param node
28     *          the parent JSON object to add properties to
29     * @param definitions
30     *          the JSON schemas to use to describe the contained data
31     * @param state
32     *          the generation state
33     */
34    protected void generateInternal(
35        @NonNull ObjectNode node,
36        @NonNull Collection<? extends IJsonSchemaDefinable> definitions,
37        @NonNull IJsonGenerationState state) {
38      assert !definitions.isEmpty();
39  
40      if (definitions.size() == 1) {
41        // build the item type reference
42        definitions.iterator().next().generateJsonSchemaOrDefinitionRef(node, state);
43      } else if (definitions.size() > 1) {
44        // build an anyOf of the item type references
45        ArrayNode anyOf = node.putArray("anyOf");
46        for (IJsonSchemaDefinable definition : definitions) {
47          ObjectNode defNode = ObjectUtils.notNull(anyOf.addObject());
48          definition.generateJsonSchemaOrDefinitionRef(defNode, state);
49        }
50      }
51    }
52  }
