6   package dev.metaschema.schemagen.xml.impl.schematype;
7   
8   import javax.xml.namespace.QName;
9   import javax.xml.stream.XMLStreamException;
10  
11  import dev.metaschema.core.datatype.IDataTypeAdapter;
12  import dev.metaschema.core.model.IFieldDefinition;
13  import dev.metaschema.core.model.IFlagInstance;
14  import dev.metaschema.schemagen.xml.impl.IXmlGenerationState;
15  import dev.metaschema.schemagen.xml.impl.XmlDatatypeManager;
16  import edu.umd.cs.findbugs.annotations.NonNull;
17  
18  /**
19   * An XML Schema complex type implementation for Metaschema field definitions
20   * that have flags.
21   * <p>
22   * Fields with flags require a complex type in XML Schema because they need both
23   * content (the field value) and attributes (the flags). This class generates
24   * either simpleContent or complexContent extensions depending on whether the
25   * field's data type produces mixed XML content.
26   */
27  public class XmlComplexTypeFieldDefinition
28      extends AbstractXmlComplexType<IFieldDefinition> {
29    /**
30     * Construct a new complex type for a field definition.
31     *
32     * @param qname
33     *          the qualified name for the XML Schema type
34     * @param definition
35     *          the Metaschema field definition to generate the type for
36     */
37    public XmlComplexTypeFieldDefinition(
38        @NonNull QName qname,
39        @NonNull IFieldDefinition definition) {
40      super(qname, definition);
41    }
42  
43    @Override
44    protected void generateTypeBody(IXmlGenerationState state) throws XMLStreamException {
45      IFieldDefinition definition = getDefinition();
46      IXmlSimpleType valueType = state.getSimpleType(definition);
47      IDataTypeAdapter<?> datatype = valueType.getDataTypeAdapter();
48  
49      String xmlContentType;
50      if (datatype.isXmlMixed()) {
51        xmlContentType = "complexContent"; // with attributes
52      } else {
53        xmlContentType = "simpleContent"; // without attributes
54      }
55      state.writeStartElement(XmlDatatypeManager.PREFIX_XML_SCHEMA, xmlContentType, XmlDatatypeManager.NS_XML_SCHEMA);
56      state.writeStartElement(XmlDatatypeManager.PREFIX_XML_SCHEMA, "extension", XmlDatatypeManager.NS_XML_SCHEMA);
57      state.writeAttribute("base", valueType.getTypeReference());
58  
59      for (IFlagInstance flagInstance : definition.getFlagInstances()) {
60        assert flagInstance != null;
61        generateFlagInstance(flagInstance, state);
62      }
63      state.writeEndElement(); // xs:extension
64  
65      state.writeEndElement(); // xs:simpleContent or xs:complexContent
66    }
67  }
