6   package dev.metaschema.schemagen.xml.impl.schematype;
7   
8   import java.util.LinkedList;
9   import java.util.List;
10  import java.util.stream.Collectors;
11  
12  import javax.xml.namespace.QName;
13  import javax.xml.stream.XMLStreamException;
14  
15  import dev.metaschema.core.model.IValuedDefinition;
16  import dev.metaschema.core.util.CollectionUtil;
17  import dev.metaschema.core.util.ObjectUtils;
18  import dev.metaschema.schemagen.SchemaGenerationException;
19  import dev.metaschema.schemagen.xml.impl.IXmlGenerationState;
20  import dev.metaschema.schemagen.xml.impl.XmlDatatypeManager;
21  import edu.umd.cs.findbugs.annotations.NonNull;
22  
23  /**
24   * An XML Schema simple type that is a union of multiple member types.
25   * <p>
26   * This class generates an xs:simpleType with xs:union that combines multiple
27   * simple types. Member types may be referenced by name or inlined depending on
28   * their generation requirements.
29   */
30  public class XmlSimpleTypeUnion
31      extends AbstractXmlSimpleType {
32    @NonNull
33    private final List<IXmlSimpleType> simpleTypes;
34  
35    /**
36     * Construct a new union type.
37     *
38     * @param qname
39     *          the qualified name for the XML Schema type
40     * @param definition
41     *          the Metaschema definition that this union applies to
42     * @param simpleTypes
43     *          the member types to include in the union
44     */
45    public XmlSimpleTypeUnion(
46        @NonNull QName qname,
47        @NonNull IValuedDefinition definition,
48        @NonNull IXmlSimpleType... simpleTypes) {
49      super(qname, definition);
50      this.simpleTypes = CollectionUtil.requireNonEmpty(CollectionUtil.listOrEmpty(simpleTypes));
51    }
52  
53    /**
54     * Get the member types that make up this union.
55     *
56     * @return an unmodifiable list of member simple types
57     */
58    @NonNull
59    public List<IXmlSimpleType> getSimpleTypes() {
60      return simpleTypes;
61    }
62  
63    @Override
64    public boolean isInline(IXmlGenerationState state) {
65      return true;
66    }
67  
68    @Override
69    public void generate(IXmlGenerationState state) { // NOPMD unavoidable complexity
70      try {
71        state.writeStartElement(XmlDatatypeManager.PREFIX_XML_SCHEMA, "simpleType", XmlDatatypeManager.NS_XML_SCHEMA);
72  
73        if (!isInline(state)) {
74          state.writeAttribute("name", ObjectUtils.notNull(getQName().getLocalPart()));
75        }
76  
77        state.writeStartElement(XmlDatatypeManager.PREFIX_XML_SCHEMA, "union", XmlDatatypeManager.NS_XML_SCHEMA);
78  
79        List<IXmlSimpleType> memberTypes = new LinkedList<>();
80        List<IXmlSimpleType> inlineTypes = new LinkedList<>();
81        for (IXmlSimpleType unionType : simpleTypes) {
82          if (unionType.isGeneratedType(state) && unionType.isInline(state)) {
83            inlineTypes.add(unionType);
84          } else {
85            memberTypes.add(unionType);
86          }
87        }
88  
89        if (!memberTypes.isEmpty()) {
90          state.writeAttribute(
91              "memberTypes",
92              ObjectUtils.notNull(memberTypes.stream()
93                  .map(IXmlSimpleType::getTypeReference)
94                  .collect(Collectors.joining(" "))));
95        }
96  
97        for (IXmlSimpleType inlineType : inlineTypes) {
98          inlineType.generate(state);
99        }
100 
101       state.writeEndElement(); // xs:union
102       state.writeEndElement(); // xs:simpleType
103 
104       for (IXmlSimpleType memberType : memberTypes) {
105         memberType.generate(state);
106       }
107     } catch (XMLStreamException ex) {
108       throw new SchemaGenerationException(ex);
109     }
110   }
111 }
