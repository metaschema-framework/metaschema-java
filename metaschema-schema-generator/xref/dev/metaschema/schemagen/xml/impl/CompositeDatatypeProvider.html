6   package dev.metaschema.schemagen.xml.impl;
7   
8   import org.codehaus.stax2.XMLStreamWriter2;
9   
10  import java.util.ArrayList;
11  import java.util.HashSet;
12  import java.util.LinkedHashMap;
13  import java.util.List;
14  import java.util.Map;
15  import java.util.Set;
16  import java.util.function.Function;
17  import java.util.stream.Collectors;
18  
19  import javax.xml.stream.XMLStreamException;
20  
21  import dev.metaschema.core.util.CollectionUtil;
22  import dev.metaschema.core.util.ObjectUtils;
23  import edu.umd.cs.findbugs.annotations.NonNull;
24  
25  /**
26   * A composite datatype provider that aggregates multiple datatype providers.
27   * <p>
28   * This class implements the composite pattern, delegating datatype operations
29   * to a collection of underlying providers.
30   */
31  public class CompositeDatatypeProvider implements IDatatypeProvider {
32    @NonNull
33    private final List<IDatatypeProvider> proxiedProviders;
34  
35    /**
36     * Constructs a new composite datatype provider with the given providers.
37     *
38     * @param proxiedProviders
39     *          the list of providers to aggregate
40     */
41    public CompositeDatatypeProvider(@NonNull List<IDatatypeProvider> proxiedProviders) {
42      this.proxiedProviders = CollectionUtil.unmodifiableList(new ArrayList<>(proxiedProviders));
43    }
44  
45    /**
46     * Retrieves the list of proxied datatype providers.
47     *
48     * @return an unmodifiable list of the underlying providers
49     */
50    @NonNull
51    protected List<IDatatypeProvider> getProxiedProviders() {
52      return proxiedProviders;
53    }
54  
55    @Override
56    public Map<String, IDatatypeContent> getDatatypes() {
57      return ObjectUtils.notNull(proxiedProviders.stream()
58          .flatMap(provider -> provider.getDatatypes().values().stream())
59          .collect(Collectors.toMap(
60              IDatatypeContent::getTypeName,
61              Function.identity(),
62              (e1, e2) -> e2,
63              LinkedHashMap::new)));
64    }
65  
66    @Override
67    public Set<String> generateDatatypes(Set<String> requiredTypes, XMLStreamWriter2 writer)
68        throws XMLStreamException {
69      Set<String> retval = new HashSet<>();
70  
71      for (IDatatypeProvider provider : getProxiedProviders()) {
72        retval.addAll(provider.generateDatatypes(requiredTypes, writer));
73      }
74      return retval;
75    }
76  
77  }
