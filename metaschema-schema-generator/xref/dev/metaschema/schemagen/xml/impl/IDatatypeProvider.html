6   package dev.metaschema.schemagen.xml.impl;
7   
8   import org.codehaus.stax2.XMLStreamWriter2;
9   
10  import java.util.Map;
11  import java.util.Set;
12  
13  import javax.xml.stream.XMLStreamException;
14  
15  import edu.umd.cs.findbugs.annotations.NonNull;
16  
17  /**
18   * Provides XML Schema datatype definitions for schema generation.
19   * <p>
20   * Implementations supply datatype content that can be written to an XML Schema
21   * document.
22   */
23  public interface IDatatypeProvider {
24    /**
25     * Retrieves all datatypes provided by this provider.
26     *
27     * @return a map of datatype names to their content definitions
28     */
29    @NonNull
30    Map<String, IDatatypeContent> getDatatypes();
31  
32    /**
33     * Generates XML Schema datatype definitions for the specified required types.
34     * <p>
35     * Only datatypes that are both provided by this provider and included in the
36     * required types set will be generated.
37     *
38     * @param requiredTypes
39     *          the set of datatype names that are required
40     * @param writer
41     *          the XML stream writer to write datatype definitions to
42     * @return the set of datatype names that were actually generated
43     * @throws XMLStreamException
44     *           if an error occurs while writing XML content
45     */
46    @NonNull
47    Set<String> generateDatatypes(Set<String> requiredTypes, @NonNull XMLStreamWriter2 writer)
48        throws XMLStreamException;
49  
50  }
