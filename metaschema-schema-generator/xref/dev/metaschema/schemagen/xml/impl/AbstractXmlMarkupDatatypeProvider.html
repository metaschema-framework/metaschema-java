6   package dev.metaschema.schemagen.xml.impl;
7   
8   import org.eclipse.jdt.annotation.Owning;
9   import org.w3c.dom.Element;
10  
11  import java.io.InputStream;
12  import java.util.List;
13  import java.util.Map;
14  import java.util.stream.Collectors;
15  
16  import dev.metaschema.core.model.IModule;
17  import dev.metaschema.core.util.CollectionUtil;
18  import dev.metaschema.core.util.ObjectUtils;
19  import edu.umd.cs.findbugs.annotations.NonNull;
20  
21  /**
22   * Provides a base implementation for XML markup datatype providers.
23   * <p>
24   * This class handles the loading and processing of XML schema resources that
25   * define markup datatypes (such as markup-line and markup-multiline).
26   */
27  public abstract class AbstractXmlMarkupDatatypeProvider
28      extends AbstractXmlDatatypeProvider {
29  
30    @SuppressWarnings("null")
31    @Owning
32    @Override
33    protected InputStream getSchemaResource() {
34      return IModule.class.getResourceAsStream(getSchemaResourcePath());
35    }
36  
37    /**
38     * Get the absolute classpath of the schema resource.
39     *
40     * @return the resource path
41     */
42    @NonNull
43    protected abstract String getSchemaResourcePath();
44  
45    @Override
46    protected List<Element> queryElements(XmlSchemaLoader loader) {
47      return loader.getContent(
48          "/xs:schema/*",
49          CollectionUtil.singletonMap("xs", XmlSchemaLoader.NS_XML_SCHEMA));
50    }
51  
52    /**
53     * Get the name of the data type provided by this provider.
54     *
55     * @return the data type name
56     */
57    @NonNull
58    protected abstract String getDataTypeName();
59  
60    @Override
61    protected Map<String, IDatatypeContent> handleResults(@NonNull List<Element> items) {
62      String dataTypeName = getDataTypeName();
63      return CollectionUtil.singletonMap(
64          dataTypeName,
65          new DomDatatypeContent(
66              dataTypeName,
67              ObjectUtils.notNull(items.stream()
68                  .filter(element -> !"include".equals(element.getLocalName()))
69                  .collect(Collectors.toList())),
70              CollectionUtil.emptyList()));
71    }
72  
73  }
