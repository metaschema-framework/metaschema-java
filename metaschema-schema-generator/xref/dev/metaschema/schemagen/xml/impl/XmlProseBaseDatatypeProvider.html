6   package dev.metaschema.schemagen.xml.impl;
7   
8   import org.eclipse.jdt.annotation.Owning;
9   import org.w3c.dom.Element;
10  
11  import java.io.InputStream;
12  import java.util.List;
13  import java.util.Map;
14  
15  import dev.metaschema.core.model.IModule;
16  import dev.metaschema.core.util.CollectionUtil;
17  import dev.metaschema.core.util.ObjectUtils;
18  import edu.umd.cs.findbugs.annotations.NonNull;
19  
20  /**
21   * Provides prose base datatype from the metaschema-prose-base.xsd resource.
22   */
23  public class XmlProseBaseDatatypeProvider
24      extends AbstractXmlDatatypeProvider {
25    private static final String DATATYPE_NAME = "ProseBase";
26  
27    @SuppressWarnings("resource")
28    @Override
29    @Owning
30    protected InputStream getSchemaResource() {
31      return ObjectUtils.requireNonNull(IModule.class.getResourceAsStream("/schema/xml/metaschema-prose-base.xsd"));
32    }
33  
34    @Override
35    protected List<Element> queryElements(XmlSchemaLoader loader) {
36      return loader.getContent(
37          "/xs:schema/*",
38          CollectionUtil.singletonMap("xs", XmlSchemaLoader.NS_XML_SCHEMA));
39    }
40  
41    @Override
42    @NonNull
43    protected Map<String, IDatatypeContent> handleResults(@NonNull List<Element> items) {
44      return CollectionUtil.singletonMap(
45          DATATYPE_NAME,
46          new DomDatatypeContent(
47              DATATYPE_NAME,
48              items,
49              CollectionUtil.emptyList()));
50    }
51  }
