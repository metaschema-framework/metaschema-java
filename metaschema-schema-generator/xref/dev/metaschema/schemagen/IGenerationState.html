6   package dev.metaschema.schemagen;
7   
8   import java.io.IOException;
9   import java.util.Collection;
10  import java.util.Locale;
11  
12  import dev.metaschema.core.model.IAssemblyDefinition;
13  import dev.metaschema.core.model.IDefinition;
14  import dev.metaschema.core.model.IModule;
15  import edu.umd.cs.findbugs.annotations.NonNull;
16  import edu.umd.cs.findbugs.annotations.Nullable;
17  
18  /**
19   * Represents the state information used during schema generation.
20   *
21   * @param <WRITER>
22   *          the type of writer used for schema output
23   */
24  public interface IGenerationState<WRITER> {
25    /**
26     * Get the Metaschema module being processed for schema generation.
27     *
28     * @return the module
29     */
30    @NonNull
31    IModule getModule();
32  
33    /**
34     * Get the writer used for schema output.
35     * <p>
36     * The caller does not own this writer and must not close it.
37     *
38     * @return the writer instance
39     */
40    @NonNull
41    WRITER getWriter();
42  
43    /**
44     * Get the collection of root assembly definitions exported by the module.
45     *
46     * @return the root assembly definitions
47     */
48    @NonNull
49    default Collection<? extends IAssemblyDefinition> getRootDefinitions() {
50      return getModule().getExportedRootAssemblyDefinitions();
51    }
52  
53    /**
54     * Determine if the provided definition should be inlined in the generated
55     * schema.
56     *
57     * @param definition
58     *          the definition to check
59     * @return {@code true} if the definition should be inlined, {@code false}
60     *         otherwise
61     */
62    boolean isInline(@NonNull IDefinition definition);
63  
64    /**
65     * Flush any buffered content to the underlying writer.
66     *
67     * @throws IOException
68     *           if an I/O error occurs while flushing
69     */
70    void flushWriter() throws IOException;
71  
72    /**
73     * Generate a type name for the provided definition with an optional suffix.
74     *
75     * @param definition
76     *          the definition to generate a type name for
77     * @param suffix
78     *          an optional suffix to append to the type name, or {@code null} if no
79     *          suffix is needed
80     * @return the generated type name
81     */
82    @NonNull
83    String getTypeNameForDefinition(@NonNull IDefinition definition, @Nullable String suffix);
84  
85    /**
86     * Convert a text string to camel case by splitting on punctuation and
87     * capitalizing each segment.
88     *
89     * @param text
90     *          the text to convert
91     * @return the camel case representation of the text
92     */
93    @NonNull
94    static CharSequence toCamelCase(String text) {
95      StringBuilder builder = new StringBuilder();
96      for (String segment : text.split("\\p{Punct}")) {
97        if (segment.length() > 0) {
98          builder.append(segment.substring(0, 1).toUpperCase(Locale.ROOT));
99        }
100       if (segment.length() > 1) {
101         builder.append(segment.substring(1).toLowerCase(Locale.ROOT));
102       }
103     }
104     return builder;
105   }
106 }
