33  class DomDatatypeContentTest {

35    private static final String NS_XML_SCHEMA = "http://www.w3.org/2001/XMLSchema";
36  
37    private Element createSchemaElement(Document doc, String name) {
38      Element element = doc.createElementNS(NS_XML_SCHEMA, "xs:simpleType");
39      element.setAttribute("name", name);
40  
41      Element restriction = doc.createElementNS(NS_XML_SCHEMA, "xs:restriction");
42      restriction.setAttribute("base", "xs:string");
43      element.appendChild(restriction);
44  
45      return element;
46    }
47  
48    private Document createDocument() throws Exception {
49      DocumentBuilderFactory factory = DocumentBuilderFactory.newInstance();
50      factory.setNamespaceAware(true);
51      DocumentBuilder builder = factory.newDocumentBuilder();
52      return builder.newDocument();
53    }
54  
55    @Nested
56    @DisplayName("Basic Properties Tests")
57    class BasicPropertiesTests {
58  
59      @Test
60      @DisplayName("getTypeName returns correct type name")
61      void testGetTypeName() throws Exception {
62        Document doc = createDocument();
63        Element element = createSchemaElement(doc, "TestDatatype");
64  
65        DomDatatypeContent content = new DomDatatypeContent(
66            "TestDatatype",
67            CollectionUtil.singletonList(element),
68            CollectionUtil.emptyList());
69  
70        assertEquals("TestDatatype", content.getTypeName());
71      }
72  
73      @Test
74      @DisplayName("getDependencies returns correct dependencies")
75      void testGetDependencies() throws Exception {
76        Document doc = createDocument();
77        Element element = createSchemaElement(doc, "TestDatatype");
78        List<String> dependencies = Arrays.asList("BaseDatatypeA", "BaseDatatypeB");
79  
80        DomDatatypeContent content = new DomDatatypeContent(
81            "TestDatatype",
82            CollectionUtil.singletonList(element),
83            dependencies);
84  
85        assertEquals(2, content.getDependencies().size());
86        assertTrue(content.getDependencies().contains("BaseDatatypeA"));
87        assertTrue(content.getDependencies().contains("BaseDatatypeB"));
88      }
89  
90      @Test
91      @DisplayName("empty dependencies list works correctly")
92      void testEmptyDependencies() throws Exception {
93        Document doc = createDocument();
94        Element element = createSchemaElement(doc, "TestDatatype");
95  
96        DomDatatypeContent content = new DomDatatypeContent(
97            "TestDatatype",
98            CollectionUtil.singletonList(element),
99            CollectionUtil.emptyList());
100 
101       assertNotNull(content.getDependencies());
102       assertTrue(content.getDependencies().isEmpty());
103     }
104   }

106   @Nested
107   @DisplayName("Write to XMLStreamWriter Tests")
108   class WriteTests {
109 
110     @Test
111     @DisplayName("writes simple element correctly")
112     void testWriteSimpleElement() throws Exception {
113       Document doc = createDocument();
114       Element element = createSchemaElement(doc, "StringDatatype");
115 
116       DomDatatypeContent content = new DomDatatypeContent(
117           "StringDatatype",
118           CollectionUtil.singletonList(element),
119           CollectionUtil.emptyList());
120 
121       StringWriter sw = new StringWriter();
122       XMLOutputFactory factory = XMLOutputFactory.newInstance();
123       XMLStreamWriter writer = factory.createXMLStreamWriter(sw);
124 
125       writer.writeStartDocument();
126       writer.writeStartElement("wrapper");
127       content.write(writer);
128       writer.writeEndElement();
129       writer.writeEndDocument();
130       writer.close();
131 
132       String result = sw.toString();
133 
134       // Verify the element was written
135       assertTrue(result.contains("simpleType"), "Should contain simpleType element");
136       assertTrue(result.contains("StringDatatype"), "Should contain type name attribute");
137       assertTrue(result.contains("restriction"), "Should contain child elements");
138     }
139 
140     @Test

142     void testWriteMultipleElements() throws Exception {

168     }

198     @Test
199     @DisplayName("handles element with text content")
200     void testElementWithTextContent() throws Exception {
201       Document doc = createDocument();

205       element.appendChild(docElement);

208           "AnnotatedType",

210           CollectionUtil.emptyList());
211 
212       StringWriter sw = new StringWriter();

226       assertTrue(result.contains("This is documentation text"), "Should contain text content");
227     }
