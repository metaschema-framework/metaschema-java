34  class AnyJsonSchemaGenerationTest {
35  
36    private static final Path METASCHEMA_FILE
37        = ObjectUtils.notNull(Paths.get("src/test/resources/metaschema/any-test_metaschema.xml"));
38  
39    private static IBindingContext getBindingContext() throws IOException {
40      return IBindingContext.builder()
41          .compilePath(ObjectUtils.notNull(Files.createTempDirectory(Paths.get("target"), "modules-")))
42          .build();
43    }
44  
45    @Test
46    void testAnyAssemblyHasAdditionalPropertiesTrue() throws MetaschemaException, IOException {
47      IBindingContext bindingContext = getBindingContext();
48      IBindingModuleLoader loader = bindingContext.newModuleLoader();
49      loader.allowEntityResolution();
50  
51      IModule module = loader.load(METASCHEMA_FILE);
52  
53      IMutableConfiguration<SchemaGenerationFeature<?>> features
54          = new DefaultConfiguration<>();
55      features.disableFeature(SchemaGenerationFeature.INLINE_DEFINITIONS);
56  
57      ISchemaGenerator schemaGenerator = new JsonSchemaGenerator();
58      StringWriter writer = new StringWriter();
59      schemaGenerator.generateFromModule(module, writer, features);
60  
61      String schemaJson = writer.toString();
62      assertNotNull(schemaJson, "Generated schema should not be null");
63  
64      ObjectMapper mapper = new ObjectMapper();
65      JsonNode schema = mapper.readTree(schemaJson);
66  
67      // The root assembly 'root' should be in the definitions
68      JsonNode definitions = schema.get("definitions");
69      assertNotNull(definitions, "Schema should have definitions");
70  
71      // Find the root assembly definition (named "AssemblyAnyTestRootType")
72      JsonNode rootDef = null;
73      var fieldNames = definitions.fieldNames();
74      while (fieldNames.hasNext()) {
75        String name = fieldNames.next();
76        if (name.contains("Root")) {
77          rootDef = definitions.get(name);
78          break;
79        }
80      }
81      assertNotNull(rootDef, "Should find 'root' assembly definition");
82  
83      // The root assembly has <any/>, so additionalProperties should be true
84      JsonNode additionalProperties = rootDef.get("additionalProperties");
85      assertNotNull(additionalProperties,
86          "Root assembly with <any/> should have additionalProperties defined. Schema: " + rootDef);
87      assertTrue(additionalProperties.isBoolean(),
88          "additionalProperties should be a boolean value");
89      assertEquals(true, additionalProperties.booleanValue(),
90          "additionalProperties should be true for assembly with <any/>");
91    }
92  
93    @Test
94    void testAssemblyWithoutAnyHasAdditionalPropertiesFalse() throws MetaschemaException, IOException {
95      IBindingContext bindingContext = getBindingContext();
96      IBindingModuleLoader loader = bindingContext.newModuleLoader();
97      loader.allowEntityResolution();
98  
99      // Use the metaschema-module-metaschema as an example of an assembly without
100     // <any/>
101     Path metaschemaFile = ObjectUtils.notNull(
102         Paths.get("../core/metaschema/schema/metaschema/metaschema-module-metaschema.xml"));
103     IModule module = loader.load(metaschemaFile);
104 
105     IMutableConfiguration<SchemaGenerationFeature<?>> features
106         = new DefaultConfiguration<>();
107     features.disableFeature(SchemaGenerationFeature.INLINE_DEFINITIONS);
108 
109     ISchemaGenerator schemaGenerator = new JsonSchemaGenerator();
110     StringWriter writer = new StringWriter();
111     schemaGenerator.generateFromModule(module, writer, features);
112 
113     String schemaJson = writer.toString();
114     ObjectMapper mapper = new ObjectMapper();
115     JsonNode schema = mapper.readTree(schemaJson);
116 
117     JsonNode definitions = schema.get("definitions");
118     assertNotNull(definitions, "Schema should have definitions");
119 
120     // Check that at least one definition has additionalProperties: false
121     // (meaning assemblies without <any/> retain the existing behavior)
122     var fieldNames = definitions.fieldNames();
123     boolean foundFalse = false;
124     while (fieldNames.hasNext()) {
125       String name = fieldNames.next();
126       JsonNode def = definitions.get(name);
127       JsonNode additionalProperties = def.get("additionalProperties");
128       if (additionalProperties != null && additionalProperties.isBoolean()
129           && !additionalProperties.booleanValue()) {
130         foundFalse = true;
131         break;
132       }
133     }
134     assertTrue(foundFalse,
135         "At least one assembly without <any/> should have additionalProperties: false");
136   }
