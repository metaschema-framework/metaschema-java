50  /**
51   * Base class for schema generation test suites, providing utilities for XML and
52   * JSON schema generation and validation.
53   */
54  public abstract class AbstractSchemaGeneratorTestSuite
55      extends AbstractTestSuite {
56    @NonNull
57    protected static final ISchemaGenerator XML_SCHEMA_GENERATOR = new XmlSchemaGenerator();
58    @NonNull
59    protected static final ISchemaGenerator JSON_SCHEMA_GENERATOR = new JsonSchemaGenerator();
60    @NonNull
61    protected static final IConfiguration<SchemaGenerationFeature<?>> SCHEMA_GENERATION_CONFIG;
62    @NonNull
63    protected static final BiFunction<IModule, Writer, Void> XML_SCHEMA_PROVIDER;
64    @NonNull
65    protected static final BiFunction<IModule, Writer, Void> JSON_SCHEMA_PROVIDER;
66    @NonNull
67    protected static final JsonSchemaContentValidator JSON_SCHEMA_VALIDATOR;
68    @NonNull
69    protected static final Function<Path, JsonSchemaContentValidator> JSON_CONTENT_VALIDATOR_PROVIDER;
70    @NonNull
71    protected static final Function<Path, XmlSchemaContentValidator> XML_CONTENT_VALIDATOR_PROVIDER;
72  
73    private static final String UNIT_TEST_CONFIG
74        = "../core/metaschema/test-suite/schema-generation/unit-tests.xml";
75  
76    static {
77      IMutableConfiguration<SchemaGenerationFeature<?>> features = new DefaultConfiguration<>();
78      // features.enableFeature(SchemaGenerationFeature.INLINE_DEFINITIONS);
79      features.disableFeature(SchemaGenerationFeature.INLINE_DEFINITIONS);
80      SCHEMA_GENERATION_CONFIG = features;
81  
82      BiFunction<IModule, Writer, Void> xmlProvider = (module, writer) -> {
83        assert module != null;
84        assert writer != null;
85        try {
86          XML_SCHEMA_GENERATOR.generateFromModule(module, writer, SCHEMA_GENERATION_CONFIG);
87        } catch (SchemaGenerationException ex) {
88          throw new JUnitException("IO error", ex);
89        }
90        return null;
91      };
92      XML_SCHEMA_PROVIDER = xmlProvider;
93  
94      BiFunction<IModule, Writer, Void> jsonProvider = (module, writer) -> {
95        assert module != null;
96        assert writer != null;
97        try {
98          JSON_SCHEMA_GENERATOR.generateFromModule(module, writer, SCHEMA_GENERATION_CONFIG);
99        } catch (SchemaGenerationException ex) {
100         throw new JUnitException("IO error", ex);
101       }
102       return null;
103     };
104     JSON_SCHEMA_PROVIDER = jsonProvider;
105     // Module module = ModuleLayer.boot()
106     // .findModule("dev.metaschema.core")
107     // .orElseThrow();
108     //
109     // try (InputStream is
110     // = module.getResourceAsStream("schema.json/json-schema.json")) {
111     try (InputStream is = MetaschemaModelConstants.class.getResourceAsStream("/schema/json/json-schema.json")) {
112       assert is != null : "unable to get JSON schema resource";
113       JsonSchemaContentValidator schemaValidator = new JsonSchemaContentValidator(is);
114       JSON_SCHEMA_VALIDATOR = schemaValidator;
115     } catch (IOException ex) {
116       throw new IllegalStateException(ex);
117     }
118 
119     @SuppressWarnings("null")
120     @NonNull
121     Function<Path, XmlSchemaContentValidator> xmlContentValidatorProvider = path -> {
122       try {
123         URL schemaResource = path.toUri().toURL();
124         @SuppressWarnings("resource")
125         StreamSource source
126             = new StreamSource(schemaResource.openStream(), schemaResource.toString());
127         List<? extends Source> schemaSources = Collections.singletonList(source);
128         return new XmlSchemaContentValidator(schemaSources);
129       } catch (IOException ex) {
130         throw new IllegalStateException(ex);
131       }
132     };
133     XML_CONTENT_VALIDATOR_PROVIDER = xmlContentValidatorProvider;
134 
135     @NonNull
136     Function<Path, JsonSchemaContentValidator> jsonContentValidatorProvider = path -> {
137       try (InputStream is = Files.newInputStream(path, StandardOpenOption.READ)) {
138         assert is != null;
139         return new JsonSchemaContentValidator(is);
140       } catch (IOException ex) {
141         throw new JUnitException("Failed to create content validator for schema: " + path.toString(), ex);
142       }
143     };
144     JSON_CONTENT_VALIDATOR_PROVIDER = jsonContentValidatorProvider;
145   }

169   protected static IBindingContext newBindingContext(@NonNull Collection<IConstraintSet> constraints)
170       throws IOException {
171     Path generationDir = Paths.get("target/generated-modules");
172     Files.createDirectories(generationDir);
173 
174     return IBindingContext.builder()
175         .compilePath(ObjectUtils.notNull(Files.createTempDirectory(generationDir, "modules-")))
176         .constraintSet(constraints)
177         .build();
178   }

226   /**
227    * Runs a schema generation and validation test.
228    *
229    * @param collectionName
230    *          the name of the test collection directory
231    * @param metaschemaName
232    *          the name of the Metaschema file to load
233    * @param generatedSchemaName
234    *          the base name for the generated schema file
235    * @param contentCases
236    *          the content validation test cases to run
237    * @throws IOException
238    *           if an I/O error occurs
239    * @throws MetaschemaException
240    *           if a Metaschema error occurs
241    */
242   @SuppressWarnings("null")
243   protected void doTest(
244       @NonNull String collectionName,
245       @NonNull String metaschemaName,
246       @NonNull String generatedSchemaName,
247       @NonNull ContentCase... contentCases) throws IOException, MetaschemaException {
248     Path generationDir = getGenerationPath();
249 
250     Path testSuite = Paths.get("../core/metaschema/test-suite/schema-generation/");
251     Path collectionPath = testSuite.resolve(collectionName);
252 
253     IBindingContext bindingContext = newBindingContext();
254 
255     // load the metaschema module
256     IBindingModuleLoader loader = bindingContext.newModuleLoader();
257     loader.allowEntityResolution();
258     Path modulePath = collectionPath.resolve(metaschemaName);
259     IModule module = loader.load(modulePath);
260 
261     // generate the schema
262     Path schemaPath;
263     Format requiredContentFormat = getRequiredContentFormat();
264     switch (requiredContentFormat) {
265     case JSON:
266     case YAML:
267       Path jsonSchema = produceJsonSchema(module, generationDir.resolve(generatedSchemaName + ".json"));
268       assertTrue(validateWithSchema(JSON_SCHEMA_VALIDATOR, jsonSchema),
269           String.format("JSON schema '%s' was invalid", jsonSchema.toString()));
270       schemaPath = jsonSchema;
271       break;
272     case XML:
273       schemaPath = produceXmlSchema(module, generationDir.resolve(generatedSchemaName + ".xsd"));
274       break;
275     default:
276       throw new IllegalStateException();
277     }
278 
279     // create content test cases
280     for (ContentCase contentCase : contentCases) {
281       Path contentPath = collectionPath.resolve(contentCase.getName());
282 
283       if (!requiredContentFormat.equals(contentCase.getActualFormat())) {
284         contentPath = convertContent(contentPath.toUri(), generationDir, bindingContext);
285       }
286 
287       assertEquals(contentCase.isValid(),
288           validateWithSchema(getContentValidatorSupplier().apply(schemaPath), contentPath),
289           String.format("validation of '%s' did not match expectation", contentPath));
290     }
291   }

293   /**
294    * Creates a new content validation test case.
295    *
296    * @param actualFormat
297    *          the format of the content file
298    * @param contentName
299    *          the name of the content file
300    * @param valid
301    *          whether the content is expected to be valid
302    * @return a new content case
303    */
304   @NonNull
305   protected ContentCase contentCase(
306       @NonNull Format actualFormat,
307       @NonNull String contentName,
308       boolean valid) {
309     return new ContentCase(contentName, actualFormat, valid);

312   /**
313    * Represents a content validation test case.
314    */
315   protected static class ContentCase {
316     @NonNull
317     private final String name;
318     @NonNull
319     private final Format actualFormat;
320     private final boolean valid;
321 
322     /**
323      * Constructs a new content case.
324      *
325      * @param name
326      *          the name of the content file
327      * @param actualFormat
328      *          the format of the content file
329      * @param valid
330      *          whether the content is expected to be valid
331      */
332     public ContentCase(@NonNull String name, @NonNull Format actualFormat, boolean valid) {
333       this.name = name;
334       this.actualFormat = actualFormat;
335       this.valid = valid;
336     }
337 
338     /**
339      * Gets the name of the content file.
340      *
341      * @return the content file name
342      */
343     @NonNull
344     public String getName() {
345       return name;
346     }
347 
348     /**
349      * Gets the format of the content file.
350      *
351      * @return the content format
352      */
353     @NonNull
354     public Format getActualFormat() {
355       return actualFormat;
356     }
357 
358     /**
359      * Checks whether the content is expected to be valid.
360      *
361      * @return {@code true} if the content should be valid, {@code false} otherwise
362      */
363     public boolean isValid() {
364       return valid;
365     }
366   }
