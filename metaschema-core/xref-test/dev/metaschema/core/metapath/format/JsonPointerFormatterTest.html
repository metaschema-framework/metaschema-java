475   @Nested
476   @DisplayName("Full Path Formatting")
477   class FullPathTests {
478 
479     @Test
480     @DisplayName("format produces complete JSON Pointer path")
481     void testFormatCompletePath() {
482       IEnhancedQName rootQname = IEnhancedQName.of(TEST_NS, "catalog");
483       IEnhancedQName assemblyQname = IEnhancedQName.of(TEST_NS, "control");
484       IEnhancedQName flagQname = IEnhancedQName.of(TEST_NS, "id");
485 
486       IFlagNodeItem flag = mockFactory.flag(flagQname, IStringItem.valueOf("ac-1"));
487       IAssemblyNodeItem assembly = mockFactory.assembly(
488           assemblyQname,
489           CollectionUtil.singletonList(flag),
490           CollectionUtil.emptyList());
491 
492       // Set up instance for assembly
493       IAssemblyInstance assemblyInstance = mock(IAssemblyInstance.class);
494       doReturn(assemblyInstance).when(assembly).getInstance();
495       doReturn(JsonGroupAsBehavior.LIST).when(assemblyInstance).getJsonGroupAsBehavior();
496       doReturn("controls").when(assemblyInstance).getJsonName();
497 
498       // Create document - this sets up the root assembly with a definition
499       IDocumentNodeItem document = mockFactory.document(
500           URI.create("https://example.com/catalog.json"),
501           rootQname,
502           CollectionUtil.emptyList(),
503           CollectionUtil.singletonList(assembly));
504 
505       // Get the root and set up JSON name on its definition
506       IRootAssemblyNodeItem root = document.getRootAssemblyNodeItem();
507       IAssemblyDefinition rootDef = root.getDefinition();
508       doReturn("catalog").when(rootDef).getJsonName();
509 
510       String result = formatter.format(flag);
511 
512       // Expected: /catalog/controls/0/id
513       assertEquals("/catalog/controls/0/id", result);
514     }
