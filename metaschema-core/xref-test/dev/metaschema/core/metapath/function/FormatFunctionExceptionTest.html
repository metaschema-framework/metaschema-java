6   package dev.metaschema.core.metapath.function;
7   
8   import static org.junit.jupiter.api.Assertions.assertEquals;
9   import static org.junit.jupiter.api.Assertions.assertNotNull;
10  import static org.junit.jupiter.api.Assertions.assertNull;
11  import static org.junit.jupiter.api.Assertions.assertSame;
12  
13  import org.junit.jupiter.api.Test;
14  
15  import dev.metaschema.core.metapath.IErrorCode;
16  
17  class FormatFunctionExceptionTest {
18  
19    @Test
20    void testConstructionWithCodeAndMessage() {
21      String message = "invalid format token '~'";
22      FormatFunctionException ex
23          = new FormatFunctionException(FormatFunctionException.INVALID_FORMAT_TOKEN, message);
24  
25      IErrorCode errorCode = ex.getErrorCode();
26      assertNotNull(errorCode);
27      assertEquals("FODF", errorCode.getPrefix());
28      assertEquals(1310, errorCode.getCode());
29      assertEquals("FODF1310: " + message, ex.getMessage());
30      assertNull(ex.getCause());
31    }
32  
33    @Test
34    void testErrorCodePrefix() {
35      FormatFunctionException ex
36          = new FormatFunctionException(FormatFunctionException.INVALID_FORMAT_TOKEN, "test");
37  
38      IErrorCode errorCode = ex.getErrorCode();
39      assertEquals("FODF", errorCode.getPrefix());
40      assertEquals(1310, errorCode.getCode());
41      assertEquals("FODF1310", errorCode.getCodeAsString());
42    }
43  
44    @Test
45    void testConstructionWithCodeMessageAndCause() {
46      String message = "invalid format token";
47      Throwable cause = new IllegalArgumentException("bad token");
48      FormatFunctionException ex
49          = new FormatFunctionException(FormatFunctionException.INVALID_FORMAT_TOKEN, message, cause);
50  
51      IErrorCode errorCode = ex.getErrorCode();
52      assertEquals("FODF", errorCode.getPrefix());
53      assertEquals(1310, errorCode.getCode());
54      assertEquals("FODF1310: " + message, ex.getMessage());
55      assertSame(cause, ex.getCause());
56    }
57  
58    @Test
59    void testConstructionWithCodeAndCause() {
60      Throwable cause = new IllegalArgumentException("bad token");
61      FormatFunctionException ex
62          = new FormatFunctionException(FormatFunctionException.INVALID_FORMAT_TOKEN, cause);
63  
64      IErrorCode errorCode = ex.getErrorCode();
65      assertEquals("FODF", errorCode.getPrefix());
66      assertEquals(1310, errorCode.getCode());
67      assertSame(cause, ex.getCause());
68    }
69  }
