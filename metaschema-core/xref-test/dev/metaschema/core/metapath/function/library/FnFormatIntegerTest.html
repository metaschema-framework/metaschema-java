131   @ParameterizedTest
132   @MethodSource("provideValues")
133   void testExpression(@NonNull IStringItem expected, @NonNull String metapath) {
134     assertEquals(
135         expected,
136         IMetapathExpression.compile(metapath)
137             .evaluateAs(null, IMetapathExpression.ResultType.ITEM, newDynamicContext()));
138   }
139 
140   @Test
141   void testOrdinalWithWordFormatFallsBackToCardinal() {
142     // Spec: "If ordinal numbering is not supported for the combination of the
143     // format token, the language, and the string appearing in parentheses, the
144     // request is ignored and cardinal numbers are generated instead."
145     // Our implementation does not support word ordinals (first, second, etc.),
146     // so it must fall back to cardinal (one, two, etc.)
147     assertEquals(
148         string("one"),
149         IMetapathExpression.compile("format-integer(1, 'w;o', 'en')")
150             .evaluateAs(null, IMetapathExpression.ResultType.ITEM, newDynamicContext()));
151   }
152 
153   @Test
154   void testInvalidFormatToken() {
155     assertThrows(MetapathException.class, () -> {
156       IMetapathExpression.compile("format-integer(1, '')")
157           .evaluateAs(null, IMetapathExpression.ResultType.ITEM, newDynamicContext());
158     });
