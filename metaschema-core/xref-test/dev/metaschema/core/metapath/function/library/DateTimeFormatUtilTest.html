30  /**
31   * Unit tests for the picture string parser and formatting engine in
32   * {@link DateTimeFormatUtil}.
33   * <p>
34   * These tests verify that {@link DateTimeFormatUtil#parsePictureString(String)}
35   * correctly parses XPath 3.1 picture strings (spec section 9.8) into a list of
36   * {@link DateTimeFormatUtil.FormatComponent} objects, including literal text,
37   * escaped brackets, variable markers with modifiers, width specifications, and
38   * error handling.
39   * <p>
40   * Additionally, tests verify that
41   * {@link DateTimeFormatUtil#formatDateTime(ITemporalItem, String, String, String, String, Set)}
42   * correctly formats temporal values according to picture strings, covering all
43   * component specifiers (Y, M, D, d, F, W, w, H, h, P, m, s, f, Z, z, C, E),
44   * presentation modifiers, width modifiers, and error cases.
45   *
46   * @see <a href=
47   *      "https://www.w3.org/TR/xpath-functions-31/#date-picture-string">XPath
48   *      Functions 3.1 - Date Picture String</a>
49   */
50  class DateTimeFormatUtilTest {

82    /**
83     * Assert that the given component is a
84     * {@link DateTimeFormatUtil.LiteralComponent} with the expected text.
85     *
86     * @param component
87     *          the component to check
88     * @param expectedText
89     *          the expected literal text
90     */
91    private static void assertLiteral(
92        @NonNull DateTimeFormatUtil.FormatComponent component,
93        @NonNull String expectedText) {
94      assertInstanceOf(DateTimeFormatUtil.LiteralComponent.class, component);
95      assertEquals(expectedText,
96          ((DateTimeFormatUtil.LiteralComponent) component).getText());
97    }

99    /**
100    * Assert that the given component is a
101    * {@link DateTimeFormatUtil.VariableMarkerComponent} with the expected field
102    * values.
103    *
104    * @param component
105    *          the component to check
106    * @param specifier
107    *          the expected component specifier character
108    * @param primaryModifier
109    *          the expected primary modifier string, or {@code null}
110    * @param secondModifier
111    *          the expected second modifier character, or {@code null}
112    * @param minWidth
113    *          the expected minimum width, or {@code null}
114    * @param maxWidth
115    *          the expected maximum width, or {@code null}
116    */
117   private static void assertMarker(
118       @NonNull DateTimeFormatUtil.FormatComponent component,
119       char specifier,
120       @Nullable String primaryModifier,
121       @Nullable Character secondModifier,
122       @Nullable Integer minWidth,
123       @Nullable Integer maxWidth) {
124     assertInstanceOf(DateTimeFormatUtil.VariableMarkerComponent.class, component);
125     DateTimeFormatUtil.VariableMarkerComponent marker
126         = (DateTimeFormatUtil.VariableMarkerComponent) component;
127     assertEquals(specifier, marker.getSpecifier(), "specifier");
128     assertEquals(primaryModifier, marker.getPrimaryModifier(), "primaryModifier");
129     assertEquals(secondModifier, marker.getSecondModifier(), "secondModifier");
130     assertEquals(minWidth, marker.getMinWidth(), "minWidth");
131     assertEquals(maxWidth, marker.getMaxWidth(), "maxWidth");
132   }

134   /**
135    * Format a temporal value using the given picture string, with default
136    * language, calendar, and place parameters.
137    *
138    * @param value
139    *          the temporal value to format
140    * @param picture
141    *          the picture string
142    * @param allowedMarkers
143    *          the set of allowed component specifier characters
144    * @return the formatted string
145    */
146   private static String format(
147       @NonNull ITemporalItem value,
148       @NonNull String picture,
149       @NonNull Set<Character> allowedMarkers) {
150     return DateTimeFormatUtil.formatDateTime(value, picture, null, null, null, allowedMarkers);
151   }

536   private static Stream<Arguments> provideYearFormats() {
537     return Stream.of(
538         // [Y] default - full year
539         Arguments.of(TEST_DATE, "[Y]", "2002"),
540         // [Y0001] - 4 digit padded
541         Arguments.of(TEST_DATE, "[Y0001]", "2002"),
542         // [Y01] - modulo 10^2 = last 2 digits (spec 9.8.4.4 modulo rule)
543         Arguments.of(TEST_DATE, "[Y01]", "02"),
544         // [Y,4-4] - padded to exactly 4 with width modifier
545         Arguments.of(IDateItem.valueOf("0005-01-01"), "[Y,4-4]", "0005"),
546         // [Y1] - single digit pattern = no truncation, full year
547         Arguments.of(TEST_DATE, "[Y1]", "2002"));
548   }
