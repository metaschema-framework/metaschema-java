11  import static org.junit.jupiter.api.Assertions.assertThrows;

13  import org.junit.jupiter.api.Test;

18  import java.util.stream.Stream;

20  import dev.metaschema.core.metapath.ExpressionTestBase;

25  
26  /**
27   * Tests for the fn:format-time function.
28   */
29  class FnFormatTimeTest
30      extends ExpressionTestBase {
31  
32    private static Stream<Arguments> provideValues() {
33      return Stream.of(
34          // W3C spec examples (section 9.8.5)
35          Arguments.of(
36              string("3:58 PM"),
37              "format-time(meta:time('15:58:45.762+02:00'), '[h]:[m01] [PN]', 'en', (), ())"),
38          Arguments.of(
39              string("3:58:45 pm"),
40              "format-time(meta:time('15:58:45.762+02:00'), '[h]:[m01]:[s01] [Pn]', 'en', (), ())"),
41          Arguments.of(
42              string("15:58"),
43              "format-time(meta:time('15:58:45.762+02:00'), '[H01]:[m01]')"),
44          Arguments.of(
45              string("15:58:45.762"),
46              "format-time(meta:time('15:58:45.762+02:00'), '[H01]:[m01]:[s01].[f001]')"),
47          // 12-hour with midnight
48          Arguments.of(
49              string("12:00 AM"),
50              "format-time(meta:time('00:00:00+00:00'), '[h]:[m01] [PN]', 'en', (), ())"),
51          // 12-hour with noon
52          Arguments.of(
53              string("12:00 PM"),
54              "format-time(meta:time('12:00:00+00:00'), '[h]:[m01] [PN]', 'en', (), ())"),
55          // Timezone formatting
56          Arguments.of(
57              string("+02:00"),
58              "format-time(meta:time('15:58:45+02:00'), '[Z]')"));
59    }
60  
61    @Test
62    void testEmptySequenceReturnsEmptySequence() {
63      // Spec: "If $value is the empty sequence, the function returns the empty
64      // sequence."
65      assertNull(
66          IMetapathExpression.compile("format-time((), '[H01]')")
67              .evaluateAs(null, IMetapathExpression.ResultType.ITEM, newDynamicContext()));
68    }
69  
70    @ParameterizedTest
71    @MethodSource("provideValues")
72    void testExpression(@NonNull IStringItem expected, @NonNull String metapath) {
73      assertEquals(
74          expected,
75          IMetapathExpression.compile(metapath)
76              .evaluateAs(null, IMetapathExpression.ResultType.ITEM, newDynamicContext()));
77    }
78  
79    @Test
80    void testDateMarkerRejectsForTime() {
81      // format-time should reject date-only markers (Y, M, D, d, F, W, w)
82      assertThrows(MetapathException.class, () -> {
83        IMetapathExpression.compile("format-time(meta:time('15:58:45+02:00'), '[Y0001]')")
84            .evaluateAs(null, IMetapathExpression.ResultType.ITEM, newDynamicContext());
85      });
86    }
87  
88    @Test
89    void testEraMarkerRejectsForTime() {
90      // Era (E) is "a baseline for the numbering of years" per W3C spec,
91      // so it is not available for xs:time values (FOFD1350)
92      assertThrows(MetapathException.class, () -> {
93        IMetapathExpression.compile("format-time(meta:time('15:58:45+02:00'), '[En]')")
94            .evaluateAs(null, IMetapathExpression.ResultType.ITEM, newDynamicContext());
95      });
96    }
97  }
