30  class ArrowExpressionTest
31      extends ExpressionTestBase {
32    private static Stream<Arguments> provideValues() { // NOPMD - false positive
33      return Stream.of(
34          Arguments.of(ISequence.of(string("ABC")), "'abc' => upper-case()"),
35          Arguments.of(ISequence.of(string("123")), "'1' => concat('2') => concat('3')"),
36          Arguments.of(ISequence.of(bool(true)), "() => $ex:var1()"));
37    }
38  
39    /**
40     * Tests the casting functionality using various input strings and target types.
41     * <p>
42     * The dynamic context is created fresh for each test case to ensure isolation.
43     *
44     * @param text
45     *          The input string to cast
46     * @param type
47     *          The target type to cast to
48     * @param expected
49     *          The expected result after casting
50     */
51    @ParameterizedTest
52    @MethodSource("provideValues")
53    void testArrowExpression(@NonNull ISequence<?> expected, @NonNull String metapath) {
54      StaticContext staticContext = StaticContext.builder()
55          .namespace("ex", NS)
56          .build();
57      DynamicContext dynamicContext = new DynamicContext(staticContext);
58      dynamicContext.bindVariableValue(IEnhancedQName.of(NS, "var1"), ISequence.of(string("fn:empty")));
59  
60      assertEquals(
61          expected,
62          IMetapathExpression.compile(metapath, staticContext).evaluate(null, dynamicContext));
63    }
64  
65    @Test
66    void testArrowExpressionWithUndefinedVariable() {
67      StaticContext staticContext = StaticContext.builder()
68          .namespace("ex", NS)
69          .build();
70      DynamicContext dynamicContext = new DynamicContext(staticContext);
71  
72      StaticMetapathException thrown = assertThrows(StaticMetapathException.class,
73          () -> IMetapathExpression.compile("() => $ex:undefined()", staticContext)
74              .evaluate(null, dynamicContext));
75      assertThat(thrown)
76          .isExactlyInstanceOf(StaticMetapathException.class)
77          .extracting(StaticMetapathException::getErrorCode)
78          .extracting(code -> code.getCode())
79          .isEqualTo(StaticMetapathException.NOT_DEFINED);
80    }
81  }
