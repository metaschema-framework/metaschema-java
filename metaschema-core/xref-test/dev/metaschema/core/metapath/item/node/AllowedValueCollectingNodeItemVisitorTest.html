234   void testAllowOtherFlagPreserved() {
235     MockedModelTestSupport mocking = new MockedModelTestSupport();
236     ISource source = ISource.externalSource(URI.create(TEST_NAMESPACE));
237 
238     IModule module = IModuleBuilder.builder()
239         .namespace(TEST_NAMESPACE)
240         .shortName("allow-other-test")
241         .version("1.0.0")
242         .source(source)
243         .assembly(mocking.assembly()
244             .name("root")
245             .rootName("root")
246             .flags(List.of(
247                 mocking.flag().name("strict"),
248                 mocking.flag().name("lax"))))
249         .toModule();
250 
251     IAssemblyDefinition rootDef = module.getAssemblyDefinitions().iterator().next();
252     IValueConstrained constraintSupport = rootDef.getConstraintSupport();
253     // Explicit allow-other=false (strict)
254     constraintSupport.addConstraint(
255         IAllowedValuesConstraint.builder()
256             .source(source)
257             .target(IMetapathExpression.compile("@strict"))
258             .allowedValue(IAllowedValue.of("x", MarkupLine.fromMarkdown("X"), null))
259             .allowsOther(false)
260             .build());
261     // Explicit allow-other=true (lax)
262     constraintSupport.addConstraint(
263         IAllowedValuesConstraint.builder()
264             .source(source)
265             .target(IMetapathExpression.compile("@lax"))
266             .allowedValue(IAllowedValue.of("y", MarkupLine.fromMarkdown("Y"), null))
267             .allowsOther(true)
268             .build());
269 
270     AllowedValueCollectingNodeItemVisitor visitor = new AllowedValueCollectingNodeItemVisitor();
271     visitor.visit(module);
272 
273     List<AllowedValuesRecord> allRecords = visitor.getAllowedValueLocations().stream()
274         .flatMap(loc -> loc.getAllowedValues().stream())
275         .collect(java.util.stream.Collectors.toList());
276     assertEquals(2, allRecords.size(), "Both constraints should be captured");
277 
278     boolean foundStrict = false;
279     boolean foundLax = false;
280     for (AllowedValuesRecord record : allRecords) {
281       String targetName = record.getTarget().getDefinition().getName();
282       if ("strict".equals(targetName)) {
283         foundStrict = true;
284         assertFalse(record.getAllowedValues().isAllowedOther(),
285             "allow-other=false should be preserved on the strict constraint");
286       } else if ("lax".equals(targetName)) {
287         foundLax = true;
288         assertTrue(record.getAllowedValues().isAllowedOther(),
289             "allow-other=true should be preserved on the lax constraint");
290       }
291     }
292     assertTrue(foundStrict, "Expected a record for the 'strict' flag");
293     assertTrue(foundLax, "Expected a record for the 'lax' flag");
294   }

476     MockedModelTestSupport mocking = new MockedModelTestSupport();
477     ISource source = ISource.externalSource(URI.create(TEST_NAMESPACE));
478 
479     IModule module = IModuleBuilder.builder()

502             null));

504         IAllowedValuesConstraint.builder()
505             .source(source)
506             .target(IMetapathExpression.compile("@href"))
507             .allowedValue(IAllowedValue.of("http://example.com/a", MarkupLine.fromMarkdown("A"), null))
508             .build());
509 
510     AllowedValueCollectingNodeItemVisitor visitor = new AllowedValueCollectingNodeItemVisitor();
511     visitor.visit(module);
512 
513     assertFalse(capturedAnyMessageContaining("Skipping let expression"),
514         "A let whose function-argument atomization would raise FOTY on a no-data flag"
515             + " must not produce a warning when the visitor enables"
516             + " METAPATH_ATOMIZE_NO_DATA_AS_EMPTY");
517     assertEquals(1, visitor.getAllowedValueLocations().size(),
518         "The allowed-values constraint targeting @href must still be collected");
519   }

521   @Test
522   void testTwoArgVisitOverloadEnablesModuleWalkFeaturesOnCallerContext() {
523     MockedModelTestSupport mocking = new MockedModelTestSupport();
524     ISource source = ISource.externalSource(URI.create(TEST_NAMESPACE));
525 
526     IModule module = IModuleBuilder.builder()
527         .namespace(TEST_NAMESPACE)
528         .shortName("two-arg-overload-test")
529         .version("1.0.0")
530         .source(source)
531         .assembly(mocking.assembly()
532             .name("root")
533             .rootName("root")
534             .flags(List.of(mocking.flag().name("href"))))
535         .toModule();
536 
537     IAssemblyDefinition rootDef = module.getAssemblyDefinitions().iterator().next();
538     rootDef.getConstraintSupport().addLetExpression(
539         ILet.of(
540             IEnhancedQName.of("resolved"),
541             IMetapathExpression.compile("upper-case(@href)"),
542             source,
543             null));
544     rootDef.getConstraintSupport().addConstraint(
545         IAllowedValuesConstraint.builder()
546             .source(source)
547             .target(IMetapathExpression.compile("@href"))
548             .allowedValue(IAllowedValue.of("http://example.com/a", MarkupLine.fromMarkdown("A"), null))
549             .build());
550 
551     // Callers of the two-arg overload (including the existing oscal-cli and
552     // metaschema-cli list-allowed-values commands) supply their own
553     // DynamicContext. The visitor must enable the module-walk features on
554     // that context itself so callers do not need to remember to do it, and
555     // do not observe FOTY warnings on OSCAL-style lets.
556     DynamicContext callerContext = new DynamicContext(
557         StaticContext.builder()
558             .defaultModelNamespace(module.getXmlNamespace())
559             .build());
560     // Explicitly set the feature flags to the opposite of their expected
561     // post-visit state. This proves that the two-arg overload actually flips
562     // them rather than relying on DynamicContext construction defaults.
563     callerContext.enablePredicateEvaluation();
564     callerContext.disableAtomizeNoDataAsEmpty();
565 
566     AllowedValueCollectingNodeItemVisitor visitor = new AllowedValueCollectingNodeItemVisitor();
567     visitor.visit(INodeItemFactory.instance().newModuleNodeItem(module), callerContext);
568 
569     assertFalse(capturedAnyMessageContaining("Skipping let expression"),
570         "The two-arg visit overload must enable METAPATH_ATOMIZE_NO_DATA_AS_EMPTY"
571             + " on the caller-supplied context so no FOTY warning is produced");
572     assertTrue(callerContext.getConfiguration()
573         .isFeatureEnabled(dev.metaschema.core.metapath.MetapathEvaluationFeature.METAPATH_ATOMIZE_NO_DATA_AS_EMPTY),
574         "The two-arg visit overload must enable METAPATH_ATOMIZE_NO_DATA_AS_EMPTY on the caller-supplied context");
575     assertFalse(callerContext.getConfiguration()
576         .isFeatureEnabled(dev.metaschema.core.metapath.MetapathEvaluationFeature.METAPATH_EVALUATE_PREDICATES),
577         "The two-arg visit overload must disable METAPATH_EVALUATE_PREDICATES on the caller-supplied context");
578   }
