6   package dev.metaschema.core.metapath.item.node;
7   
8   import static org.junit.jupiter.api.Assertions.assertEquals;
9   import static org.junit.jupiter.api.Assertions.assertFalse;
10  
11  import org.junit.jupiter.api.Test;
12  
13  import java.net.URI;
14  import java.util.List;
15  import java.util.Set;
16  
17  import dev.metaschema.core.model.IModule;
18  import dev.metaschema.core.model.ISource;
19  import dev.metaschema.core.testsupport.MockedModelTestSupport;
20  import dev.metaschema.core.testsupport.builder.IModuleBuilder;
21  
22  class RecursionCollectingNodeItemVisitorTest {
23  
24    private static final String TEST_NAMESPACE = "http://example.com/ns/recursion-test";
25  
26    @Test
27    void testAssemblyRecursion() {
28      // Build a module with a self-referencing assembly
29      MockedModelTestSupport mocking = new MockedModelTestSupport();
30      ISource source = ISource.externalSource(URI.create(TEST_NAMESPACE));
31  
32      IModule module = IModuleBuilder.builder()
33          .namespace(TEST_NAMESPACE)
34          .shortName("recursion-test")
35          .version("1.0.0")
36          .source(source)
37          .assembly(mocking.assembly()
38              .name("recursive-node")
39              .rootName("recursive-node")
40              .modelInstances(List.of(
41                  mocking.assemblyRef("recursive-node"))))
42          .toModule();
43  
44      RecursionCollectingNodeItemVisitor walker = new RecursionCollectingNodeItemVisitor();
45      walker.visit(module);
46      Set<RecursionCollectingNodeItemVisitor.AssemblyRecord> recursiveAssemblies
47          = walker.getRecursiveAssemblyDefinitions();
48  
49      // Verify that the recursive assembly was detected
50      assertFalse(recursiveAssemblies.isEmpty(), "Should detect recursive assembly");
51      assertEquals(1, recursiveAssemblies.size(), "Should have exactly one recursive assembly");
52  
53      RecursionCollectingNodeItemVisitor.AssemblyRecord record = recursiveAssemblies.iterator().next();
54      assertEquals("recursive-node", record.getDefinition().getName(),
55          "Recursive assembly should be 'recursive-node'");
56    }
57  
58  }
