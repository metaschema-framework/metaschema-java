178   @Test
179   void testThreadSafety() throws Exception {
180     TimingCollector collector = new TimingCollector();
181     ISource source = ISource.externalSource("https://example.com/module");
182 
183     int threadCount = 8;
184     int iterationsPerThread = 100;
185     ExecutorService executor = ObjectUtils.notNull(Executors.newFixedThreadPool(threadCount));
186     CountDownLatch startLatch = new CountDownLatch(1);
187     CountDownLatch doneLatch = new CountDownLatch(threadCount);
188 
189     try {
190       for (int t = 0; t < threadCount; t++) {
191         final int threadId = t;
192         executor.submit(() -> {
193           try {
194             startLatch.await();
195             // Each thread uses its own constraint but they share the collector
196             IExpectConstraint constraint = IExpectConstraint.builder()
197                 .source(source)
198                 .identifier("thread-constraint-" + threadId)
199                 .test(IMetapathExpression.compile("true()"))
200                 .build();
201 
202             INodeItem target = mock(INodeItem.class);
203 
204             for (int i = 0; i < iterationsPerThread; i++) {
205               collector.beforeConstraintEvaluation(constraint, target);
206               collector.afterConstraintEvaluation(constraint, target);
207             }
208           } catch (Exception e) {
209             // fail silently; assertions below will catch issues
210           } finally {
211             doneLatch.countDown();
212           }
213         });
214       }
215 
216       startLatch.countDown();
217       assertTrue(doneLatch.await(30, TimeUnit.SECONDS), "threads should complete");
218 
219       // Verify each thread's constraint timing is correct
220       for (int t = 0; t < threadCount; t++) {
221         TimingRecord record = collector.getConstraintTiming("thread-constraint-" + t);
222         assertNotNull(record, "timing for thread " + t + " should exist");
223         assertEquals(iterationsPerThread, record.getCount(),
224             "thread " + t + " should have correct count");
225       }
226     } finally {
227       executor.shutdown();
228     }
229   }

232   void testNestedConstraintAndLetEvents() {
233     TimingCollector collector = new TimingCollector();
234     ISource source = ISource.externalSource("https://example.com/module");
235 
236     IExpectConstraint constraint = IExpectConstraint.builder()
237         .source(source)
238         .identifier("outer-constraint")
239         .test(IMetapathExpression.compile("true()"))
240         .build();
241 
242     ILet let = ILet.of(
243         IEnhancedQName.of("nested-var"),
244         IMetapathExpression.compile("count(//item)"),
245         source,
246         null);
247 
248     INodeItem target = mock(INodeItem.class);
249 
250     // Simulate constraint evaluation that triggers let evaluation
251     collector.beforeConstraintEvaluation(constraint, target);
252     collector.beforeLetEvaluation(let);
253     collector.afterLetEvaluation(let);
254     collector.afterConstraintEvaluation(constraint, target);
255 
256     // Both should be recorded correctly
257     TimingRecord constraintRecord = collector.getConstraintTiming("outer-constraint");
258     assertNotNull(constraintRecord);
259     assertEquals(1, constraintRecord.getCount());
260 
261     TimingRecord letRecord = collector.getLetTiming(let);
262     assertNotNull(letRecord);
263     assertEquals(1, letRecord.getCount());
264 
265     // Constraint time should be >= let time (it contains the let)
266     assertTrue(constraintRecord.getTotalTimeNs() >= letRecord.getTotalTimeNs(),
267         "constraint time should include let time");
268   }
