46    @ParameterizedTest
47    @MethodSource("provideValuesForEncodeDecode")
48    void testEncodeDecodeEncode(@NonNull String expectedDecodedString, @NonNull String expectedEncodedString) {
49      // test encode to buffer
50      ByteBuffer encodedBuffer = MetaschemaDataTypeProvider.BASE64.encodeToByteBuffer(expectedDecodedString);
51      encodedBuffer.mark();
52      String encodedString = MetaschemaDataTypeProvider.BASE64.asString(encodedBuffer);
53      assertEquals(expectedEncodedString, encodedString);
54  
55      // test decode from buffer
56      ByteBuffer decodedBuffer = MetaschemaDataTypeProvider.BASE64.decode(encodedBuffer);
57      decodedBuffer.mark();
58      String decodedString = MetaschemaDataTypeProvider.BASE64.asString(decodedBuffer);
59      assertEquals(expectedDecodedString, decodedString);
60    }
