28  /**
29   * Supports generating a simple Metapath node item-based document model for use
30   * in unit tests.
31   */
32  public class MockedDocumentGenerator {
33    /**
34     * The document base URI used by documents generated by the
35     * {@link #generateDocumentNodeItem()} method.
36     */
37    @NonNull
38    public static final URI BASE_URI = ObjectUtils.notNull(URI.create("https://example.com/resource"));
39    /**
40     * The namespace used by Metaschema module definitions generated by the
41     * {@link #generateDocumentNodeItem()} method.
42     */
43    @NonNull
44    public static final String NS = "http://example.com/ns";
45    /**
46     * The qualified name for the root node ({@code /Q{http://example.com/ns}root})
47     * of a document generated by the {@link #generateDocumentNodeItem()} method.
48     */
49    @NonNull
50    public static final IEnhancedQName ROOT_QNAME = IEnhancedQName.of(NS, "root");
51    /**
52     * The qualified name for the assembly child node
53     * ({@code /Q{http://example.com/ns}root/Q{http://example.com/ns}assembly}) of a
54     * document generated by the {@link #generateDocumentNodeItem()} method.
55     */
56    @NonNull
57    public static final IEnhancedQName ASSEMBLY_QNAME = IEnhancedQName.of(NS, "assembly");
58    /**
59     * The qualified name for the field child node
60     * ({@code /Q{http://example.com/ns}root/Q{http://example.com/ns}field}) of a
61     * document generated by the {@link #generateDocumentNodeItem()} method.
62     */
63    @NonNull
64    public static final IEnhancedQName FIELD_QNAME = IEnhancedQName.of(NS, "field");
65    /**
66     * The qualified name for the flag child node
67     * ({@code /Q{http://example.com/ns}root/Q{http://example.com/ns}assembly/@field-flag})
68     * of a document generated by the {@link #generateDocumentNodeItem()} method.
69     */
70    @NonNull
71    public static final IEnhancedQName ASSEMBLY_FLAG_QNAME = IEnhancedQName.of("assembly-flag");
72    /**
73     * The qualified name for the flag child node
74     * ({@code /Q{http://example.com/ns}root/Q{http://example.com/ns}field/@field-flag})
75     * of a document generated by the {@link #generateDocumentNodeItem()} method.
76     */
77    @NonNull
78    public static final IEnhancedQName FIELD_FLAG_QNAME = IEnhancedQName.of("field-flag");
79    /**
80     * The qualified name for the flag node ({@code /Q{http://example.com/ns}flag})
81     * of a document generated by the {@link #generateDocumentNodeItem()} method.
82     */
83    @NonNull
84    public static final IEnhancedQName FLAG_QNAME = IEnhancedQName.of("flag");
85  
86    /**
87     * Generate a simple Metaschema-based document model.
88     *
89     * @return the document node item for this model
90     */
91    public static IDMDocumentNodeItem generateDocumentNodeItem() {
92      MockedModelTestSupport mocking = new MockedModelTestSupport();
93  
94      ISource moduleSource = ISource.externalSource("https://example.com/module");
95  
96      IAssemblyDefinition rootDefinition = mocking.assembly()
97          .qname(ROOT_QNAME)
98          .rootQName(ROOT_QNAME)
99          .source(moduleSource)
100         .toDefinition();
101     IAssemblyInstance assemblyInstance = mocking.assembly()
102         .qname(ASSEMBLY_QNAME)
103         .source(moduleSource)
104         .toInstance(rootDefinition);
105     IFlagInstance assemblyFlag = mocking.flag()
106         .qname(ASSEMBLY_FLAG_QNAME)
107         .source(moduleSource)
108         .toInstance(assemblyInstance.getDefinition());
109     IFieldInstance fieldInstance = mocking.field()
110         .qname(FIELD_QNAME)
111         .source(moduleSource)
112         .toInstance(rootDefinition);
113     IFlagInstance fieldFlag = mocking.flag()
114         .qname(FIELD_FLAG_QNAME)
115         .source(moduleSource)
116         .toInstance(fieldInstance.getDefinition());
117 
118     IDMDocumentNodeItem document = IDMDocumentNodeItem.newInstance(
119         BASE_URI,
120         rootDefinition);
121     IDMRootAssemblyNodeItem root = document.getRootAssemblyNodeItem();
122     IDMAssemblyNodeItem assembly = root.newAssembly(assemblyInstance);
123     assembly.newFlag(assemblyFlag, IStringItem.valueOf("assembly-flag"));
124     IDMFieldNodeItem field = root.newField(fieldInstance, IStringItem.valueOf("field"));
125     field.newFlag(fieldFlag, IStringItem.valueOf("field-flag"));
126     return document;
127   }
128 
129   /**
130    * Generate a simple Metaschema-based document model.
131    *
132    * @return the document node item for this model
133    */
134   public static IDMFlagNodeItem generateOrphanedFlagNodeItem() {
135     MockedModelTestSupport mocking = new MockedModelTestSupport();
136 
137     ISource moduleSource = ISource.externalSource("https://example.com/module");
138 
139     IFlagDefinition flagDefinition = mocking.flag()
140         .qname(FLAG_QNAME)
141         .source(moduleSource)
142         .toDefinition();
143 
144     return IDMFlagNodeItem.newInstance(flagDefinition, IStringItem.valueOf("flag"), moduleSource.getStaticContext());
145   }
