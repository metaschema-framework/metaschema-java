33  class CastFunctionTest
34      extends ExpressionTestBase {
35  
36    private static Stream<Arguments> provideValues() { // NOPMD - false positive
37      return Stream.of(
38          // Boolean cases
39          Arguments.of(string("ABCD0"), IBooleanItem.type(), bool(false)),
40          Arguments.of(string("true"), IBooleanItem.type(), bool(true)),
41          Arguments.of(string("0"), IBooleanItem.type(), bool(false)),
42          Arguments.of(string("1"), IBooleanItem.type(), bool(true)),
43          Arguments.of(string("yes"), IBooleanItem.type(), bool(false)),
44          Arguments.of(string("no"), IBooleanItem.type(), bool(false)),
45          Arguments.of(string("TRUE"), IBooleanItem.type(), bool(true)),
46          Arguments.of(string(""), IBooleanItem.type(), bool(false)),
47          Arguments.of(string("   "), IBooleanItem.type(), bool(false)),
48          // Integer cases
49          Arguments.of(string("1234567"), IIntegerItem.type(), integer(1234567)),
50          Arguments.of(string("-1234567"), IIntegerItem.type(), integer(-1234567)),
51          Arguments.of(string("0"), IIntegerItem.type(), integer(0)),
52          Arguments.of(
53              string(ObjectUtils.notNull(Integer.toString(Integer.MAX_VALUE))),
54              IIntegerItem.type(),
55              integer(Integer.MAX_VALUE)),
56          Arguments.of(
57              string(ObjectUtils.notNull(Integer.toString(Integer.MIN_VALUE))),
58              IIntegerItem.type(),
59              integer(Integer.MIN_VALUE)));
60    }
61  
62    @ParameterizedTest
63    @MethodSource("provideValues")
64    void testExpression(@NonNull IStringItem text, @NonNull IAtomicOrUnionType<?> type,
65        @NonNull IAnyAtomicItem expected) {
66      IAnyAtomicItem result = IMetapathExpression.compile(type.getQName().toEQName() + "('" + text.asString() + "')")
67          .evaluateAs(null, IMetapathExpression.ResultType.ITEM, newDynamicContext());
68      assertEquals(expected, result);
69    }
70  
71    @Test
72    void testInvalidCasts() {
73      assertThrows(MetapathException.class, () -> {
74        IMetapathExpression.compile(IIntegerItem.type().getQName().toEQName() + "('invalid')")
75            .evaluateAs(null, IMetapathExpression.ResultType.ITEM, newDynamicContext());
76      });
77    }
78  }
