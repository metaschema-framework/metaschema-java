22  import edu.umd.cs.findbugs.annotations.NonNull;
23  
24  class DefaultNodeItemFactoryTest
25      extends MockedModelTestSupport {
26    @NonNull
27    private static final String NS = ObjectUtils.notNull("http://example.com/ns");
28  
29    @Test
30    void testGenerateModelItems() {
31      IAssemblyDefinition assembly = assembly()
32          .namespace(NS)
33          .name("assembly1")
34          .source(ISource.externalSource("http://example.com/module"))
35          .flags(List.of(
36              flag().namespace(NS).name("flag1")))
37          .modelInstances(List.of(
38              field().namespace(NS).name("field1")))
39          .toDefinition();
40  
41      // Setup the value calls
42      StaticContext staticContext = StaticContext.instance();
43      IDMAssemblyNodeItem parentItem = IDMAssemblyNodeItem.newInstance(assembly, staticContext);
44      assembly.getFlagInstances()
45          .forEach(flag -> parentItem.newFlag(flag, IStringItem.valueOf(flag.getName() + " value")));
46      assembly.getFieldInstances()
47          .forEach(field -> {
48            parentItem.newField(field, IStringItem.valueOf(field.getName() + " value"));
49          });
50  
51      Collection<? extends IFlagNodeItem> flagItems = parentItem.getFlags();
52      Collection<? extends IModelNodeItem<?, ?>> modelItems = parentItem.modelItems()
53          .collect(Collectors.toUnmodifiableList());
54      assertAll(
55          () -> assertThat(flagItems).extracting(IFlagNodeItem::getQName)
56              .isEqualTo(List.of(IEnhancedQName.of(NS, "flag1"))),
57          () -> assertThat(flagItems).extracting(flag -> flag.toAtomicItem().asString())
58              .isEqualTo(List.of("flag1 value")),
59          () -> assertThat(modelItems).extracting(IModelNodeItem::getQName)
60              .isEqualTo(List.of(IEnhancedQName.of(NS, "field1"))),
61          () -> assertThat(modelItems).extracting(field -> field.toAtomicItem().asString())
62              .isEqualTo(List.of("field1 value")));
63    }
64  }
