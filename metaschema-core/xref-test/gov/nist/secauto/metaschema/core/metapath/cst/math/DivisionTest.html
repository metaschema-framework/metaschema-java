6   package gov.nist.secauto.metaschema.core.metapath.cst.math;
7   
8   import static gov.nist.secauto.metaschema.core.metapath.TestUtils.dayTimeDuration;
9   import static gov.nist.secauto.metaschema.core.metapath.TestUtils.decimal;
10  import static gov.nist.secauto.metaschema.core.metapath.TestUtils.yearMonthDuration;
11  import static org.junit.jupiter.api.Assertions.assertEquals;
12  
13  import gov.nist.secauto.metaschema.core.metapath.ExpressionTestBase;
14  import gov.nist.secauto.metaschema.core.metapath.IMetapathExpression;
15  import gov.nist.secauto.metaschema.core.metapath.item.atomic.IAnyAtomicItem;
16  
17  import org.junit.jupiter.params.ParameterizedTest;
18  import org.junit.jupiter.params.provider.Arguments;
19  import org.junit.jupiter.params.provider.MethodSource;
20  
21  import java.util.stream.Stream;
22  
23  import edu.umd.cs.findbugs.annotations.NonNull;
24  
25  public class DivisionTest
26      extends ExpressionTestBase {
27    private static Stream<Arguments> provideValues() {
28      return Stream.of(
29          Arguments.of(
30              dayTimeDuration("PT17H40M7S"),
31              "meta:day-time-duration('P1DT2H30M10.5S') div 1.5"),
32          Arguments.of(
33              decimal("1.437834967320261"),
34              "meta:day-time-duration('P2DT53M11S') div meta:day-time-duration('P1DT10H')"),
35          Arguments.of(
36              decimal("175991.0"),
37              "meta:day-time-duration('P2DT53M11S') div meta:day-time-duration('PT1S')"),
38          Arguments.of(
39              yearMonthDuration("P1Y11M"),
40              "meta:year-month-duration('P2Y11M') div 1.5"),
41          Arguments.of(
42              decimal("-2.5"),
43              "meta:year-month-duration('P3Y4M') div meta:year-month-duration('-P1Y4M')"),
44          Arguments.of(
45              decimal("40"),
46              "meta:year-month-duration('P3Y4M') div meta:year-month-duration('P1M')"),
47          Arguments.of(decimal("0"), "0 div 1"),
48          Arguments.of(decimal("1"), "1 div 1"),
49          Arguments.of(decimal("0.5"), "1 div 2"),
50          Arguments.of(decimal("1.0"), "1.0 div 1"),
51          Arguments.of(decimal("1.0"), "1 div 1.0"),
52          Arguments.of(decimal("1.0"), "1.0 div 1.0"));
53    }
54  
55    @ParameterizedTest
56    @MethodSource("provideValues")
57    void testExpression(@NonNull IAnyAtomicItem expected, @NonNull String metapath) {
58      IAnyAtomicItem result = IMetapathExpression.compile(metapath)
59          .evaluateAs(null, IMetapathExpression.ResultType.ITEM, newDynamicContext());
60      assertEquals(expected, result);
61    }
62  }
