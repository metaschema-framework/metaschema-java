54    @ParameterizedTest
55    @MethodSource("provideValuesForEncodeDecode")
56    void testEncodeDecodeEncode(@NonNull String expectedDecodedString, @NonNull String expectedEncodedString) {
57      assertAll(
58          // test encode
59          () -> assertEquals(
60              expectedEncodedString,
61              ObjectUtils.requireNonNull(
62                  ObjectUtils.requireNonNull((IBase64BinaryItem) IMetapathExpression
63                      .compile(ObjectUtils.notNull(String.format("fn:base64-encode-text('%s')", expectedDecodedString)))
64                      .evaluateAs(null, IMetapathExpression.ResultType.ITEM, newDynamicContext())))
65                  .asString(),
66              "Expected same encoded text"),
67          // test decode
68          () -> assertEquals(
69              expectedDecodedString,
70              ObjectUtils.requireNonNull(
71                  (IStringItem) IMetapathExpression
72                      .compile(ObjectUtils.notNull(
73                          String.format("fn:base64-decode-text(meta:base64('%s'))", expectedEncodedString)))
74                      .evaluateAs(null, IMetapathExpression.ResultType.ITEM, newDynamicContext()))
75                  .asString(),
76              "Expected same decoded text"));
77    }
