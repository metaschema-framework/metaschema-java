6   package gov.nist.secauto.metaschema.core.datatype.adapter;
7   
8   import static org.junit.jupiter.api.Assertions.assertAll;
9   import static org.junit.jupiter.api.Assertions.assertEquals;
10  
11  import org.junit.jupiter.params.ParameterizedTest;
12  import org.junit.jupiter.params.provider.Arguments;
13  import org.junit.jupiter.params.provider.MethodSource;
14  
15  import java.util.UUID;
16  import java.util.stream.Stream;
17  
18  import edu.umd.cs.findbugs.annotations.NonNull;
19  
20  class UuidAdapterTest {
21    private static Stream<Arguments> provideValues() {
22      return Stream.of(
23          Arguments.of(
24              "f0b23719-2687-407c-b22f-f4b7ee832571"),
25          Arguments.of(
26              "8eb2fb0f-47ac-40be-adb0-1a70fe7fad76"));
27    }
28  
29    @ParameterizedTest
30    @MethodSource("provideValues")
31    void testParseUUID(@NonNull String expected) {
32      UUID uuid = MetaschemaDataTypeProvider.UUID.parse(expected);
33      assertAll(
34          () -> assertEquals(expected, uuid.toString()),
35          () -> assertEquals(expected, MetaschemaDataTypeProvider.UUID.asString(uuid)));
36    }
37  }
