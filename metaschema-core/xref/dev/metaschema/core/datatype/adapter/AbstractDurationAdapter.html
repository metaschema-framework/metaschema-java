26  /**
27   * Supports working with duration-based data values.
28   *
29   * @param <TYPE>
30   *          the raw Java type this adapter supports
31   * @param <ITEM_TYPE>
32   *          the metapath item type supported by the adapter
33   */
34  public abstract class AbstractDurationAdapter<TYPE, ITEM_TYPE extends IDurationItem>
35      extends AbstractDataTypeAdapter<TYPE, ITEM_TYPE> {
36    private static final Pattern DURATION_PATTERN = Pattern.compile(
37        "^(?<sign>-)?"
38            + "P"
39            + "(?:(?<year>[0-9]+)Y)?"
40            + "(?:(?<month>[0-9]+)M)?"
41            + "(?:(?<day>[0-9]+)D)?"
42            + "(?:T"
43            + "(?:(?<hour>[0-9]+)H)?"
44            + "(?:(?<minute>[0-9]+)M)?"
45            + "(?:(?<second2>[0-9]+(?:\\.[0-9]+)?)S)?"
46            + ")?$");
47  
48    @NonNull
49    private static IDurationItem validate(
50        @NonNull String value,
51        @Nullable Period period,
52        @Nullable Duration duration) {
53  
54      if (period != null && duration != null) {
55        throw new IllegalArgumentException(
56            String.format("Invalid year/month or day/time duration value '%s'.",
57                value));
58      }
59  
60      IDurationItem retval;
61      if (period != null && duration == null) {
62        retval = IYearMonthDurationItem.valueOf(period);
63      } else if (period == null && duration != null) {
64        retval = IDayTimeDurationItem.valueOf(duration);
65      } else { // both are null or zero
66        retval = IDayTimeDurationItem.valueOf(ObjectUtils.notNull(Duration.ZERO));
67      }
68      return retval;
69    }
70  
71    /**
72     * Parse a set of text tokens as a period value.
73     *
74     * @param negative
75     *          {@code true} if the period is negative or {@code false} otherwise
76     * @param yearsFragment
77     *          an integer value indicating the amount of years in the duration
78     * @param monthsFragment
79     *          an integer value indicating the amount of months in the duration
80     * @return a period based on the amount of years and months
81     */
82    @NonNull
83    protected static Period parsePeriod(
84        boolean negative,
85        @Nullable String yearsFragment,
86        @Nullable String monthsFragment) {
87      int years = parseIntegerValue(yearsFragment);
88      int months = parseIntegerValue(monthsFragment);
89  
90      Period period = Period.of(years, months, 0);
91      return ObjectUtils.notNull(negative
92          // negate for '-' sign
93          ? period.negated()
94          : period);
95    }
96  
97    /**
98     * Parses the provided duration value.
99     *
100    * @param value
101    *          the value to parse
102    * @return the parsed duration
103    * @throws IllegalArgumentException
104    *           if the provided value is invalid
105    */
106   @NonNull
107   public static IDurationItem parseDuration(@NonNull String value) {
108     Matcher matcher = DURATION_PATTERN.matcher(value);
109 
110     // yearMonth; null is zero months
111     Period period = null;
112     // dayTime; null is zero seconds
113     Duration duration = null;
114 
115     if (matcher.matches()) {
116       boolean negative = matcher.group(1) != null;
117 
118       Period parsedPeriod = parsePeriod(
119           negative,
120           matcher.group(2),
121           matcher.group(3));
122       if (!Period.ZERO.equals(parsedPeriod)) {
123         period = parsedPeriod;
124       }
125 
126       Duration parsedDuration;
127       try {
128         parsedDuration = parseDuration(
129             negative,
130             matcher.group(4),
131             matcher.group(5),
132             matcher.group(6),
133             matcher.group(7));
134       } catch (ArithmeticException ex) {
135         throw new IllegalArgumentException(
136             String.format("Invalid duration value '%s'.", value),
137             ex);
138       }
139 
140       if (!Duration.ZERO.equals(parsedDuration)) {
141         duration = parsedDuration;
142       }
143     }
144 
145     return validate(value, period, duration);
146   }
147 
148   /**
149    * Parse a set of text tokens as a duration value.
150    *
151    * @param negative
152    *          {@code true} if the duration is negative or {@code false} otherwise
153    * @param daysFragment
154    *          an integer value indicating the amount of days in the duration
155    * @param hoursFragment
156    *          an integer value indicating the amount of hours in the duration
157    * @param minutesFragment
158    *          an integer value indicating the amount of minutes in the duration
159    * @param secondsFragment
160    *          a decimal value indicating the amount of fractional seconds in the
161    *          duration
162    * @return a duration based on the calculated seconds and fractional seconds
163    * @throws ArithmeticException
164    *           if the calculated value of seconds in the duration overflow an
165    *           integer value
166    */
167   @NonNull
168   protected static Duration parseDuration(
169       boolean negative,
170       @Nullable String daysFragment,
171       @Nullable String hoursFragment,
172       @Nullable String minutesFragment,
173       @Nullable String secondsFragment) {
174     int days = parseIntegerValue(daysFragment);
175     int hours = parseIntegerValue(hoursFragment);
176     int minutes = parseIntegerValue(minutesFragment);
177 
178     BigDecimal bigSeconds = parseDecimalValue(secondsFragment);
179     long totalSeconds = toWholeSeconds(bigSeconds);
180     totalSeconds = Math.addExact(totalSeconds, Math.multiplyExact(days, 86_400));
181     totalSeconds = Math.addExact(totalSeconds, Math.multiplyExact(hours, 3_600));
182     totalSeconds = Math.addExact(totalSeconds, Math.multiplyExact(minutes, 60));
183 
184     Duration duration = Duration.ofSeconds(
185         totalSeconds,
186         toFractionalNanoSeconds(bigSeconds));
187     return ObjectUtils.notNull(negative
188         // negate for '-' sign
189         ? duration.negated()
190         : duration);
191   }

193   /**
194    * Construct a new Java type adapter for a provided class.
195    *
196    * @param valueClass
197    *          the Java value object type this adapter supports
198    * @param itemClass
199    *          the Java type of the Matepath item this adapter supports
200    * @param castExecutor
201    *          the method to call to cast an item to an item based on this type
202    */
203   protected AbstractDurationAdapter(
204       @NonNull Class<TYPE> valueClass,
205       @NonNull Class<ITEM_TYPE> itemClass,
206       @NonNull AbstractAtomicOrUnionType.ICastExecutor<ITEM_TYPE> castExecutor) {
207     super(valueClass, itemClass, castExecutor);
208   }
209 
210   @Override
211   public JsonFormatTypes getJsonRawType() {
212     return JsonFormatTypes.STRING;
213   }
214 
215   private static int parseIntegerValue(@Nullable String value) {
216     return value == null ? 0 : Integer.parseInt(value);
217   }
218 
219   @NonNull
220   private static BigDecimal parseDecimalValue(@Nullable String value) {
221     return value == null ? ObjectUtils.notNull(BigDecimal.ZERO) : new BigDecimal(value, DecimalAdapter.mathContext());
222   }
223 
224   private static long toWholeSeconds(@NonNull BigDecimal bigSeconds) {
225     try {
226       return bigSeconds.toBigInteger().longValueExact();
227     } catch (ArithmeticException ex) {
228       ArithmeticException ex2 = new ArithmeticException(
229           String.format("Whole seconds '%s' is out of range for a long.", bigSeconds.toPlainString()));
230       ex2.addSuppressed(ex);
231       throw ex2;
232     }
233   }
234 
235   private static long toFractionalNanoSeconds(@NonNull BigDecimal seconds) {
236     BigDecimal remainder = seconds.remainder(BigDecimal.ONE, DecimalAdapter.mathContext());
237 
238     BigDecimal result = remainder.multiply(new BigDecimal("1e9", DecimalAdapter.mathContext()))
239         .setScale(0, RoundingMode.HALF_EVEN);
240     try {
241       return result.longValueExact();
242     } catch (ArithmeticException ex) {
243       ArithmeticException ex2 = new ArithmeticException(
244           String.format("Nano seconds '%s' is out of range for a long.", result.toPlainString()));
245       ex2.addSuppressed(ex);
246       throw ex2;
247     }
248   }
