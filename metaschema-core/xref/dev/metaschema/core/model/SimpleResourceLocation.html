6   package dev.metaschema.core.model;
7   
8   import com.fasterxml.jackson.core.JsonLocation;
9   
10  import javax.xml.stream.Location;
11  
12  import edu.umd.cs.findbugs.annotations.NonNull;
13  
14  /**
15   * A simple implementation of {@link IMetaschemaData} that stores location
16   * information from various parser sources.
17   * <p>
18   * This class can be used both for bound object metadata during parsing and for
19   * validation context in error messages.
20   */
21  public final class SimpleResourceLocation implements IMetaschemaData {
22    /** A constant representing an unknown location. */
23    @NonNull
24    public static final IMetaschemaData UNKNOWN = new SimpleResourceLocation(-1, -1, -1L, -1L);
25  
26    private final int line;
27    private final int column;
28    private final long charOffset;
29    private final long byteOffset;
30  
31    /**
32     * Construct a new resource location with the specified values.
33     *
34     * @param line
35     *          the line number (1-based), or -1 if unknown
36     * @param column
37     *          the column number (1-based), or -1 if unknown
38     * @param charOffset
39     *          the character offset (0-based), or -1 if unknown
40     * @param byteOffset
41     *          the byte offset (0-based), or -1 if unknown
42     */
43    public SimpleResourceLocation(int line, int column, long charOffset, long byteOffset) {
44      this.line = line;
45      this.column = column;
46      this.charOffset = charOffset;
47      this.byteOffset = byteOffset;
48    }
49  
50    /**
51     * Create a resource location from an XML stream location.
52     *
53     * @param location
54     *          the XML stream location, may be {@code null}
55     * @return a new resource location, or {@link #UNKNOWN} if the input is null
56     */
57    @NonNull
58    public static IMetaschemaData fromXmlLocation(Location location) {
59      if (location == null) {
60        return UNKNOWN;
61      }
62      return new SimpleResourceLocation(
63          location.getLineNumber(),
64          location.getColumnNumber(),
65          location.getCharacterOffset(),
66          -1L);
67    }
68  
69    /**
70     * Create a resource location from a Jackson JSON location.
71     *
72     * @param location
73     *          the JSON location, may be {@code null}
74     * @return a new resource location, or {@link #UNKNOWN} if the input is null
75     */
76    @NonNull
77    public static IMetaschemaData fromJsonLocation(JsonLocation location) {
78      if (location == null) {
79        return UNKNOWN;
80      }
81      return new SimpleResourceLocation(
82          location.getLineNr(),
83          location.getColumnNr(),
84          location.getCharOffset(),
85          location.getByteOffset());
86    }
87  
88    /**
89     * Create a resource location with just line and column information.
90     *
91     * @param line
92     *          the line number (1-based), or -1 if unknown
93     * @param column
94     *          the column number (1-based), or -1 if unknown
95     * @return a new resource location
96     */
97    @NonNull
98    public static IMetaschemaData of(int line, int column) {
99      return new SimpleResourceLocation(line, column, -1L, -1L);
100   }
101 
102   @Override
103   public int getLine() {
104     return line;
105   }
106 
107   @Override
108   public int getColumn() {
109     return column;
110   }
111 
112   @Override
113   public long getCharOffset() {
114     return charOffset;
115   }
116 
117   @Override
118   public long getByteOffset() {
119     return byteOffset;
120   }
121 
122   @Override
123   public String toString() {
124     StringBuilder sb = new StringBuilder();
125     if (line >= 0) {
126       sb.append(line);
127       if (column >= 0) {
128         sb.append(':').append(column);
129       }
130     } else {
131       sb.append("unknown");
132     }
133     return sb.toString();
134   }
135 }
