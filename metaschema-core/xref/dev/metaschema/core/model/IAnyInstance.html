6   package dev.metaschema.core.model;
7   
8   import java.util.Locale;
9   
10  import dev.metaschema.core.qname.IEnhancedQName;
11  import edu.umd.cs.findbugs.annotations.NonNull;
12  
13  /**
14   * A marker interface for an {@code any} instance in a Metaschema model.
15   * <p>
16   * An {@code any} instance represents unmodeled content that may appear within
17   * an assembly's model. Unlike other model instances, an {@code any} instance
18   * does not have a definition or name; it acts as a wildcard that captures
19   * content not explicitly declared by the model.
20   * <p>
21   * An {@code any} instance is always optional ({@link #getMinOccurs()} returns
22   * {@code 0}) and unbounded ({@link #getMaxOccurs()} returns {@code -1}).
23   */
24  public interface IAnyInstance extends IModelInstanceAbsolute {
25  
26    /**
27     * Provides the Metaschema model type of "ANY".
28     *
29     * @return the model type
30     */
31    @Override
32    default ModelType getModelType() {
33      return ModelType.ANY;
34    }
35  
36    @Override
37    default IAssemblyDefinition getContainingDefinition() {
38      return getParentContainer().getOwningDefinition();
39    }
40  
41    @Override
42    default int getMinOccurs() {
43      return 0;
44    }
45  
46    @Override
47    default int getMaxOccurs() {
48      return -1;
49    }
50  
51    @Override
52    default IEnhancedQName getEffectiveXmlGroupAsQName() {
53      // never grouped
54      return null;
55    }
56  
57    @Override
58    default boolean isEffectiveValueWrappedInXml() {
59      // any content is never wrapped
60      return false;
61    }
62  
63    @SuppressWarnings("null")
64    @Override
65    default String toCoordinates() {
66      return String.format("%s-instance:%s:%s@%d",
67          getModelType().toString().toLowerCase(Locale.ROOT),
68          getContainingDefinition().getContainingModule().getShortName(),
69          getContainingDefinition().getName(),
70          hashCode());
71    }
72  
73    /**
74     * A visitor callback.
75     *
76     * @param <CONTEXT>
77     *          the type of the context parameter
78     * @param <RESULT>
79     *          the type of the visitor result
80     * @param visitor
81     *          the calling visitor
82     * @param context
83     *          a parameter used to pass contextual information between visitors
84     * @return the visitor result
85     */
86    @Override
87    default <CONTEXT, RESULT> RESULT accept(@NonNull IModelElementVisitor<CONTEXT, RESULT> visitor, CONTEXT context) {
88      return visitor.visitAny(this, context);
89    }
90  }
