26  /**
27   * The base class for {@link IArrayItem} implementations, that provides an
28   * implementation of common methods.
29   *
30   * @param <ITEM>
31   *          the Java type of the items contained within the sequence
32   */
33  public abstract class AbstractArrayItem<ITEM extends ICollectionValue>
34      extends ImmutableCollections.AbstractImmutableDelegatedList<ITEM>
35      implements IArrayItem<ITEM>, IFeatureCollectionFunctionItem {
36    @NonNull
37    private static final IEnhancedQName QNAME = IEnhancedQName.of("array");
38    /**
39     * The function arguments, lazily initialized to prevent class initialization
40     * deadlock when multiple threads trigger class loading simultaneously.
41     */
42    @NonNull
43    private static final Lazy<List<IArgument>> ARGUMENTS = ObjectUtils.notNull(Lazy.of(() -> ObjectUtils.notNull(List.of(
44        IArgument.builder().name("position").type(IIntegerItem.type()).one().build()))));
45    /**
46     * An empty array item singleton, lazily initialized to prevent class
47     * initialization deadlock.
48     */
49    @NonNull
50    private static final Lazy<IArrayItem<?>> EMPTY = ObjectUtils.notNull(Lazy.of(ArrayItemN::new));
51  
52    /**
53     * Get an immutable array item that is empty.
54     *
55     * @param <T>
56     *          the item Java type
57     * @return the empty array item
58     */
59    @SuppressWarnings("unchecked")
60    @NonNull
61    public static <T extends ICollectionValue> IArrayItem<T> empty() {
62      return (IArrayItem<T>) EMPTY.get();
63    }
64  
65    @Override
66    public IEnhancedQName getQName() {
67      return QNAME;
68    }
69  
70    @Override
71    public List<IArgument> getArguments() {
72      return ARGUMENTS.get();
73    }
74  
75    @Override
76    public ISequence<?> execute(List<? extends ISequence<?>> arguments, DynamicContext dynamicContext,
77        ISequence<?> focus) {
78      ISequence<? extends IIntegerItem> arg = FunctionUtils.asType(
79          ObjectUtils.notNull(arguments.get(0)));
80  
81      IIntegerItem position = arg.getFirstItem(true);
82      if (position == null) {
83        return ISequence.empty();
84      }
85  
86      int index = position.toIntValueExact() - 1;
87      ICollectionValue result = getValue().get(index);
88      return result.toSequence();
89    }
90  
91    @Override
92    public int hashCode() {
93      return Objects.hash(getValue());
94    }
95  
96    @Override
97    public boolean equals(Object other) {
98      return other == this
99          || other instanceof IArrayItem && getValue().equals(((IArrayItem<?>) other).getValue());
100   }
101 
102   @Override
103   public boolean deepEquals(ICollectionValue other, DynamicContext dynamicContext) {
104     if (!(other instanceof IArrayItem)) {
105       return false;
106     }
107 
108     IArrayItem<?> otherArray = (IArrayItem<?>) other;
109     if (size() != otherArray.size()) {
110       return false;
111     }
112 
113     Iterator<? extends ICollectionValue> thisIterator = iterator();
114     Iterator<? extends ICollectionValue> otherIterator = otherArray.iterator();
115     boolean retval = true;
116     while (thisIterator.hasNext() && otherIterator.hasNext()) {
117       ICollectionValue i1 = thisIterator.next();
118       ICollectionValue i2 = otherIterator.next();
119       if (!i1.deepEquals(i2, dynamicContext)) {
120         retval = false;
121         break;
122       }
123     }
124     return retval;
125   }
126 
127   @Override
128   public String toSignature() {
129     return ObjectUtils.notNull(stream()
130         .map(ICollectionValue::toSignature)
131         .collect(Collectors.joining(",", "[", "]")));
132   }
133 
134   @Override
135   public String toString() {
136     return toSignature();
137   }
138 }
