6   package dev.metaschema.core.metapath.impl;
7   
8   import dev.metaschema.core.metapath.item.function.IDecimalMapKey;
9   import dev.metaschema.core.metapath.item.function.IMapKey;
10  
11  /**
12   * An implementation of a {@link IMapKey} that uses a string-based value.
13   */
14  public abstract class AbstractDecimalMapKey
15      extends AbstractMapKey
16      implements IDecimalMapKey {
17  
18    /**
19     * {@inheritDoc}
20     * <p>
21     * The hash code is based on the least precision to ensure this matches the
22     * logic in {@link #isSameKey(IMapKey)}.
23     */
24    @Override
25    protected int generateHashCode() {
26      return asDecimal().stripTrailingZeros().hashCode();
27    }
28  }
