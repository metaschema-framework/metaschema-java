6   package dev.metaschema.core.metapath.function.library;
7   
8   import java.time.Duration;
9   import java.util.List;
10  
11  import dev.metaschema.core.metapath.DynamicContext;
12  import dev.metaschema.core.metapath.MetapathConstants;
13  import dev.metaschema.core.metapath.function.FunctionUtils;
14  import dev.metaschema.core.metapath.function.IArgument;
15  import dev.metaschema.core.metapath.function.IFunction;
16  import dev.metaschema.core.metapath.item.IItem;
17  import dev.metaschema.core.metapath.item.ISequence;
18  import dev.metaschema.core.metapath.item.atomic.IDayTimeDurationItem;
19  import dev.metaschema.core.metapath.item.atomic.IDecimalItem;
20  import dev.metaschema.core.metapath.item.atomic.IDurationItem;
21  import dev.metaschema.core.metapath.item.atomic.IIntegerItem;
22  import dev.metaschema.core.metapath.item.atomic.IYearMonthDurationItem;
23  import edu.umd.cs.findbugs.annotations.NonNull;
24  
25  /**
26   * Implements the XPath 3.1 <a href=
27   * "https://www.w3.org/TR/xpath-functions-31/#func-seconds-from-duration">fn:seconds-from-duration</a>.
28   * function.
29   */
30  public final class FnSecondsFromDuration {
31    private static final String NAME = "seconds-from-duration";
32    @NonNull
33    static final IFunction SIGNATURE = IFunction.builder()
34        .name(NAME)
35        .namespace(MetapathConstants.NS_METAPATH_FUNCTIONS)
36        .deterministic()
37        .contextIndependent()
38        .focusIndependent()
39        .argument(IArgument.builder()
40            .name("arg")
41            .type(IDurationItem.type())
42            .zeroOrOne()
43            .build())
44        .returnType(IIntegerItem.type())
45        .returnZeroOrOne()
46        .functionHandler(FnSecondsFromDuration::execute)
47        .build();
48  
49    private FnSecondsFromDuration() {
50      // disable construction
51    }
52  
53    @SuppressWarnings("unused")
54    @NonNull
55    private static ISequence<IDecimalItem> execute(@NonNull IFunction function,
56        @NonNull List<ISequence<?>> arguments,
57        @NonNull DynamicContext dynamicContext,
58        IItem focus) {
59      IDurationItem arg = FunctionUtils.asTypeOrNull(arguments.get(0).getFirstItem(true));
60      return arg == null
61          // Per spec, return empty sequence if the arg is null
62          ? ISequence.empty()
63          : arg instanceof IYearMonthDurationItem
64              // year-month durations do not have hour granularity
65              ? ISequence.of(IIntegerItem.ZERO)
66              // get the hours
67              : ISequence.of(fnSecondsFromDuration((IDayTimeDurationItem) arg));
68    }
69  
70    /**
71     * Implements <a href=
72     * "https://www.w3.org/TR/xpath-functions-31/#func-seconds-from-duration">fn:seconds-from-duration</a>.
73     *
74     * @param arg
75     *          the meta:duration item from which to extract the second component
76     * @return the second component from the duration as a decimal item
77     */
78    @NonNull
79    public static IDecimalItem fnSecondsFromDuration(@NonNull IDayTimeDurationItem arg) {
80      Duration duration = arg.asDuration();
81      return IDecimalItem.valueOf(duration.getSeconds())
82          // remove the non-second quantity
83          .mod(IIntegerItem.valueOf(60))
84          // add the factional seconds
85          .add(IDecimalItem.valueOf(duration.getNano()).divide(IDecimalItem.valueOf(1_000_000_000)));
86    }
87  }
