6   package dev.metaschema.core.metapath;
7   
8   import dev.metaschema.core.configuration.AbstractConfigurationFeature;
9   import edu.umd.cs.findbugs.annotations.NonNull;
10  
11  /**
12   * Provides a mechanism to configure Metapath evaluation settings.
13   *
14   * @param <V>
15   *          the feature value Java type
16   */
17  public final class MetapathEvaluationFeature<V>
18      extends AbstractConfigurationFeature<V> {
19    /**
20     * If enabled, evaluate <a href=
21     * "https://www.w3.org/TR/xpath-31/#id-filter-expression">predicates</a>,
22     * otherwise skip evaluating them.
23     */
24    @NonNull
25    public static final MetapathEvaluationFeature<Boolean> METAPATH_EVALUATE_PREDICATES
26        = new MetapathEvaluationFeature<>("evaluate-predicates", Boolean.class, true);
27  
28    /**
29     * If enabled, atomization of a node item that has no associated typed value
30     * (for example, a flag or field node reached while walking a module definition
31     * rather than an instance document) yields a {@code null} atomic value instead
32     * of raising
33     * {@link dev.metaschema.core.metapath.function.InvalidTypeFunctionException}
34     * with code
35     * {@link dev.metaschema.core.metapath.function.InvalidTypeFunctionException#NODE_HAS_NO_TYPED_VALUE}.
36     * <p>
37     * This is intended for visitors and tools that traverse an
38     * {@link dev.metaschema.core.metapath.item.node.IModuleNodeItem} graph and need
39     * downstream function calls (for example {@code fn:resolve-uri} or
40     * {@code fn:doc}) to degrade gracefully when they receive a no-data flag rather
41     * than an instance value.
42     */
43    @NonNull
44    public static final MetapathEvaluationFeature<Boolean> METAPATH_ATOMIZE_NO_DATA_AS_EMPTY
45        = new MetapathEvaluationFeature<>("atomize-no-data-as-empty", Boolean.class, false);
46  
47    private MetapathEvaluationFeature(
48        @NonNull String name,
49        @NonNull Class<V> valueClass,
50        @NonNull V defaultValue) {
51      super(name, valueClass, defaultValue);
52    }
53  }
