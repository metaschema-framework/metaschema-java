8   import java.util.LinkedHashMap;
9   import java.util.Map;
10

12  import dev.metaschema.core.metapath.cst.IExpressionVisitor;
13  import dev.metaschema.core.metapath.function.FunctionUtils;
14  import dev.metaschema.core.metapath.function.impl.OperationFunctions;
15  import dev.metaschema.core.metapath.item.atomic.IAnyAtomicItem;
16  import dev.metaschema.core.metapath.item.atomic.IDayTimeDurationItem;
17  import dev.metaschema.core.metapath.item.atomic.INumericItem;
18  import dev.metaschema.core.metapath.item.atomic.IYearMonthDurationItem;
19  import dev.metaschema.core.metapath.type.InvalidTypeMetapathException;
20  import dev.metaschema.core.util.CollectionUtil;

24  /**
25   * An XPath 3.1
26   * <a href="https://www.w3.org/TR/xpath-31/#id-arithmetic">arithmetic
27   * expression</a> supporting multiplication.
28   * <p>
29   * Supports multiplication between:
30   * <ul>
31   * <li>Numeric values
32   * <li>YearMonthDuration × Numeric
33   * <li>DayTimeDuration × Numeric
34   * </ul>
35   *
36   * <p>
37   * Numeric operands are automatically converted using
38   * {@link FunctionUtils#toNumeric}.
39   */
40  public class Multiplication
41      extends AbstractBasicArithmeticExpression {
42    @NonNull
43    private static final Map<Class<? extends IAnyAtomicItem>,
44        Map<Class<? extends IAnyAtomicItem>, OperationStrategy>> MULTIPLICATION_STRATEGIES = generateStrategies();
45  
46    /**
47     * An expression that gets the product result by multiplying two values.
48     *
49     * @param text
50     *          the parsed text of the expression
51     * @param left
52     *          the item to be divided
53     * @param right
54     *          the item to divide by
55     */
56    public Multiplication(
57        @NonNull String text,
58        @NonNull IExpression left,
59        @NonNull IExpression right) {
60      super(text, left, right);
61    }
62  
63    @Override
64    public <RESULT, CONTEXT> RESULT accept(IExpressionVisitor<RESULT, CONTEXT> visitor, CONTEXT context) {
65      return visitor.visitMultiplication(this, context);
66    }
67  
68    @Override
69    protected Map<
70        Class<? extends IAnyAtomicItem>,
71        Map<Class<? extends IAnyAtomicItem>, OperationStrategy>> getStrategies() {
72      return MULTIPLICATION_STRATEGIES;
73    }
74  
75    @Override
76    protected String unsupportedMessage(String dividend, String divisor) {
77      return ObjectUtils.notNull(String.format("Multiplication of '%s' by '%s' is not supported.", dividend, divisor));
78    }
79  
80    @Override
81    protected INumericItem operationAsNumeric(INumericItem dividend, INumericItem divisor) {
82      // Default to numeric multiplication
83      return OperationFunctions.opNumericMultiply(dividend, divisor);
84    }
85  
86    @SuppressWarnings("PMD.UseConcurrentHashMap")
87    @NonNull
88    private static Map<
89        Class<? extends IAnyAtomicItem>,
90        Map<Class<? extends IAnyAtomicItem>, OperationStrategy>> generateStrategies() {
91      Map<Class<? extends IAnyAtomicItem>, Map<Class<? extends IAnyAtomicItem>, OperationStrategy>> strategies
92          = new LinkedHashMap<>();
93  
94      // IYearMonthDurationItem strategies
95      Map<Class<? extends IAnyAtomicItem>, OperationStrategy> typeStrategies = new LinkedHashMap<>();
96      typeStrategies.put(INumericItem.class,
97          (dividend, divisor, dynamicContext) -> OperationFunctions.opMultiplyYearMonthDuration(
98              (IYearMonthDurationItem) dividend,
99              (INumericItem) divisor));
100     strategies.put(IYearMonthDurationItem.class, CollectionUtil.unmodifiableMap(typeStrategies));
101 
102     // IDayTimeDurationItem strategies
103     typeStrategies = new LinkedHashMap<>();
104     typeStrategies.put(INumericItem.class,
105         (dividend, divisor, dynamicContext) -> OperationFunctions.opMultiplyDayTimeDuration(
106             (IDayTimeDurationItem) dividend,
107             (INumericItem) divisor));
108     strategies.put(IDayTimeDurationItem.class, CollectionUtil.unmodifiableMap(typeStrategies));
109 
110     // INumericItem strategies
111     typeStrategies = new LinkedHashMap<>();
112     typeStrategies.put(INumericItem.class,
113         (dividend, divisor, dynamicContext) -> OperationFunctions.opNumericMultiply(
114             (INumericItem) dividend,
115             (INumericItem) divisor));
116     typeStrategies.put(IYearMonthDurationItem.class,
117         (dividend, divisor, dynamicContext) -> OperationFunctions.opMultiplyYearMonthDuration(
118             (IYearMonthDurationItem) divisor,
119             (INumericItem) dividend));
120     typeStrategies.put(IDayTimeDurationItem.class,
121         (dividend, divisor, dynamicContext) -> OperationFunctions.opMultiplyDayTimeDuration(
122             (IDayTimeDurationItem) divisor,
123             (INumericItem) dividend));
124     strategies.put(INumericItem.class, CollectionUtil.unmodifiableMap(typeStrategies));
125 
126     return CollectionUtil.unmodifiableMap(strategies);
127   }
128 
129   /**
130    * Get the sum of two atomic items.
131    *
132    * @param leftItem
133    *          the first item to multiply
134    * @param rightItem
135    *          the second item to multiply
136    * @return the product of both items
137    * @throws InvalidTypeMetapathException
138    *           for unsupported operand combinations.
139    */
140   // TODO: Is this used?
141   @SuppressWarnings("PMD.CyclomaticComplexity")
142   @NonNull
143   public static IAnyAtomicItem multiply(
144       @NonNull IAnyAtomicItem leftItem,
145       @NonNull IAnyAtomicItem rightItem) {
146     IAnyAtomicItem retval = null;
147     if (leftItem instanceof IYearMonthDurationItem) {
148       IYearMonthDurationItem left = (IYearMonthDurationItem) leftItem;
149       if (rightItem instanceof INumericItem) {
150         retval = OperationFunctions.opMultiplyYearMonthDuration(left, (INumericItem) rightItem);
151       }
152     } else if (leftItem instanceof IDayTimeDurationItem) {
153       IDayTimeDurationItem left = (IDayTimeDurationItem) leftItem;
154       if (rightItem instanceof INumericItem) {
155         retval = OperationFunctions.opMultiplyDayTimeDuration(left, (INumericItem) rightItem);
156       }
157     } else {
158       // handle as numeric
159       INumericItem left = FunctionUtils.castToNumeric(leftItem);
160       if (rightItem instanceof INumericItem) {
161         INumericItem right = FunctionUtils.castToNumeric(rightItem);
162         retval = OperationFunctions.opNumericMultiply(left, right);
163       } else if (rightItem instanceof IYearMonthDurationItem) {
164         retval = OperationFunctions.opMultiplyYearMonthDuration((IYearMonthDurationItem) rightItem, left);
165       } else if (rightItem instanceof IDayTimeDurationItem) {
166         retval = OperationFunctions.opMultiplyDayTimeDuration((IDayTimeDurationItem) rightItem, left);
167       }
168     }
169     if (retval == null) {
170       throw new InvalidTypeMetapathException(
171           null,
172           String.format("Multiplication between %s and %s is not supported.",
173               leftItem.toSignature(),
174               rightItem.toSignature()));
175     }
176     return retval;
177   }
