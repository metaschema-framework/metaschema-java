57    /**
58     * Parse the provided sequence type.
59     *
60     * @param ctx
61     *          the sequence type parse tree
62     * @param staticContext
63     *          the static context used to resolve types and other information
64     * @return the parse sequence type
65     */
66    @NonNull
67    public static ISequenceType parseSequenceType(
68        @NonNull Metapath10.SequencetypeContext ctx,
69        @NonNull StaticContext staticContext) {
70      return ObjectUtils.notNull(ctx.KW_EMPTY_SEQUENCE() == null
71          ? parseSequenceType(
72              ObjectUtils.requireNonNull(ctx.itemtype()),
73              ctx.occurrenceindicator(),
74              staticContext)
75          : ISequenceType.empty());
76    }
77  
78    private static ISequenceType parseSequenceType(
79        @NonNull Metapath10.ItemtypeContext itemTypeCtx,
80        @Nullable Metapath10.OccurrenceindicatorContext occurrenceIndicatorCtx,
81        @NonNull StaticContext staticContext) {
82      IItemType itemType = parseItemType(itemTypeCtx, staticContext);
83      Occurrence occurrence = parseOccurrence(occurrenceIndicatorCtx);
84      return ISequenceType.of(itemType, occurrence);
85    }

87    @NonNull
88    private static IItemType parseItemType(
89        @NonNull Metapath10.ItemtypeContext ctx,
90        @NonNull StaticContext staticContext) {
91      IItemType retval;
92      if (ctx.KW_ITEM() == null) {
93        ParseTree tree = ctx.getChild(0);
94        retval = ITEM_TYPE_HANDLER_MAP.get(tree.getClass()).parse(tree, staticContext);
95      } else {
96        retval = IItemType.item();
97      }
98      return retval;
99    }

101   /**
102    * Parses a kind test using the provided abstract syntax tree (AST).
103    *
104    * @param tree
105    *          the AST
106    * @param staticContext
107    *          the static context used to resolve types and other information
108    * @return the kind test
109    * @throws IllegalArgumentException
110    *           if the tree structure contains an unsupported kind test type
111    */
112   @NonNull
113   public static IItemType parseKindTest(
114       @NonNull ParseTree tree,
115       @NonNull StaticContext staticContext) {
116     ParseTree child = ObjectUtils.requireNonNull(tree.getChild(0));
117     ParseTreeParser<? extends IKindTest<?>> handler = KIND_TEST_HANDLER_MAP.get(child.getClass());
118     if (handler == null) {
119       throw new IllegalArgumentException("Unsupported kind test type: " + child.getClass());
120     }
121     return handler.parse(child, staticContext);
122   }

135   @NonNull
136   private static IKindTest<IAssemblyNodeItem> parseKindAssemblyTest(
137       @NonNull ParseTree tree,
138       @NonNull StaticContext staticContext) {
139 
140     Metapath10.AssemblytestContext ctx = (Metapath10.AssemblytestContext) tree;
141     IEnhancedQName instanceName
142         = ctx.elementnameorwildcard() == null || ctx.elementnameorwildcard().elementname() == null
143             ? null
144             : staticContext.parseModelName(ObjectUtils.notNull(
145                 ctx.elementnameorwildcard().elementname().eqname().getText()));
146     String typeName = ctx.typename_() == null ? null : ctx.typename_().eqname().getText();
147     return instanceName == null
148         ? typeName == null
149             ? IItemType.assembly()
150             : IItemType.assembly(typeName, staticContext)
151         : IItemType.assembly(instanceName, typeName, staticContext);
152   }

209   @NonNull
210   private static IItemType parseMapTest(
211       @NonNull ParseTree tree,
212       @NonNull StaticContext staticContext) {
213     Metapath10.MaptestContext ctx = (Metapath10.MaptestContext) tree;
214 
215     ctx.anymaptest();
216     ctx.typedmaptest();
217 
218     IItemType retval;
219     if (ctx.anymaptest() != null) {
220       retval = IItemType.map();
221     } else {
222       Metapath10.TypedmaptestContext typedMapCtx = ctx.typedmaptest();
223 
224       String dataTypeName = ObjectUtils.notNull(typedMapCtx.atomicoruniontype().getText());
225       IAtomicOrUnionType<?> dataType = staticContext.lookupAtomicType(dataTypeName);
226       retval = IItemType.map(
227           dataType,
228           parseSequenceType(ObjectUtils.requireNonNull(typedMapCtx.sequencetype()), staticContext));
229     }
230     return retval;
231   }

268   /**
269    * Parse the occurrence indicator.
270    *
271    * @param ctx
272    *          the occurrence parse tree
273    * @return the parsed occurrence value
274    */
275   @NonNull
276   public static Occurrence parseOccurrence(@Nullable Metapath10.OccurrenceindicatorContext ctx) {
277     Occurrence retval;
278     if (ctx == null) {
279       retval = Occurrence.ONE;
280     } else {
281       int type = ((TerminalNode) ctx.getChild(0)).getSymbol().getType();
282       switch (type) {
283       case Metapath10Lexer.QM:
284         retval = Occurrence.ZERO_OR_ONE;
285         break;
286       case Metapath10Lexer.STAR:
287         retval = Occurrence.ZERO_OR_MORE;
288         break;
289       case Metapath10Lexer.PLUS:
290         retval = Occurrence.ONE_OR_MORE;
291         break;
292       default:
293         throw new UnsupportedOperationException(((TerminalNode) ctx.getChild(0)).getSymbol().getText());
294       }
295     }
296     return retval;
297   }
