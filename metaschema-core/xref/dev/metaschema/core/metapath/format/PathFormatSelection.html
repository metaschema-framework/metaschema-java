6   package dev.metaschema.core.metapath.format;
7   
8   /**
9    * Enumeration of path format selection options for validation output.
10   * <p>
11   * This enum allows users to control how node paths are formatted in validation
12   * messages and other output. The selection can be explicit (forcing a specific
13   * format) or automatic (selecting based on document format).
14   *
15   * @see IPathFormatter#METAPATH_PATH_FORMATER
16   * @see IPathFormatter#XPATH_PATH_FORMATTER
17   * @see IPathFormatter#JSON_POINTER_PATH_FORMATTER
18   */
19  public enum PathFormatSelection {
20    /**
21     * Auto-select the path format based on document format.
22     * <p>
23     * When this option is selected:
24     * <ul>
25     * <li>JSON documents use JSON Pointer format (RFC 6901)</li>
26     * <li>YAML documents use JSON Pointer format (RFC 6901)</li>
27     * <li>XML documents use XPath 3.1 EQName format</li>
28     * </ul>
29     * <p>
30     * This is the default selection when no explicit format is specified.
31     */
32    AUTO,
33  
34    /**
35     * Always use Metapath format regardless of document type.
36     * <p>
37     * Produces paths like: {@code /root/assembly[1]/field[1]/@flag}
38     */
39    METAPATH,
40  
41    /**
42     * Always use XPath 3.1 EQName format regardless of document type.
43     * <p>
44     * Produces namespace-qualified paths like:
45     * {@code /Q{http://example.com}root/Q{http://example.com}assembly[1]}
46     */
47    XPATH,
48  
49    /**
50     * Always use RFC 6901 JSON Pointer format regardless of document type.
51     * <p>
52     * Produces paths like: {@code /root/assemblies/0/id}
53     */
54    JSON_POINTER
55  }
