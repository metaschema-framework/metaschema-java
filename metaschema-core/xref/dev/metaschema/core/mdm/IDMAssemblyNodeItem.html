6   package dev.metaschema.core.mdm;
7   
8   import dev.metaschema.core.mdm.impl.DefinitionAssemblyNodeItem;
9   import dev.metaschema.core.mdm.impl.IDMModelNodeItem;
10  import dev.metaschema.core.metapath.StaticContext;
11  import dev.metaschema.core.metapath.item.atomic.IAnyAtomicItem;
12  import dev.metaschema.core.metapath.item.node.IAssemblyNodeItem;
13  import dev.metaschema.core.model.IAssemblyDefinition;
14  import dev.metaschema.core.model.IAssemblyInstance;
15  import dev.metaschema.core.model.IFieldInstance;
16  import edu.umd.cs.findbugs.annotations.NonNull;
17  
18  /**
19   * Represents a Metapath assembly node item that is backed by a simple
20   * Metaschema module-based data model.
21   * <p>
22   * The {@link #newInstance(IAssemblyDefinition, StaticContext)} method can be
23   * used to create a node from an {@link IAssemblyDefinition} that is orphaned
24   * from a document model.
25   * <p>
26   * Child nodes can be created using the
27   * {@link #newFlag(dev.metaschema.core.model.IFlagInstance, IAnyAtomicItem)},
28   * {@link #newAssembly(IAssemblyInstance)}, and
29   * {@link #newField(IFieldInstance, IAnyAtomicItem)} methods. These children are
30   * added to this assembly.
31   */
32  public interface IDMAssemblyNodeItem
33      extends IAssemblyNodeItem, IDMModelNodeItem<IAssemblyDefinition, IAssemblyInstance> {
34    /**
35     * Create new assembly node item that is detached from a parent node item.
36     *
37     * @param definition
38     *          the Metaschema field definition describing the assembly
39     * @param staticContext
40     *          the atomic field value
41     * @return the new field node item
42     */
43    @NonNull
44    static IDMAssemblyNodeItem newInstance(
45        @NonNull IAssemblyDefinition definition,
46        @NonNull StaticContext staticContext) {
47      return new DefinitionAssemblyNodeItem(definition, staticContext);
48    }
49  
50    /**
51     * Create and add a new field to the underlying data model.
52     *
53     * @param instance
54     *          the Metaschema field instance describing the field
55     * @param value
56     *          the atomic field value
57     * @return the new field node item
58     */
59    @NonNull
60    IDMFieldNodeItem newField(
61        @NonNull IFieldInstance instance,
62        @NonNull IAnyAtomicItem value);
63  
64    /**
65     * Create and add a new assembly to the underlying data model.
66     *
67     * @param instance
68     *          the Metaschema assembly instance describing the assembly
69     * @return the new assembly node item
70     */
71    @NonNull
72    IDMAssemblyNodeItem newAssembly(
73        @NonNull IAssemblyInstance instance);
74  }
