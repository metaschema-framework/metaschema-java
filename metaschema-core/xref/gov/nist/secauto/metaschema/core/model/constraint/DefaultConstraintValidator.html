263   /**
264    * Evaluates the provided {@code constraint} against each of the
265    * {@code targets}.
266    *
267    * @param constraint
268    *          the constraint to execute
269    * @param node
270    *          the original focus of Metapath evaluation for identifying the
271    *          targets
272    * @param targets
273    *          the focus of Metapath evaluation for evaluating any constraint
274    *          Metapath clauses
275    * @throws ConstraintValidationException
276    *           if an unexpected error occurred while validating a constraint
277    */
278   private void validateHasCardinality(
279       @NonNull ICardinalityConstraint constraint,
280       @NonNull IAssemblyNodeItem node,
281       @NonNull ISequence<? extends INodeItem> targets,
282       @NonNull DynamicContext dynamicContext) throws ConstraintValidationException {
283     int itemCount = targets.size();
284 
285     IConstraintValidationHandler handler = getConstraintValidationHandler();
286 
287     boolean violation = false;
288     Integer minOccurs = constraint.getMinOccurs();
289     if (minOccurs != null && itemCount < minOccurs) {
290       handler.handleCardinalityMinimumViolation(constraint, node, targets, dynamicContext);
291       violation = true;
292     }
293 
294     Integer maxOccurs = constraint.getMaxOccurs();
295     if (maxOccurs != null && itemCount > maxOccurs) {
296       handler.handleCardinalityMaximumViolation(constraint, node, targets, dynamicContext);
297       violation = true;
298     }
299 
300     if (!violation) {
301       handlePass(constraint, node, node, dynamicContext);
302     }
303   }

353   private void validateIndex(
354       @NonNull IIndexConstraint constraint,
355       @NonNull IAssemblyNodeItem node,
356       @NonNull ISequence<? extends INodeItem> targets,
357       @NonNull DynamicContext dynamicContext) throws ConstraintValidationException {
358     IConstraintValidationHandler handler = getConstraintValidationHandler();
359     if (indexNameToIndexMap.containsKey(constraint.getName())) {
360       handler.handleIndexDuplicateViolation(constraint, node, dynamicContext);
361     } else {
362       validateIndexEntries(constraint, node, targets, dynamicContext);
363     }

366   private void validateIndexEntries(
367       @NonNull IIndexConstraint constraint,
368       @NonNull IAssemblyNodeItem node,
369       @NonNull ISequence<? extends INodeItem> targets,
370       @NonNull DynamicContext dynamicContext) throws ConstraintValidationException {
371     IIndex index = IIndex.newInstance(constraint.getKeyFields());
372     for (INodeItem item : targets) {
373       assert item != null;
374       if (item.hasValue()) {
375         INodeItem oldItem = null;
376         try {
377           oldItem = index.put(item, IIndex.toKey(item, index.getKeyFields(), dynamicContext));
378         } catch (IllegalArgumentException ex) {
379           // throw by IIndex.toKey
380           handleError(constraint, item, ex, dynamicContext);
381         }
382         try {
383           if (oldItem == null) {
384             handlePass(constraint, node, item, dynamicContext);
385           } else {
386             handler.handleIndexDuplicateKeyViolation(constraint, node, oldItem, item, dynamicContext);
387           }
388         } catch (MetapathException ex) {
389           handler.handleKeyMatchError(constraint, node, item, ex, dynamicContext);
390         }
391       }
392     }
393     indexNameToIndexMap.put(constraint.getName(), index);
394   }

406   private void handleError(
407       @NonNull IConstraint constraint,
408       @NonNull INodeItem node,
409       @NonNull Throwable ex,
410       @NonNull DynamicContext dynamicContext) throws ConstraintValidationException {
411     if (isFeatureEnabled(ValidationFeature.THROW_EXCEPTION_ON_ERROR)) {
412       if (ex instanceof ConstraintValidationException) {
413         throw (ConstraintValidationException) ex;
414       }
415       throw new ConstraintValidationException(ex);
416     }
417     getConstraintValidationHandler()
418         .handleError(constraint, node, toErrorMessage(constraint, node, ex), ex, dynamicContext);
419   }

421   @NonNull
422   private static String toErrorMessage(
423       @NonNull IConstraint constraint,
424       @NonNull INodeItem item,
425       @NonNull Throwable ex) {
426     StringBuilder builder = new StringBuilder(128);
427     builder.append("A ")
428         .append(constraint.getType().getName())
429         .append(" constraint");
430 
431     String id = constraint.getId();
432     if (id == null) {
433       builder.append(" targeting the metapath '")
434           .append(constraint.getTarget().getPath())
435           .append('\'');
436     } else {
437       builder.append(" with id '")
438           .append(id)
439           .append('\'');
440     }
441 
442     builder.append(", matching the item at path '")
443         .append(item.getMetapath())
444         .append("', resulted in an unexpected error. ")
445         .append(ex.getLocalizedMessage());
446     return ObjectUtils.notNull(builder.toString());
447   }

497   private void validateUnique(
498       @NonNull IUniqueConstraint constraint,
499       @NonNull IAssemblyNodeItem node,
500       @NonNull ISequence<? extends INodeItem> targets,
501       @NonNull DynamicContext dynamicContext) throws ConstraintValidationException {
502 
503     IConstraintValidationHandler handler = getConstraintValidationHandler();
504     IIndex index = IIndex.newInstance(constraint.getKeyFields());
505 
506     for (INodeItem item : targets) {
507       assert item != null;
508       if (item.hasValue()) {
509         INodeItem oldItem = null;
510         try {
511           oldItem = index.put(item, IIndex.toKey(item, index.getKeyFields(), dynamicContext));
512         } catch (IllegalArgumentException ex) {
513           // raised by IIndex.toKey
514           handleError(constraint, item, ex, dynamicContext);
515         }
516 
517         try {
518           if (oldItem == null) {
519             handlePass(constraint, node, item, dynamicContext);
520           } else {
521             handler.handleUniqueKeyViolation(constraint, node, oldItem, item, dynamicContext);
522           }
523         } catch (MetapathException ex) {
524           handleError(constraint, item, ex, dynamicContext);
525         }
526       }
527     }

589   private void validateMatchesItem(
590       @NonNull IMatchesConstraint constraint,
591       @NonNull INodeItem node,
592       @NonNull INodeItem item,
593       @NonNull DynamicContext dynamicContext) throws ConstraintValidationException {
594     String value = item.toAtomicItem().asString();
595 
596     IConstraintValidationHandler handler = getConstraintValidationHandler();
597     boolean valid = true;
598     Pattern pattern = constraint.getPattern();
599     if (pattern != null && !pattern.asMatchPredicate().test(value)) {
600       // failed pattern match
601       handler.handleMatchPatternViolation(constraint, node, item, value, pattern, dynamicContext);
602       valid = false;
603     }
604 
605     IDataTypeAdapter<?> adapter = constraint.getDataType();
606     if (adapter != null) {
607       try {
608         adapter.parse(value);
609       } catch (IllegalArgumentException ex) {
610         handler.handleMatchDatatypeViolation(constraint, node, item, value, adapter, ex, dynamicContext);
611         valid = false;
612       }
613     }
614 
615     if (valid) {
616       handlePass(constraint, node, item, dynamicContext);
617     }
618   }

705       }

710    * Evaluates the provided {@code constraint} against each of the
711    * {@code targets}.
712    *
713    * @param constraint
714    *          the constraint to execute
715    * @param node
716    *          the original focus of Metapath evaluation for identifying the
717    *          targets
718    * @param targets
719    *          the focus of Metapath evaluation for evaluating any constraint
720    *          Metapath clauses
721    * @param dynamicContext
722    *          the Metapath dynamic execution context to use for Metapath
723    *          evaluation
724    * @throws ConstraintValidationException
725    *           if an unexpected error occurred while validating a constraint
726    */
727   private void validateExpect(
728       @NonNull IExpectConstraint constraint,
729       @NonNull INodeItem node,
730       @NonNull ISequence<? extends INodeItem> targets,
731       @NonNull DynamicContext dynamicContext) throws ConstraintValidationException {
732     IMetapathExpression test = constraint.getTest();
733     IConstraintValidationHandler handler = getConstraintValidationHandler();
734     for (INodeItem item : targets) {
735       assert item != null;
736 
737       if (item.hasValue()) {
738         try {
739           ISequence<?> result = test.evaluate(item, dynamicContext);
740           if (FnBoolean.fnBoolean(result).toBoolean()) {
741             handlePass(constraint, node, item, dynamicContext);
742           } else {
743             handler.handleExpectViolation(constraint, node, item, dynamicContext);
744           }
745         } catch (MetapathException ex) {
746           handleError(constraint, item, ex, dynamicContext);
747         }
748       }
749     }

867     // key references
868     for (Map.Entry<String, List<KeyRef>> entry : indexNameToKeyRefMap.entrySet()) {
869       String indexName = ObjectUtils.notNull(entry.getKey());
870       IIndex index = indexNameToIndexMap.get(indexName);

883           } catch (MetapathException ex) {

885           }
886         }

891   private void validateKeyRef(
892       @NonNull IIndexHasKeyConstraint constraint,
893       @NonNull INodeItem contextNode,
894       @NonNull INodeItem item,
895       @NonNull String indexName,
896       @Nullable IIndex index,
897       @NonNull DynamicContext dynamicContext) throws ConstraintValidationException {
898     IConstraintValidationHandler handler = getConstraintValidationHandler();
899     try {
900       List<String> key;
901       try {
902         key = IIndex.toKey(item, constraint.getKeyFields(), dynamicContext);
903       } catch (IllegalArgumentException ex) {
904         handler.handleError(constraint, item, toErrorMessage(constraint, item, ex), ex, dynamicContext);
905         throw ex;
906       }
907 
908       if (index == null) {
909         handler.handleMissingIndexViolation(
910             constraint,
911             contextNode,
912             item,
913             ObjectUtils.notNull(String.format("Key reference to undefined index with name '%s'",
914                 indexName)),
915             dynamicContext);
916       } else {
917         INodeItem referencedItem = index.get(key);
918 
919         if (referencedItem == null) {
920           handler.handleIndexMiss(constraint, contextNode, item, key, dynamicContext);
921         } else {
922           handlePass(constraint, contextNode, item, dynamicContext);
923         }
924       }
925     } catch (MetapathException ex) {
926       handler.handleKeyMatchError(constraint, contextNode, item, ex, dynamicContext);
927     }

930   private class ValueStatus {
931     @NonNull
932     private final List<Pair<IAllowedValuesConstraint, IDefinitionNodeItem<?, ?>>> constraints = new LinkedList<>();
933     @NonNull
934     private final String value;
935     @NonNull
936     private final INodeItem item;
937     private boolean allowOthers = true;
938     @NonNull
939     private IAllowedValuesConstraint.Extensible extensible = IAllowedValuesConstraint.Extensible.EXTERNAL;
940 
941     public ValueStatus(@NonNull INodeItem item) {
942       this.item = item;
943       this.value = item.toAtomicItem().asString();
944     }
945 
946     public void registerAllowedValues(
947         @NonNull IAllowedValuesConstraint allowedValues,
948         @NonNull IDefinitionNodeItem<?, ?> node) throws ConstraintValidationException {
949       IAllowedValuesConstraint.Extensible newExtensible = allowedValues.getExtensible();
950       if (newExtensible.ordinal() > extensible.ordinal()) {
951         // record the most restrictive value
952         extensible = allowedValues.getExtensible();
953       } else if (newExtensible == IAllowedValuesConstraint.Extensible.NONE
954           && extensible == IAllowedValuesConstraint.Extensible.NONE) {
955         // this is an error, where there are two none constraints that conflict
956         // TODO: find a different exception type to use
957         throw new ConstraintValidationException(
958             ObjectUtils.notNull(String.format(
959                 "Multiple constraints matching path '%s' have scope='none', which prevents extension. Involved" +
960                     " constraints are: %s",
961                 Stream.concat(
962                     Stream.of(allowedValues),
963                     constraints.stream()
964                         .map(Pair::getLeft)
965                         .filter(
966                             constraint -> constraint.getExtensible() == IAllowedValuesConstraint.Extensible.NONE))
967                     .map(IConstraint::getConstraintIdentity)
968                     .collect(Collectors.joining(", ", "{", "}")),
969                 item.getMetapath())));
970       } else if (allowedValues.getExtensible().ordinal() < extensible.ordinal()) {
971         String msg = ObjectUtils.notNull(String.format(
972             "An allowed values constraint with an extensibility scope '%s'"
973                 + " exceeds the allowed scope '%s' at path '%s'",
974             allowedValues.getExtensible().name(), extensible.name(), item.getMetapath()));
975         LOGGER.atError().log(msg);
976         throw new ConstraintValidationException(msg);
977       }
978       this.constraints.add(Pair.of(allowedValues, node));
979       if (!allowedValues.isAllowedOther()) {
980         // record the most restrictive value
981         allowOthers = false;
982       }
983     }
984 
985     public void validate(@NonNull DynamicContext dynamicContext) {
986       if (!constraints.isEmpty()) {
987         boolean match = false;
988         List<IAllowedValuesConstraint> failedConstraints = new LinkedList<>();
989         IConstraintValidationHandler handler = getConstraintValidationHandler();
990         for (Pair<IAllowedValuesConstraint, IDefinitionNodeItem<?, ?>> pair : constraints) {
991           IAllowedValuesConstraint allowedValues = pair.getLeft();
992           IDefinitionNodeItem<?, ?> node = ObjectUtils.notNull(pair.getRight());
993           IAllowedValue matchingValue = allowedValues.getAllowedValue(value);
994           if (matchingValue != null) {
995             match = true;
996             handlePass(allowedValues, node, item, dynamicContext);
997           } else if (allowedValues.getExtensible() == IAllowedValuesConstraint.Extensible.NONE) {
998             // hard failure, since no other values can satisfy this constraint
999             failedConstraints = CollectionUtil.singletonList(allowedValues);
1000             match = false;
1001             break;
1002           } else {
1003             failedConstraints.add(allowedValues);
1004           } // this constraint passes, but we need to make sure other constraints do as well
1005         }
1006 
1007         // it's not a failure if allow others is true
1008         if (!match && !allowOthers) {
1009           handler.handleAllowedValuesViolation(failedConstraints, item, dynamicContext);
1010         }
1011       }
1012     }
1013   }

1035               name,
1036               // ensure the sequence is list backed
1037               result.reusable());
1038         }
1039         retval = subContext;
1040       }
1041       return retval;
1042     }
