6   package gov.nist.secauto.metaschema.core.model.constraint.impl;
7   
8   import gov.nist.secauto.metaschema.core.datatype.markup.MarkupLine;
9   import gov.nist.secauto.metaschema.core.datatype.markup.MarkupMultiline;
10  import gov.nist.secauto.metaschema.core.metapath.IMetapathExpression;
11  import gov.nist.secauto.metaschema.core.metapath.item.atomic.IBooleanItem;
12  import gov.nist.secauto.metaschema.core.model.IAttributable;
13  import gov.nist.secauto.metaschema.core.model.ISource;
14  import gov.nist.secauto.metaschema.core.model.constraint.IExpectConstraint;
15  
16  import java.util.Map;
17  import java.util.Set;
18  
19  import edu.umd.cs.findbugs.annotations.NonNull;
20  import edu.umd.cs.findbugs.annotations.Nullable;
21  
22  /**
23   * Represents an expect constraint.
24   * <p>
25   * Requires that the associated test evaluates to {@link IBooleanItem#TRUE}
26   * against the target.
27   */
28  public final class DefaultExpectConstraint
29      extends AbstractConfigurableMessageConstraint
30      implements IExpectConstraint {
31    @NonNull
32    private final IMetapathExpression test;
33  
34    /**
35     * Construct a new expect constraint.
36     *
37     * @param id
38     *          the optional identifier for the constraint
39     * @param formalName
40     *          the constraint's formal name or {@code null} if not provided
41     * @param description
42     *          the constraint's semantic description or {@code null} if not
43     *          provided
44     * @param source
45     *          information about the constraint source
46     * @param level
47     *          the significance of a violation of this constraint
48     * @param target
49     *          the Metapath expression identifying the nodes the constraint targets
50     * @param properties
51     *          a collection of associated properties
52     * @param test
53     *          a Metapath expression that is evaluated against the target node to
54     *          determine if the constraint passes
55     * @param message
56     *          an optional message to emit when the constraint is violated
57     * @param remarks
58     *          optional remarks describing the intent of the constraint
59     */
60    @SuppressWarnings("PMD.ExcessiveParameterList")
61    public DefaultExpectConstraint(
62        @Nullable String id,
63        @Nullable String formalName,
64        @Nullable MarkupLine description,
65        @NonNull ISource source,
66        @NonNull Level level,
67        @NonNull IMetapathExpression target,
68        @NonNull Map<IAttributable.Key, Set<String>> properties,
69        @NonNull IMetapathExpression test,
70        @Nullable String message,
71        @Nullable MarkupMultiline remarks) {
72      super(id, formalName, description, source, level, target, properties, message, remarks);
73      this.test = test;
74    }
75  
76    @Override
77    public IMetapathExpression getTest() {
78      return test;
79    }
80  }
