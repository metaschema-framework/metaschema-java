6   package gov.nist.secauto.metaschema.core.mdm;
7   
8   import gov.nist.secauto.metaschema.core.mdm.impl.DocumentNodeItem;
9   import gov.nist.secauto.metaschema.core.metapath.item.node.IDocumentNodeItem;
10  import gov.nist.secauto.metaschema.core.model.IAssemblyDefinition;
11  
12  import java.net.URI;
13  
14  import edu.umd.cs.findbugs.annotations.NonNull;
15  
16  /**
17   * Represents a Metapath document node item that is backed by a simple
18   * Metaschema module-based data model.
19   * <p>
20   * The {@link #newInstance(URI, IAssemblyDefinition)} method can be used to
21   * create a new document-based data model.
22   */
23  public interface IDMDocumentNodeItem
24      extends IDocumentNodeItem {
25    @Override
26    IDMRootAssemblyNodeItem getRootAssemblyNodeItem();
27  
28    /**
29     * Create a new Metaschema document-based data model.
30     *
31     * @param resource
32     *          the base URI of the document resource
33     * @param rootAssembly
34     *          the assembly that is at the root of the node tree for this document
35     * @return the document node item
36     */
37    @NonNull
38    static IDMDocumentNodeItem newInstance(
39        @NonNull URI resource,
40        @NonNull IAssemblyDefinition rootAssembly) {
41      return new DocumentNodeItem(resource, rootAssembly);
42    }
43  }
