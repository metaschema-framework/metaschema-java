6   package gov.nist.secauto.metaschema.core.mdm.impl;
7   
8   import gov.nist.secauto.metaschema.core.mdm.IDMAssemblyNodeItem;
9   import gov.nist.secauto.metaschema.core.mdm.IDMFieldNodeItem;
10  import gov.nist.secauto.metaschema.core.metapath.item.atomic.IAnyAtomicItem;
11  import gov.nist.secauto.metaschema.core.model.IAssemblyDefinition;
12  import gov.nist.secauto.metaschema.core.model.IAssemblyInstance;
13  import gov.nist.secauto.metaschema.core.model.IFieldInstance;
14  import gov.nist.secauto.metaschema.core.qname.IEnhancedQName;
15  import gov.nist.secauto.metaschema.core.util.CollectionUtil;
16  import gov.nist.secauto.metaschema.core.util.ObjectUtils;
17  
18  import java.util.Collection;
19  import java.util.Collections;
20  import java.util.LinkedList;
21  import java.util.List;
22  import java.util.Map;
23  import java.util.concurrent.ConcurrentHashMap;
24  
25  import edu.umd.cs.findbugs.annotations.NonNull;
26  
27  /**
28   * This abstract Metapath assmebly node item implementation supports creating a
29   * Metaschema module-based data model.
30   */
31  public abstract class AbstractDMAssemblyNodeItem
32      extends AbstractDMModelNodeItem<IAssemblyDefinition, IAssemblyInstance>
33      implements IDMAssemblyNodeItem {
34    @NonNull
35    private final Map<IEnhancedQName, List<IDMModelNodeItem<?, ?>>> modelItems
36        = new ConcurrentHashMap<>();
37  
38    /**
39     * Construct a new node item.
40     */
41    protected AbstractDMAssemblyNodeItem() {
42      // only allow extending classes to create instances
43    }
44  
45    @Override
46    public String stringValue() {
47      return "";
48    }
49  
50    @Override
51    protected String getValueSignature() {
52      return "";
53    }
54  
55    @Override
56    public Collection<List<IDMModelNodeItem<?, ?>>> getModelItems() {
57      return ObjectUtils.notNull(modelItems.values());
58    }
59  
60    @Override
61    public List<? extends IDMModelNodeItem<?, ?>> getModelItemsByName(IEnhancedQName name) {
62      List<? extends IDMModelNodeItem<?, ?>> retval = modelItems.get(name);
63      return retval == null ? CollectionUtil.emptyList() : retval;
64    }
65  
66    @Override
67    public IDMFieldNodeItem newField(IFieldInstance instance, IAnyAtomicItem value) {
68      List<IDMModelNodeItem<?, ?>> result = modelItems.computeIfAbsent(
69          instance.getQName(),
70          name -> Collections.synchronizedList(new LinkedList<>()));
71      IDMFieldNodeItem field = new ChildFieldNodeItem(instance, this, value);
72      result.add(field);
73      return field;
74    }
75  
76    @Override
77    public IDMAssemblyNodeItem newAssembly(IAssemblyInstance instance) {
78      List<IDMModelNodeItem<?, ?>> result = modelItems.computeIfAbsent(
79          instance.getQName(),
80          name -> Collections.synchronizedList(new LinkedList<>()));
81      IDMAssemblyNodeItem assembly = new ChildAssemblyNodeItem(instance, this);
82      result.add(assembly);
83      return assembly;
84    }
85  }
