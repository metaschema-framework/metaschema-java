26  /**
27   * An atomic Metapath item representing a date value in the Metapath system with
28   * or without an explicit time zone.
29   * <p>
30   * This interface provides functionality for handling date/time values with or
31   * without time zone information, supporting parsing, casting, and comparison
32   * operations. It works in conjunction with {@link ZonedDateTime} to handle time
33   * zone ambiguity.
34   */
35  public interface IDateItem extends ICalendarTemporalItem {
36    /**
37     * Get the type information for this item.
38     *
39     * @return the type information
40     */
41    @NonNull
42    static IAtomicOrUnionType<IDateItem> type() {
43      return MetaschemaDataTypeProvider.DATE.getItemType();
44    }
45  
46    @Override
47    default IAtomicOrUnionType<? extends IDateItem> getType() {
48      return type();
49    }
50  
51    /**
52     * Construct a new date item using the provided string {@code value}.
53     *
54     * @param value
55     *          a string representing a date
56     * @return the new item
57     */
58    @NonNull
59    static IDateItem valueOf(@NonNull String value) {
60      try {
61        return valueOf(MetaschemaDataTypeProvider.DATE.parse(value));
62      } catch (IllegalArgumentException ex) {
63        throw new InvalidTypeMetapathException(
64            null,
65            String.format("Invalid date value '%s'. %s",
66                value,
67                ex.getLocalizedMessage()),
68            ex);
69      }
70    }
71  
72    /**
73     * Construct a new date item using the provided {@code value}.
74     * <p>
75     * This method handles recording if an explicit timezone was provided using the
76     * {@code hasTimeZone} parameter. The {@link AmbiguousDate#hasTimeZone()} method
77     * can be called to determine if timezone information is present.
78     *
79     * @param value
80     *          a date, without time zone information
81     * @param hasTimeZone
82     *          {@code true} if the date/time is intended to have an associated time
83     *          zone or {@code false} otherwise
84     * @return the new item
85     * @see AmbiguousDateTime for more details on timezone handling
86     */
87    @NonNull
88    static IDateItem valueOf(@NonNull ZonedDateTime value, boolean hasTimeZone) {
89      ZonedDateTime truncated = ObjectUtils.notNull(value.truncatedTo(ChronoUnit.DAYS));
90      return hasTimeZone
91          ? IDateWithTimeZoneItem.valueOf(truncated)
92          : valueOf(new AmbiguousDate(truncated, false));
93    }
94  
95    /**
96     * Construct a new date item using the provided {@code value}.
97     * <p>
98     * This method handles recording that the timezone is implicit.
99     *
100    * @param value
101    *          a date, without time zone information
102    * @return the new item
103    * @see AmbiguousDateTime for more details on timezone handling
104    */
105   @NonNull
106   static IDateItem valueOf(@NonNull LocalDate value) {
107     return valueOf(ObjectUtils.notNull(value.atStartOfDay(ZoneOffset.UTC)), false);
108   }
109 
110   /**
111    * Construct a new date item using the provided {@code value}.
112    *
113    * @param value
114    *          an ambiguous date with time zone information already identified
115    * @return the new item
116    */
117   @NonNull
118   static IDateItem valueOf(@NonNull AmbiguousDate value) {
119     return value.hasTimeZone()
120         ? IDateWithTimeZoneItem.valueOf(value.getValue())
121         : new DateWithoutTimeZoneItemImpl(value);
122   }
123

125   default boolean hasDate() {
126     return true;
127   }
128 
129   @Override
130   default boolean hasTime() {
131     return false;
132   }
133 
134   @Override
135   default int getYear() {
136     return asZonedDateTime().getYear();
137   }
138 
139   @Override
140   default int getMonth() {
141     return asZonedDateTime().getMonthValue();
142   }
143 
144   @Override
145   default int getDay() {
146     return asZonedDateTime().getDayOfMonth();
147   }
148 
149   @Override
150   default int getHour() {
151     return 0;
152   }
153 
154   @Override
155   default int getMinute() {
156     return 0;
157   }
158 
159   @Override
160   default int getSecond() {
161     return 0;
162   }
163 
164   @Override
165   default int getNano() {
166     return 0;
167   }
168 
169   /**
170    * Get this date as a date/time value at the start of the day.
171    *
172    * @return the date time value
173    */
174   @NonNull
175   default IDateTimeItem asDateTime() {
176     return IDateTimeItem.valueOf(this);
177   }
178 
179   @Override
180   @NonNull
181   ZonedDateTime asZonedDateTime();
182 
183   /**
184    * Get the date as a {@link LocalDate}.
185    *
186    * @return the date
187    */
188   @NonNull
189   default LocalDate asLocalDate() {
190     return ObjectUtils.notNull(asZonedDateTime().toLocalDate());
191   }
192 
193   /**
194    * Adjusts an xs:dateTime value to a specific timezone, or to no timezone at
195    * all.
196    * <p>
197    * This method does one of the following things based on the arguments.
198    * <ol>
199    * <li>If the provided offset is {@code null} and the provided date/time value
200    * has a timezone, the timezone is maked absent.
201    * <li>If the provided offset is {@code null} and the provided date/time value
202    * has an absent timezone, the date/time value is returned.
203    * <li>If the provided offset is not {@code null} and the provided date/time
204    * value has an absent timezone, the date/time value is returned with the new
205    * timezone applied.
206    * <li>Otherwise, the provided timezone is applied to the date/time value
207    * adjusting the time instant.
208    * </ol>
209    * <p>
210    * Implements the XPath 3.1 <a href=
211    * "https://www.w3.org/TR/xpath-functions-31/#func-adjust-dateTime-to-timezone">fn:adjust-dateTime-to-timezone</a>
212    * function.
213    *
214    * @param offset
215    *          the timezone offset to use or {@code null}
216    * @return the adjusted date/time value
217    * @throws DateTimeFunctionException
218    *           with code
219    *           {@link DateTimeFunctionException#INVALID_TIME_ZONE_VALUE_ERROR} if
220    *           the offset is &lt; -PT14H or &gt; PT14H
221    */
222   @Override
223   default IDateItem replaceTimezone(@Nullable IDayTimeDurationItem offset) {
224     return offset == null
225         ? hasTimezone()
226             ? valueOf(ObjectUtils.notNull(asZonedDateTime().withZoneSameLocal(ZoneOffset.UTC)), false)
227             : this
228         : hasTimezone()
229             ? valueOf(IDateTimeItem.valueOf(this).replaceTimezone(offset).asZonedDateTime(),
230                 true)
231             : valueOf(
232                 ObjectUtils.notNull(asZonedDateTime().withZoneSameLocal(offset.asZoneOffset())),
233                 true);
234   }
235 
236   /**
237    * Cast the provided type to this item type.
238    *
239    * @param item
240    *          the item to cast
241    * @return the original item if it is already this type, otherwise a new item
242    *         cast to this type
243    * @throws InvalidValueForCastFunctionException
244    *           if the provided {@code item} cannot be cast to this type
245    */
246   @NonNull
247   static IDateItem cast(@NonNull IAnyAtomicItem item) {
248     IDateItem retval;
249     if (item instanceof IDateItem) {
250       retval = (IDateItem) item;
251     } else if (item instanceof IDateTimeItem) {
252       retval = ((IDateTimeItem) item).asDate();
253     } else if (item instanceof IStringItem || item instanceof IUntypedAtomicItem) {
254       try {
255         retval = valueOf(item.asString());
256       } catch (IllegalStateException | InvalidTypeMetapathException ex) {
257         // asString can throw IllegalStateException exception
258         throw new InvalidValueForCastFunctionException(ex);
259       }
260     } else {
261       throw new InvalidValueForCastFunctionException(
262           String.format("Unsupported item type '%s'", item.getClass().getName()));
263     }
264     return retval;
265   }
266 
267   @Override
268   default IDateItem castAsType(IAnyAtomicItem item) {
269     return cast(item);
270   }
