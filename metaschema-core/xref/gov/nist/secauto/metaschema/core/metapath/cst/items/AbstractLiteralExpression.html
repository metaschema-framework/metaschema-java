6   package gov.nist.secauto.metaschema.core.metapath.cst.items;
7   
8   import gov.nist.secauto.metaschema.core.metapath.DynamicContext;
9   import gov.nist.secauto.metaschema.core.metapath.cst.AbstractExpression;
10  import gov.nist.secauto.metaschema.core.metapath.item.ISequence;
11  import gov.nist.secauto.metaschema.core.metapath.item.atomic.IAnyAtomicItem;
12  
13  import edu.umd.cs.findbugs.annotations.NonNull;
14  
15  /**
16   * A base class for all Metapath
17   * <a href="https://www.w3.org/TR/xpath-31/#id-literals">literal value
18   * expressions</a>.
19   *
20   * @param <RESULT_TYPE>
21   *          the Java type of the literal result
22   */
23  public abstract class AbstractLiteralExpression<RESULT_TYPE extends IAnyAtomicItem>
24      extends AbstractExpression
25      implements ILiteralExpression<RESULT_TYPE> {
26    @NonNull
27    private final RESULT_TYPE value;
28  
29    /**
30     * Construct an expression that always returns the same literal value.
31     *
32     * @param text
33     *          the parsed text of the expression
34     * @param value
35     *          the literal value
36     */
37    public AbstractLiteralExpression(@NonNull String text, @NonNull RESULT_TYPE value) {
38      super(text);
39      this.value = value;
40    }
41  
42    @Override
43    public RESULT_TYPE getValue() {
44      return value;
45    }
46  
47    @Override
48    protected ISequence<RESULT_TYPE> evaluate(DynamicContext dynamicContext, ISequence<?> focus) {
49      return ISequence.of(getValue());
50    }
51  
52    @SuppressWarnings("null")
53    @Override
54    public String toCSTString() {
55      return String.format("%s[value=%s]", getClass().getName(), getValue().asString());
56    }
57  }
