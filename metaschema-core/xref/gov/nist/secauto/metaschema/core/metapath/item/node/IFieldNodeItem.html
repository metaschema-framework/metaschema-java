2   package gov.nist.secauto.metaschema.core.metapath.item.node;
3   
4   import gov.nist.secauto.metaschema.core.metapath.DynamicContext;
5   import gov.nist.secauto.metaschema.core.metapath.StaticContext;
6   import gov.nist.secauto.metaschema.core.metapath.format.IPathFormatter;
7   import gov.nist.secauto.metaschema.core.metapath.item.ICollectionValue;
8   import gov.nist.secauto.metaschema.core.metapath.type.IAtomicOrUnionType;
9   import gov.nist.secauto.metaschema.core.metapath.type.IItemType;
10  import gov.nist.secauto.metaschema.core.metapath.type.IKindTest;
11  import gov.nist.secauto.metaschema.core.model.IFieldDefinition;
12  import gov.nist.secauto.metaschema.core.model.IFieldInstance;
13  
14  import java.net.URI;
15  
16  import edu.umd.cs.findbugs.annotations.NonNull;
17  import edu.umd.cs.findbugs.annotations.Nullable;
18  
19  /**
20   * A Metapath node valued item representing a Metaschema module field.
21   */
22  public interface IFieldNodeItem
23      extends IModelNodeItem<IFieldDefinition, IFieldInstance>,
24      IAtomicValuedNodeItem {
25    /**
26     * Get the static type information of the node item.
27     *
28     * @return the item type
29     */
30    @NonNull
31    static IItemType type() {
32      return IItemType.field();
33    }
34  
35    @Override
36    default NodeItemKind getNodeItemKind() {
37      return NodeItemKind.FIELD;
38    }
39  
40    @Override
41    default NodeType getNodeType() {
42      return NodeType.FIELD;
43    }
44  
45    @Override
46    default IFieldNodeItem getNodeItem() {
47      return this;
48    }
49  
50    @Override
51    default IKindTest<IFieldNodeItem> getType() {
52      StaticContext staticContext = getStaticContext();
53      return IItemType.field(
54          getQName(),
55          getDefinition().getDefinitionQName().toEQName(staticContext),
56          staticContext);
57    }
58  
59    @Override
60    default IAtomicOrUnionType<?> getValueItemType() {
61      return getDefinition().getJavaTypeAdapter().getItemType();
62    }
63  
64    @Override
65    @Nullable
66    default URI getBaseUri() {
67      INodeItem parent = getParentNodeItem();
68      return parent == null ? null : parent.getBaseUri();
69    }
70  
71    @Override
72    @NonNull
73    default String format(@NonNull IPathFormatter formatter) {
74      return formatter.formatField(this);
75    }
76  
77    @Override
78    default <CONTEXT, RESULT> RESULT accept(@NonNull INodeItemVisitor<CONTEXT, RESULT> visitor, CONTEXT context) {
79      return visitor.visitField(this, context);
80    }
81  
82    @Override
83    default boolean deepEquals(ICollectionValue other, DynamicContext dynamicContext) {
84      return other instanceof IFieldNodeItem
85          && NodeComparators.compareNodeItem(this, (IFieldNodeItem) other, dynamicContext);
86    }
87  }
