12  import gov.nist.secauto.metaschema.core.model.IAssemblyDefinition;
13  import gov.nist.secauto.metaschema.core.model.IAssemblyInstance;
14  
15  import java.net.URI;
16  
17  import edu.umd.cs.findbugs.annotations.NonNull;
18  import edu.umd.cs.findbugs.annotations.Nullable;

20  /**
21   * A Metapath node valued item representing a Metaschema module assembly.
22   */
23  public interface IAssemblyNodeItem extends IModelNodeItem<IAssemblyDefinition, IAssemblyInstance> {
24    /**
25     * Get the static type information of the node item.
26     *
27     * @return the item type
28     */
29    @NonNull
30    static IItemType type() {
31      return IItemType.assembly();
32    }
33  
34    @Override
35    default IKindTest<IAssemblyNodeItem> getType() {
36      StaticContext staticContext = getStaticContext();
37      return IItemType.assembly(
38          getQName(),
39          getDefinition().getDefinitionQName().toEQName(staticContext),
40          staticContext);
41    }
42  
43    @Override
44    default NodeItemKind getNodeItemKind() {
45      return NodeItemKind.ASSEMBLY;
46    }
47  
48    @Override
49    default NodeType getNodeType() {
50      return NodeType.ASSEMBLY;
51    }
52  
53    @Override
54    default IAssemblyNodeItem getNodeItem() {
55      return this;
56    }
57  
58    @Override
59    @Nullable
60    default URI getBaseUri() {
61      INodeItem parent = getParentNodeItem();
62      return parent == null ? null : parent.getBaseUri();
63    }
64  
65    @Override
66    default String format(@NonNull IPathFormatter formatter) {
67      return formatter.formatAssembly(this);
68    }
69  
70    @Override
71    default IAnyAtomicItem toAtomicItem() {
72      throw new InvalidTypeFunctionException(InvalidTypeFunctionException.DATA_ITEM_IS_FUNCTION, this);
73    }
74  
75    @Override
76    default <CONTEXT, RESULT> RESULT accept(@NonNull INodeItemVisitor<CONTEXT, RESULT> visitor, CONTEXT context) {
77      return visitor.visitAssembly(this, context);
78    }
79  
80    @Override
81    default boolean deepEquals(ICollectionValue other, DynamicContext dynamicContext) {
82      return other instanceof IAssemblyNodeItem
83          && NodeComparators.compareNodeItem(this, (IAssemblyNodeItem) other, dynamicContext);
84    }
85  }
