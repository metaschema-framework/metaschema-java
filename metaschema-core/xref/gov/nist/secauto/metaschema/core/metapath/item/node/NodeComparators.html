2   package gov.nist.secauto.metaschema.core.metapath.item.node;
3   
4   import gov.nist.secauto.metaschema.core.metapath.DynamicContext;
5   import gov.nist.secauto.metaschema.core.metapath.item.atomic.IAnyAtomicItem;
6   import gov.nist.secauto.metaschema.core.qname.IEnhancedQName;
7   import gov.nist.secauto.metaschema.core.util.ObjectUtils;
8   
9   import java.util.ArrayList;
10  import java.util.Collection;
11  import java.util.Collections;
12  import java.util.Comparator;
13  import java.util.Iterator;
14  import java.util.List;
15  
16  import edu.umd.cs.findbugs.annotations.NonNull;
17  import edu.umd.cs.findbugs.annotations.Nullable;
18  
19  /**
20   * Provides methods for comparing nodes according to the
21   * <a href="https://www.w3.org/TR/xpath-functions-31/#func-deep-equal">Metapath
22   * specification</a>.
23   */
24  public final class NodeComparators {
25    private static final Comparator<IFlagNodeItem> FLAG_SORT
26        = Comparator.comparing(IFlagNodeItem::getQName, IEnhancedQName::compareTo);
27  
28    /**
29     * Compare two node items for equality.
30     *
31     * @param item1
32     *          the first item to compare
33     * @param item2
34     *          the second item to compare
35     * @param dynamicContext
36     *          used to provide evaluation information, including the implicit
37     *          timezone
38     * @return {@code true} if both node items are the same type and have the same
39     *         flag and model members, or {@code false} otherwise
40     */
41    public static boolean compareNodeItem(
42        @NonNull INodeItem item1,
43        @NonNull INodeItem item2,
44        @NonNull DynamicContext dynamicContext) {
45      return item1.getNodeType() == item2.getNodeType()
46          && compareFlags(item1.getFlags(), item2.getFlags(), dynamicContext)
47          && compareModelItems(item1.getModelItems(), item2.getModelItems(), dynamicContext);
48    }
49  
50    private static boolean compareAtomics(
51        @Nullable IAnyAtomicItem atomic1,
52        @Nullable IAnyAtomicItem atomic2,
53        @NonNull DynamicContext dynamicContext) {
54      return (atomic1 == null && atomic2 == null) || (atomic1 != null && atomic1.deepEquals(atomic2, dynamicContext));
55    }
56  
57    /**
58     * Compare two node items for equality.
59     *
60     * @param item1
61     *          the first item to compare
62     * @param item2
63     *          the second item to compare
64     * @param dynamicContext
65     *          used to provide evaluation information, including the implicit
66     *          timezone
67     * @return a negative integer, zero, or a positive integer if the first argument
68     *         is less than, equal to, or greater than the second.
69     */
70    public static boolean compareModelNodeItem(
71        @NonNull IModelNodeItem<?, ?> item1,
72        @NonNull IModelNodeItem<?, ?> item2,
73        @NonNull DynamicContext dynamicContext) {
74      boolean retval;
75      if (item1 instanceof IAssemblyNodeItem) {
76        retval = compareNodeItem(item1, item2, dynamicContext);
77      } else if (item1 instanceof IFieldNodeItem) {
78        retval = compareAsField(item1, item2, dynamicContext);
79      } else {
80        throw new UnsupportedOperationException("Unsupported model node item type: " + item1.getClass().getName());
81      }
82      return retval;
83    }
84  
85    private static boolean compareFlags(
86        @NonNull Collection<? extends IFlagNodeItem> flags1,
87        @NonNull Collection<? extends IFlagNodeItem> flags2,
88        @NonNull DynamicContext dynamicContext) {
89      Comparator<Collection<? extends IFlagNodeItem>> bySize = Comparator.comparingInt(Collection::size);
90      int delta = bySize.compare(flags1, flags2);
91      if (delta != 0) {
92        return false;
93      }
94  
95      // sort the collections to compare in an order independent way
96      List<IFlagNodeItem> list1 = new ArrayList<>(flags1);
97      List<IFlagNodeItem> list2 = new ArrayList<>(flags2);
98      Collections.sort(list1, FLAG_SORT);
99      Collections.sort(list2, FLAG_SORT);
100 
101     // compare the results
102     for (int i = 0; i < list1.size(); i++) {
103       if (!compareAsFlag(
104           ObjectUtils.requireNonNull(list1.get(i)),
105           ObjectUtils.requireNonNull(list2.get(i)),
106           dynamicContext)) {
107         return false;
108       }
109     }
110     return true;
111   }
112 
113   private static boolean compareModelItems(
114       @NonNull Collection<? extends List<? extends IModelNodeItem<?, ?>>> items1,
115       @NonNull Collection<? extends List<? extends IModelNodeItem<?, ?>>> items2,
116       @NonNull DynamicContext dynamicContext) {
117     Comparator<Collection<? extends List<? extends IModelNodeItem<?, ?>>>> bySize
118         = Comparator.comparingInt(Collection::size);
119     int delta = bySize.compare(items1, items2);
120     if (delta != 0) {
121       return false;
122     }
123 
124     Iterator<? extends List<? extends IModelNodeItem<?, ?>>> thisIterator = items1.iterator();
125     Iterator<? extends List<? extends IModelNodeItem<?, ?>>> otherIterator = items2.iterator();
126     while (thisIterator.hasNext() && otherIterator.hasNext()) {
127       List<? extends IModelNodeItem<?, ?>> l1 = thisIterator.next();
128       List<? extends IModelNodeItem<?, ?>> l2 = otherIterator.next();
129 
130       Iterator<? extends IModelNodeItem<?, ?>> il1 = l1.iterator();
131       Iterator<? extends IModelNodeItem<?, ?>> il2 = l2.iterator();
132       while (thisIterator.hasNext() && otherIterator.hasNext()) {
133         IModelNodeItem<?, ?> item1 = ObjectUtils.requireNonNull(il1.next());
134         IModelNodeItem<?, ?> item2 = ObjectUtils.requireNonNull(il2.next());
135 
136         if (!compareModelNodeItem(item1, item2, dynamicContext)) {
137           return false;
138         }
139       }
140     }
141     return true;
142   }
143 
144   /**
145    * Compare two flag node items for equality.
146    *
147    * @param item1
148    *          the first item to compare
149    * @param item2
150    *          the second item to compare
151    * @param dynamicContext
152    *          used to provide evaluation information, including the implicit
153    *          timezone
154    * @return {@code true} if both flags have the same name and value, or
155    *         {@code false} otherwise
156    */
157   public static boolean compareAsFlag(
158       @NonNull IFlagNodeItem item1,
159       @NonNull IFlagNodeItem item2,
160       @NonNull DynamicContext dynamicContext) {
161     return item1.getQName().equals(item2.getQName())
162         && compareAtomics(item1.toAtomicItem(), item2.toAtomicItem(), dynamicContext);
163   }
164 
165   private static boolean compareAsField(
166       @NonNull IModelNodeItem<?, ?> item1,
167       @NonNull IModelNodeItem<?, ?> item2,
168       @NonNull DynamicContext dynamicContext) {
169     return compareNodeItem(item1, item2, dynamicContext)
170         && compareAtomics(item1.toAtomicItem(), item2.toAtomicItem(), dynamicContext);
171   }
172 
173   private NodeComparators() {
174     // disable construction
175   }
176 }
