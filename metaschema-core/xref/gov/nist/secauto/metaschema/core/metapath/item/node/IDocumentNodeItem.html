2   package gov.nist.secauto.metaschema.core.metapath.item.node;
3   
4   import gov.nist.secauto.metaschema.core.metapath.DynamicContext;
5   import gov.nist.secauto.metaschema.core.metapath.format.IPathFormatter;
6   import gov.nist.secauto.metaschema.core.metapath.item.ICollectionValue;
7   import gov.nist.secauto.metaschema.core.metapath.type.IItemType;
8   import gov.nist.secauto.metaschema.core.metapath.type.IKindTest;
9   
10  import java.net.URI;
11  
12  import edu.umd.cs.findbugs.annotations.NonNull;
13  
14  /**
15   * A node item that represents the root of a tree of nodes associated with a
16   * document resource.
17   */
18  public interface IDocumentNodeItem extends IDocumentBasedNodeItem {
19    /**
20     * The node item's type.
21     *
22     * @return the type
23     */
24    @NonNull
25    static IItemType type() {
26      return IItemType.document();
27    }
28  
29    @Override
30    default IKindTest<IDocumentNodeItem> getType() {
31      return IItemType.document(
32          getRootAssemblyNodeItem().getType());
33    }
34  
35    @Override
36    default NodeItemKind getNodeItemKind() {
37      return NodeItemKind.DOCUMENT;
38    }
39  
40    @Override
41    default NodeType getNodeType() {
42      return NodeType.DOCUMENT;
43    }
44  
45    @Override
46    default IDocumentNodeItem getNodeItem() {
47      return this;
48    }
49  
50    @Override
51    @NonNull
52    URI getDocumentUri();
53  
54    /**
55     * Get the node item for the document root element.
56     *
57     * @return the root node item
58     */
59    @NonNull
60    IRootAssemblyNodeItem getRootAssemblyNodeItem();
61  
62    @Override
63    default String format(@NonNull IPathFormatter formatter) {
64      return formatter.formatDocument(this);
65    }
66  
67    @Override
68    default <CONTEXT, RESULT> RESULT accept(@NonNull INodeItemVisitor<CONTEXT, RESULT> visitor, CONTEXT context) {
69      return visitor.visitDocument(this, context);
70    }
71  
72    @Override
73    default boolean deepEquals(ICollectionValue other, DynamicContext dynamicContext) {
74      return other instanceof IDocumentNodeItem
75          && NodeComparators.compareNodeItem(this, (IDocumentNodeItem) other, dynamicContext);
76    }
77  }
