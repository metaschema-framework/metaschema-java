6   package gov.nist.secauto.metaschema.core.metapath.cst.math;
7   
8   import gov.nist.secauto.metaschema.core.metapath.DynamicContext;
9   import gov.nist.secauto.metaschema.core.metapath.IExpression;
10  import gov.nist.secauto.metaschema.core.metapath.function.FunctionUtils;
11  import gov.nist.secauto.metaschema.core.metapath.item.ISequence;
12  import gov.nist.secauto.metaschema.core.metapath.item.atomic.IAnyAtomicItem;
13  import gov.nist.secauto.metaschema.core.metapath.item.atomic.INumericItem;
14  
15  import java.util.Map;
16  
17  import edu.umd.cs.findbugs.annotations.NonNull;
18  import edu.umd.cs.findbugs.annotations.Nullable;
19  
20  /**
21   * An immutable binary expression that supports basic arithmetic evaluation.
22   * <p>
23   * The result type is determined through static analysis of the sub-expressions,
24   * which may result in a more specific type that is a sub-class of the base
25   * result type.
26   * <p>
27   * The arithmetic operation method
28   * {@link #operation(IAnyAtomicItem, IAnyAtomicItem, DynamicContext)} must be
29   * implemented by extending classes to provide the evaluation logic.
30   */
31  public abstract class AbstractBasicArithmeticExpression
32      extends AbstractArithmeticExpression<IAnyAtomicItem> {
33  
34    /**
35     * An expression that represents a basic arithmetic operation on two values.
36     *
37     * @param text
38     *          the parsed text of the expression
39     * @param left
40     *          the first item
41     * @param right
42     *          the second item
43     */
44    public AbstractBasicArithmeticExpression(
45        @NonNull String text,
46        @NonNull IExpression left,
47        @NonNull IExpression right) {
48      super(text, left, right, IAnyAtomicItem.class);
49    }
50  
51    @Override
52    public Class<IAnyAtomicItem> getBaseResultType() {
53      return IAnyAtomicItem.class;
54    }
55  
56    @Override
57    protected ISequence<? extends IAnyAtomicItem> evaluate(DynamicContext dynamicContext, ISequence<?> focus) {
58      IAnyAtomicItem leftItem = ISequence.of(getLeft().accept(dynamicContext, focus).atomize()).getFirstItem(true);
59      IAnyAtomicItem rightItem = ISequence.of(getRight().accept(dynamicContext, focus).atomize()).getFirstItem(true);
60  
61      return resultOrEmpty(leftItem, rightItem, dynamicContext);
62    }
63  
64    /**
65     * Setup the operation on two atomic items.
66     *
67     * @param leftItem
68     *          the first item
69     * @param rightItem
70     *          the second item
71     * @param dynamicContext
72     *          used to provide evaluation information, including the implicit
73     *          timezone
74     * @return the result of the operation or an empty {@link ISequence} if either
75     *         item is {@code null}
76     */
77    @NonNull
78    protected ISequence<? extends IAnyAtomicItem> resultOrEmpty(
79        @Nullable IAnyAtomicItem leftItem,
80        @Nullable IAnyAtomicItem rightItem,
81        @NonNull DynamicContext dynamicContext) {
82      ISequence<? extends IAnyAtomicItem> retval;
83      if (leftItem == null || rightItem == null) {
84        retval = ISequence.empty();
85      } else {
86        IAnyAtomicItem result = operation(leftItem, rightItem, dynamicContext);
87        retval = ISequence.of(result);
88      }
89      return retval;
90    }
91  
92    /**
93     * Performs the arithmetic operation using the two provided values.
94     *
95     * @param left
96     *          the first item
97     * @param right
98     *          the second item
99     * @param dynamicContext
100    *          used to provide evaluation information, including the implicit
101    *          timezone
102    * @return the result of the operation
103    */
104   @NonNull
105   protected IAnyAtomicItem operation(
106       @NonNull IAnyAtomicItem left,
107       @NonNull IAnyAtomicItem right,
108       @NonNull DynamicContext dynamicContext) {
109 
110     Map<
111         Class<? extends IAnyAtomicItem>,
112         Map<Class<? extends IAnyAtomicItem>, OperationStrategy>> strategies = getStrategies();
113     Class<? extends IAnyAtomicItem> leftClass = left.getClass();
114 
115     // Find matching strategy for minuend type
116     Map<Class<? extends IAnyAtomicItem>, OperationStrategy> typeStrategies = null;
117     for (Map.Entry<Class<? extends IAnyAtomicItem>,
118         Map<Class<? extends IAnyAtomicItem>, OperationStrategy>> entry : strategies.entrySet()) {
119       if (entry.getKey().isAssignableFrom(leftClass)) {
120         // this is a matching strategy map
121         typeStrategies = entry.getValue();
122         break;
123       }
124     }
125 
126     if (typeStrategies == null) {
127       return operationAsNumeric(
128           FunctionUtils.castToNumeric(left),
129           FunctionUtils.castToNumeric(right));
130     }
131 
132     // Find matching strategy for subtrahend type
133     Class<? extends IAnyAtomicItem> rightClass = right.getClass();
134     for (Map.Entry<Class<? extends IAnyAtomicItem>, OperationStrategy> entry : typeStrategies.entrySet()) {
135       if (entry.getKey().isAssignableFrom(rightClass)) {
136         // this is matching strategy, execute it
137         return entry.getValue().execute(left, right, dynamicContext);
138       }
139     }
140 
141     throw new UnsupportedOperationException(unsupportedMessage(
142         left.toSignature(),
143         right.toSignature()));
144   }
145 
146   /**
147    * Provides a mapping of the left class to a mapping of the right class and the
148    * strategy to use to compute the operation.
149    * <p>
150    * This mapping is used to lookup the strategy to use based on the left and
151    * right classes.
152    *
153    * @return the mapping
154    */
155   @NonNull
156   protected abstract Map<
157       Class<? extends IAnyAtomicItem>,
158       Map<Class<? extends IAnyAtomicItem>, OperationStrategy>> getStrategies();
159 
160   /**
161    * Generates an error message for unsupported operand types.
162    *
163    * @param left
164    *          the string representation of the left operand type
165    * @param right
166    *          the string representation of the right operand type
167    * @return the formatted error message
168    */
169   @NonNull
170   protected abstract String unsupportedMessage(@NonNull String left, @NonNull String right);
171 
172   /**
173    * Performs the arithmetic operation on numeric items.
174    *
175    * @param left
176    *          the first numeric item
177    * @param right
178    *          the second numeric item
179    * @return the result of the numeric operation
180    */
181   @NonNull
182   protected abstract INumericItem operationAsNumeric(@NonNull INumericItem left, @NonNull INumericItem right);
183 
184   /**
185    * Provides a callback for resolving arithmetic operations.
186    */
187   @FunctionalInterface
188   protected interface OperationStrategy {
189     /**
190      * Called to execute an arithmetic operation.
191      *
192      * @param left
193      *          the left side of the arithmetic operation
194      * @param right
195      *          the right side of the arithmetic operation
196      * @param dynamicContext
197      *          the evaluation dynamic context
198      * @return the arithmetic result
199      */
200     @NonNull
201     IAnyAtomicItem execute(
202         @NonNull IAnyAtomicItem left,
203         @NonNull IAnyAtomicItem right,
204         @NonNull DynamicContext dynamicContext);
205   }
206 }
