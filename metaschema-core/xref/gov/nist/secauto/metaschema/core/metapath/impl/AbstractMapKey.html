6   package gov.nist.secauto.metaschema.core.metapath.impl;
7   
8   import gov.nist.secauto.metaschema.core.metapath.item.function.IMapKey;
9   
10  import nl.talsmasoftware.lazy4j.Lazy;
11  
12  public abstract class AbstractMapKey implements IMapKey {
13    private final Lazy<Integer> hashCode = Lazy.lazy(this::generateHashCode);
14  
15    @Override
16    public int hashCode() {
17      return hashCode.get();
18    }
19  
20    /**
21     * Generate the hash code for the key.
22     *
23     * @return the hash code
24     */
25    protected int generateHashCode() {
26      return getKey().hashCode();
27    }
28  
29    @Override
30    public boolean equals(Object obj) {
31      return this == obj
32          || (obj instanceof IMapKey && isSameKey((IMapKey) obj));
33    }
34  
35    @Override
36    public String toString() {
37      return getKey().toSignature();
38    }
39  }
