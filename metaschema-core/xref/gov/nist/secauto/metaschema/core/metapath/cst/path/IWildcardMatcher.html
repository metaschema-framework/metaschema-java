6   package gov.nist.secauto.metaschema.core.metapath.cst.path;
7   
8   import gov.nist.secauto.metaschema.core.metapath.item.node.IDefinitionNodeItem;
9   
10  import java.util.function.Predicate;
11  
12  import edu.umd.cs.findbugs.annotations.NonNull;
13  
14  /**
15   * A common interface for all wildcard matchers.
16   * <p>
17   * Based on the XPath 3.1
18   * <a href="https://www.w3.org/TR/xpath-31/#node-tests">wildcard node test</a>
19   * syntax.
20   */
21  @FunctionalInterface
22  public interface IWildcardMatcher extends Predicate<IDefinitionNodeItem<?, ?>> {
23    /**
24     * Construct a wildcard matcher that matches the provided local name in any
25     * namespace.
26     *
27     * @param localName
28     *          the name used to match nodes
29     * @return the matcher
30     */
31    @NonNull
32    static IWildcardMatcher anyNamespace(@NonNull String localName) {
33      // the grammar should ensure that localName is not empty
34      assert !localName.isEmpty();
35      return new MatchAnyNamespace(localName);
36    }
37  
38    /**
39     * Construct a wildcard matcher that matches any local name in the provided
40     * namespace.
41     *
42     * @param namespace
43     *          the namespace used to match nodes
44     * @return the matcher
45     */
46    @NonNull
47    static IWildcardMatcher anyLocalName(@NonNull String namespace) {
48      return new MatchAnyLocalName(namespace);
49    }
50  
51    @Override
52    @NonNull
53    String toString();
54  }
