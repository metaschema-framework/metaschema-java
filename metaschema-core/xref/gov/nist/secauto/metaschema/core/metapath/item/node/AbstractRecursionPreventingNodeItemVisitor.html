2   package gov.nist.secauto.metaschema.core.metapath.item.node;
3   
4   import gov.nist.secauto.metaschema.core.model.IAssemblyDefinition;
5   
6   /**
7    * A node item visitor that prevents infinite recursion when traversing assembly
8    * nodes by detecting and avoiding cycles in the node hierarchy.
9    *
10   * @param <CONTEXT>
11   *          the Java type of the visiting context
12   * @param <RESULT>
13   *          the Java type of the visiting result
14   */
15  public abstract class AbstractRecursionPreventingNodeItemVisitor<CONTEXT, RESULT>
16      extends AbstractNodeItemVisitor<CONTEXT, RESULT> {
17  
18    @Override
19    public RESULT visitAssembly(IAssemblyNodeItem item, CONTEXT context) {
20      // only walk new records to avoid looping
21      // check if this item's definition is the same as an ancestor's
22      return isDecendant(item, item.getDefinition())
23          ? defaultResult()
24          : super.visitAssembly(item, context);
25    }
26  
27    /**
28     * Determines if the provided node is a descendant of the assembly definition.
29     *
30     * @param node
31     *          the node item to test
32     * @param assemblyDefinition
33     *          the assembly definition to determine as an ancestor of the node
34     * @return {@code true} if the assembly definition is the node's ancestor, or
35     *         {@code false} otherwise
36     */
37    protected boolean isDecendant(IAssemblyNodeItem node, IAssemblyDefinition assemblyDefinition) {
38      return node.ancestor()
39          .map(ancestor -> {
40            boolean retval = false;
41            if (ancestor instanceof IAssemblyNodeItem) {
42              IAssemblyDefinition ancestorDef = ((IAssemblyNodeItem) ancestor).getDefinition();
43              retval = ancestorDef.equals(assemblyDefinition);
44            }
45            return retval;
46          }).anyMatch(value -> value);
47    }
48  }
