19  /**
20   * The CST node for a Metapath
21   * <a href="https://www.w3.org/TR/xpath-31/#dt-expanded-qname">expanded QName
22   * name test</a>.
23   */
24  @SuppressWarnings("PMD.TestClassWithoutTestCases")
25  public class KindNodeTest
26      extends AbstractExpression
27      implements INodeTestExpression {
28  
29    @NonNull
30    private final IItemType type;
31  
32    /**
33     * Construct a new kind test expression.
34     *
35     * @param text
36     *          the parsed text of the expression
37     * @param type
38     *          the expected item type to test against
39     */
40    public KindNodeTest(@NonNull String text, @NonNull IItemType type) {
41      super(text);
42      this.type = type;
43    }
44  
45    @Override
46    public <RESULT, CONTEXT> RESULT accept(IExpressionVisitor<RESULT, CONTEXT> visitor, CONTEXT context) {
47      return visitor.visitKindNodeTest(this, context);
48    }
49  
50    @Override
51    protected ISequence<? extends INodeItem> evaluate(DynamicContext dynamicContext, ISequence<?> focus) {
52      return ISequence.of(filterStream(ObjectUtils.notNull(focus.stream()
53          .map(item -> ItemUtils.checkItemIsNodeItem(dynamicContext, item)))));
54    }
55  
56    @SuppressWarnings("null")
57    @Override
58    public String toCSTString() {
59      return String.format("%s[kind=%s]", getClass().getName(), type.toSignature());
60    }
61  
62    @Override
63    public boolean match(INodeItem item) {
64      return type.isInstance(item);
65    }
66  }
