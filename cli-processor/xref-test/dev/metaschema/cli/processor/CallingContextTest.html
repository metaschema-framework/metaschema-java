44    @BeforeEach

46      nullOutput = new PrintStream(OutputStream.nullOutputStream(), true, StandardCharsets.UTF_8);
47      processor = new CLIProcessor(
48          "test-cli",
49          ObjectUtils.notNull(Map.of(CLIProcessor.COMMAND_VERSION, new TestVersionInfo())),

58    @NonNull
59    private CallingContext createContext(@NonNull String... args) {

269   @Nested
270   @DisplayName("wrapText()")
271   class WrapTextTests {
272 
273     @Test
274     @DisplayName("returns text unchanged when shorter than max width")
275     void returnsTextUnchangedWhenShorterThanMaxWidth() {
276       String text = "Short text";
277       String result = CallingContext.wrapText(text, 80, "    ");
278       assertEquals(text, result);
279     }
280 
281     @Test
282     @DisplayName("returns text unchanged when exactly max width")
283     void returnsTextUnchangedWhenExactlyMaxWidth() {
284       String text = "Exactly twenty chars"; // 20 chars
285       String result = CallingContext.wrapText(text, 20, "    ");
286       assertEquals(text, result);
287     }
288 
289     @Test
290     @DisplayName("wraps text at word boundary")
291     void wrapsTextAtWordBoundary() {
292       String text = "This is a long text that should wrap at word boundaries";
293       String result = CallingContext.wrapText(text, 30, "    ");
294 
295       // Should wrap after "that" (position 26) and continue with indent
296       assertTrue(result.contains(System.lineSeparator()));
297       assertTrue(result.startsWith("This is a long text that"));
298       assertTrue(result.contains("    should wrap"));
299     }
300 
301     @Test
302     @DisplayName("uses correct indentation for continuation lines")
303     void usesCorrectIndentationForContinuationLines() {
304       String text = "First part of text second part of text third part of text";
305       String indent = "        ";
306       String result = CallingContext.wrapText(text, 25, indent);
307 
308       String[] lines = result.split(System.lineSeparator());
309       assertTrue(lines.length > 1, "Expected multiple lines");
310       for (int i = 1; i < lines.length; i++) {
311         assertTrue(lines[i].startsWith(indent),
312             "Line " + i + " should start with indent: [" + lines[i] + "]");
313       }
314     }
315 
316     @Test
317     @DisplayName("handles text with no spaces (force break)")
318     void handlesTextWithNoSpaces() {
319       String text = "ThisIsAVeryLongWordWithNoSpacesAtAll";
320       String result = CallingContext.wrapText(text, 15, "  ");
321 
322       // Should force break at width since there are no word boundaries
323       String[] lines = result.split(System.lineSeparator());
324       assertTrue(lines.length > 1, "Expected multiple lines for long word");
325     }
326 
327     @Test
328     @DisplayName("handles empty indent")
329     void handlesEmptyIndent() {
330       String text = "This text should wrap without indentation on continuation";
331       String result = CallingContext.wrapText(text, 20, "");
332 
333       assertTrue(result.contains(System.lineSeparator()));
334       String[] lines = result.split(System.lineSeparator());
335       assertTrue(lines.length > 1);
336     }
337 
338     @Test
339     @DisplayName("handles single word longer than width")
340     void handlesSingleWordLongerThanWidth() {
341       String text = "Supercalifragilisticexpialidocious";
342       String result = CallingContext.wrapText(text, 10, "  ");
343 
344       // Should break the word at width boundaries
345       String[] lines = result.split(System.lineSeparator());
346       assertTrue(lines.length > 1);
347     }
348 
349     @Test
350     @DisplayName("preserves text content after wrapping")
351     void preservesTextContentAfterWrapping() {
352       String text = "The quick brown fox jumps over the lazy dog";
353       String result = CallingContext.wrapText(text, 20, "    ");
354 
355       // Remove line separators and indents to verify content preserved
356       String normalized = result.replace(System.lineSeparator(), " ").replaceAll("\\s+", " ").trim();
357       assertEquals(text, normalized);
358     }
359 
360     @Test
361     @DisplayName("force-break preserves all characters without skipping")
362     void forceBreakPreservesAllCharacters() {
363       // Test case from CodeRabbit: wrapping "ABCDEFGHIJ" at width 5
364       // Should produce "ABCDE" + newline + indent + "FGHIJ" (all characters
365       // preserved)
366       String text = "ABCDEFGHIJ";
367       String result = CallingContext.wrapText(text, 5, "");
368 
369       // Remove line separators to get all characters
370       String allChars = result.replace(System.lineSeparator(), "");
371       assertEquals(text, allChars, "All characters should be preserved after force-break wrapping");
372     }
373 
374     @Test
375     @DisplayName("force-break produces correct line lengths")
376     void forceBreakProducesCorrectLineLengths() {
377       String text = "ABCDEFGHIJKLMNO"; // 15 chars
378       String result = CallingContext.wrapText(text, 5, "");
379 
380       String[] lines = result.split(System.lineSeparator());
381       assertEquals(3, lines.length, "Should produce 3 lines of 5 chars each");
382       assertEquals("ABCDE", lines[0]);
383       assertEquals("FGHIJ", lines[1]);
384       assertEquals("KLMNO", lines[2]);
385     }
386 
387     @Test
388     @DisplayName("throws IllegalArgumentException when maxWidth is zero")
389     void throwsWhenMaxWidthIsZero() {
390       assertThrows(IllegalArgumentException.class,
391           () -> CallingContext.wrapText("test", 0, ""));
392     }
393 
394     @Test
395     @DisplayName("throws IllegalArgumentException when maxWidth is negative")

408     @Test
409     @DisplayName("throws IllegalArgumentException when indent > maxWidth")
410     void throwsWhenIndentGreaterThanMaxWidth() {
411       assertThrows(IllegalArgumentException.class,
412           () -> CallingContext.wrapText("test", 5, "      ")); // indent length > maxWidth
413     }

415     @Test
416     @DisplayName("handles very narrow width with word boundaries")
417     void handlesVeryNarrowWidthWithWordBoundaries() {
418       String text = "A B C D E";
419       String result = CallingContext.wrapText(text, 3, "");
